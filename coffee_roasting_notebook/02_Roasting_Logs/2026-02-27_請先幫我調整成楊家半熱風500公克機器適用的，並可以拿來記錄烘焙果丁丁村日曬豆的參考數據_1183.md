function getTargetProfile() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName("GoldenBatch");
  if (!sheet || sheet.getLastRow() < 2) return [];
  return sheet.getRange(2, 1, sheet.getLastRow() - 1, 2).getValues();
}

function recordFullBatch(data) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName("Roast_History");
  
  let advice = [];
  if (parseFloat(data.dtr) < 15) advice.push("⚠️ 發展期過短：果酸可能偏尖銳。");
  if (data.dropTemp > 204) advice.push("🚫 風味警示：果丁丁的花香可能已流失。");
  if (data.isFlick) advice.push("🔥 偵測到末段 Flick：焦糖化反應過劇。");

  const finalAdvice = advice.length > 0 ? advice.join(" ") : "✨ 完美軟著陸：花香與果質兼具！";
  sheet.appendRow([new Date(), data.batchName, data.totalTime, data.dtr + "%", data.weightLoss + "%", data.agtron, data.acidity, data.sweetness, finalAdvice]);
  return finalAdvice;
}

function createRoastReport(reportData, chartBase64) {
  const fileName = `RoastReport_${reportData.batchName}.pdf`;
  const html = `<html><body style="font-family: Arial;"><h1>SCA Roast Report: ${reportData.batchName}</h1><hr><p>DTR: ${reportData.dtr}% | Total Time: ${reportData.totalTime}</p><img src="${chartBase64}" style="width:100%;"><p>AI Advice: ${reportData.advice}</p></body></html>`;
  const blob = Utilities.newBlob(html, 'text/html', 'temp.html');
  const pdf = blob.getAs('application/pdf').setName(fileName);
  return DriveApp.getRootFolder().createFile(pdf).getUrl();
}

function record() {
  if(!startTime) startTime = Date.now();
  const t = parseFloat(document.getElementById('tempIn').value);
  const m = (Date.now() - startTime)/60000;

  if(history.length>0){

  document.getElementById('tempIn').value = '';

### 👨‍🏫 講師操作說明 (楊家機 + 果丁丁日曬)

1.  **為什麼要標註一爆校正？**
果丁丁豆小且密，探針反應慢的人可能 198 度才聽到爆聲，反應快的人 190 度就聽到了。
當你按下 **「標註一爆並校正」** 時，灰色的參考線會自動根據你當下的溫度「跳」到正確的位置。這能讓接下來發展期的 1 分半鐘內，你依然有準確的目標可以追隨。

2. 
**楊家半熱風機的操作建議**：

- **轉黃後 (約 150℃)**：將楊家機的風門拉大（例如 3 檔調至 4 檔），日曬豆需要更多對流熱來帶走糖分焦糖化產生的煙燻味。
- **一爆後**：果丁丁日曬豆蓄熱極強。請在一爆聲響起時，立刻將瓦斯調至最小，甚至關閉。靠楊家機厚實的鑄鐵筒餘溫，就能完成最後 1 分半的「軟著陸」滑行。
3. 
**PDF 報告的應用**：
這份報告會幫你記錄 DTR 與 AI 診斷。如果杯測後覺得「果酸不夠明亮」，請看報告中的 $\Delta T$。如果實際曲線一直高於黃金曲線，下次請提前 30 秒降火。
