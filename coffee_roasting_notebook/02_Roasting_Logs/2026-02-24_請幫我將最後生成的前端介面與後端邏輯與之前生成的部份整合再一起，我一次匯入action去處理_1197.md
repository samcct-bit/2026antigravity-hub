// 1. 取得黃金曲線 (由 GoldenBatch 工作表讀取)
function getTargetProfile() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName("GoldenBatch");
  if (!sheet) return []; 
  // 假設 A 欄是時間(分)，B 欄是溫度
  return sheet.getRange(2, 1, sheet.getLastRow() - 1, 2).getValues();
}

// 2. AI 診斷邏輯引擎
function getAIAdvice(data) {
  let advice = [];
  let dtr = parseFloat(data.dtr);
  let wl = parseFloat(data.weightLoss);
  
  if (dtr < 15) advice.push("⚠️ 發展期不足：可能帶有草本味。建議下一鍋稍微降低一爆前的 RoR。");
  if (dtr > 25) advice.push("⚠️ 發展期過長：風味可能趨於平淡（Baked）。建議提高一爆時的動能。");
  if (wl > 16) advice.push("🔍 失重偏高：檢查是否入豆溫過高導致豆表脫水過快。");
  if (data.flickDetected) advice.push("🚫 偵測到 Flick：一爆末期熱能失控，建議提早 30-45 秒降火。");
  
  return advice.length > 0 ? advice.join(" ") : "✨ 曲線控制精準，重現性極佳！";
}

// 3. 儲存完整紀錄並產出建議
function recordFullBatch(data) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName("Roast_History") || ss.insertSheet("Roast_History");
  
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["日期", "批次名稱", "總時長", "DTR%", "失重率", "Agtron", "酸質", "甜感", "AI建議"]);
  }
  
  const advice = getAIAdvice(data);
  sheet.appendRow([
    new Date(), data.batchName, data.totalTime, data.dtr + "%", 
    data.weightLoss + "%", data.agtron, data.acidity, data.sweetness, advice
  ]);
  return advice;
}

// 4. PDF 報表生成
function createRoastReport(reportData, chartBase64) {
  const fileName = `RoastReport_${reportData.batchName}_${new Date().getTime()}.pdf`;
  const htmlContent = `
    <html>
      <body style="font-family: Arial, sans-serif; padding: 30px;">
        <h1 style="color: #d9534f; border-bottom: 2px solid #333;">SCA Pro Roast Report</h1>
        <p><strong>Batch:</strong> ${reportData.batchName} | <strong>Date:</strong> ${new Date().toLocaleDateString()}</p>
        <hr>
        <h3>Physical Data</h3>
        <ul>
          <li>Total Time: ${reportData.totalTime} | DTR: ${reportData.dtr}%</li>
          <li>Weight Loss: ${reportData.weightLoss}% | Agtron: ${reportData.agtron}</li>
        </ul>
        <h3>Profile Chart</h3>
        <img src="${chartBase64}" style="width: 100%; border: 1px solid #ccc;" />
        <div style="margin-top:20px; padding:15px; background:#f4f4f4;">
          <h3>AI Diagnosis & Cupping Notes</h3>
          <p><strong>Sensory:</strong> Acidity ${reportData.acidity}/10, Sweetness ${reportData.sweetness}/10</p>
          <p><strong>Advice:</strong> ${reportData.advice}</p>
        </div>
      </body>
    </html>
  `;
  const blob = Utilities.newBlob(htmlContent, 'text/html', 'temp.html');
  const pdfBlob = blob.getAs('application/pdf').setName(fileName);
  const file = DriveApp.getRootFolder().createFile(pdfBlob);
  return file.getUrl();
}

let chart;
let startTime, fcTime;
let currentData = [];
let targetData = [];
let lastRoR = 0;
let flickFlag = false;

function initChart() {
  const ctx = document.getElementById('roastChart').getContext('2d');
  chart = new Chart(ctx, {
    type: 'line',
    data: {
      labels: Array.from({length: 15}, (_, i) => i),

// 載入黃金曲線
google.script.run.withSuccessHandler(data => {
  targetData = data;
  chart.data.datasets[0].data = data.map(d => ({x: d[0], y: d[1]}));
  chart.update();
}).getTargetProfile();

function recordPoint() {
  if (!startTime) startTime = Date.now();
  const temp = parseFloat(document.getElementById('tempInput').value);
  const elapsed = (Date.now() - startTime) / 60000;
  
  // 計算 RoR
  let ror = 0;
  if (currentData.length > 0) {
    const prev = currentData[currentData.length - 1];
    ror = (temp - prev.y) / (elapsed - prev.x);
    if (elapsed > 8 && ror > lastRoR + 1.5) {
      document.getElementById('warningMsg').style.display = 'block';
      flickFlag = true;
    }
  }

  // 計算 Delta T
  const target = targetData.reduce((prev, curr) => Math.abs(curr[0]-elapsed) < Math.abs(prev[0]-elapsed) ? curr : prev, [0,0]);
  const delta = temp - target[1];
  document.getElementById('deltaDisplay').innerText = (delta > 0 ? '+' : '') + delta.toFixed(1) + ' °C';
  document.getElementById('rorDisplay').innerText = ror.toFixed(1);

  currentData.push({x: elapsed, y: temp});
  chart.data.datasets[1].data = currentData;
  chart.update();
  document.getElementById('tempInput').value = '';
  lastRoR = ror;
}

function markEvent(type) {
  if (type === 'FC') fcTime = (Date.now() - startTime) / 1000;
  alert(type + " 已標註");
}

function finishRoast() {
  document.getElementById('cuppingModal').style.display = 'block';
  const elapsedSec = (Date.now() - startTime) / 1000;
  const devTime = elapsedSec - fcTime;
  const dtr = ((devTime / elapsedSec) * 100).toFixed(1);
  
  const finalData = {
    batchName: document.getElementById('batchName').value || "Unnamed Batch",
    totalTime: Math.floor(elapsedSec/60) + ":" + Math.floor(elapsedSec%60),
    dtr: dtr,
    weightLoss: ((document.getElementById('greenW').value - document.getElementById('roastedW').value) / document.getElementById('greenW').value * 100).toFixed(1),
    agtron: Math.round(110 - (currentData[currentData.length-1].y - 190) * 2.5),
    acidity: document.getElementById('senseAcidity').value,
    sweetness: document.getElementById('senseSweet').value,
    flickDetected: flickFlag
  };

  google.script.run.withSuccessHandler(advice => {
    document.getElementById('aiAdvice').innerText = "AI 診斷：" + advice;
    window.lastReportData = {...finalData, advice: advice};
  }).recordFullData(finalData);
}

}

## 📽️ 教學：如何將此系統投入實戰？

### 1. 建立「黃金曲線」(GoldenBatch)

 在你的 Google Sheet 中，建立一個名為  GoldenBatch  的工作表：

- **A 欄**：時間 (0, 0.5, 1, 1.5...)
- **B 欄**：對應溫度 (160, 150, 155, 162...)
這將會作為背景的虛線，指引你的烘焙路徑。

### 2. SCA Professional 級別的診斷邏輯

這套系統已經寫入了進階的 **Flick 偵測**。在高級認證中，我們會關注一爆後 RoR 是否因放熱反應而失控。

- **診斷範例**：如果你的一爆發展時間（DTR%）低於 15%，AI 會在報表中提醒你可能會出現「草本味（Vegetal/Grassy）」，這是專業杯測中判定發展不足的關鍵詞。

### 3. PDF 報表的價值

每次烘焙完後點擊「匯出 PDF」，系統會自動將曲線截圖與你的感官評分整合。你可以將這些 PDF 存檔，建立屬於自己的 **「烘焙風味地圖」**。
