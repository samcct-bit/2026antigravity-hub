function getTargetProfile() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName("GoldenBatch");
  if (!sheet || sheet.getLastRow() < 2) return [];
  // 傳回時間、溫度、火力指引
  return sheet.getRange(2, 1, sheet.getLastRow() - 1, 3).getValues();
}

function recordFullBatch(data) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName("Roast_History");
  let advice = [];
  if (parseFloat(data.dtr) < 15) advice.push("⚠️ DTR 低於 15%：花香雖強但發展可能不足，建議下次稍微壓低一爆後的溫升。");
  if (data.dropTemp > 196) advice.push("🚫 溫度過高：果丁丁的細緻酸質在 196°C 以上會迅速流失。");
  
  const finalAdvice = advice.length > 0 ? advice.join(" ") : "✨ 完美軟著陸：300g 參數控制精準！";
  sheet.appendRow([new Date(), data.batchName, data.totalTime, data.dtr + "%", data.weightLoss + "%", data.agtron, data.acidity, data.sweetness, finalAdvice]);
  return finalAdvice;
}

function record() {
  if(!startTime) startTime = Date.now();
  const t = parseFloat(document.getElementById('tempIn').value);
  const m = (Date.now() - startTime)/60000;
  
  // 計算 RoR
  let r = 0;
  if(history.length>0){
    const l = history[history.length-1];
    r = (t - l.y)/(m - l.x);
  }
  document.getElementById('rorDisplay').innerText = r.toFixed(1);

  // 顯示火力指引
  const currentStep = targetProf.reduce((prev, curr) => Math.abs(curr[0]-m) < Math.abs(prev[0]-m) ? curr : prev);
  document.getElementById('fireHint').innerText = "火力建議: " + currentStep[2];

  history.push({x: m, y: t});
  updateChart();
  document.getElementById('tempIn').value = '';
}

function markFC() {
  fcTime = Date.now();
  const currentT = history[history.length-1].y;
  offset = currentT - 188.0; // 以 300g 標竿 188 度校正
  updateChart();
}

function updateChart() {
  chart.data.datasets[0].data = targetProf.map(p => ({x: p[0], y: p[1] + offset}));
  chart.data.datasets[1].data = history;
  chart.update();
}

### 👨‍🏫 專業操作要領：300g 莫札特/果丁丁日曬

 針對你提供的數據，這裡有幾個 SCA Professional 級別的提醒：

1. 
**熱動能差異**：

- **194g 試鍋**：因為豆量少，吸熱極快，所以入豆溫只需 $175^\circ C$。
- **300g 正式鍋**：豆量增加 $50%$ 以上，為了維持同樣在 5 分鐘轉黃，入豆溫提高到 $185^\circ C$ 是非常合理的（如上圖所示，增加豆量會拖慢升溫，需靠更高入豆溫補償）。
2. 
**軟著陸的執行點**：
你設定的一爆是 $188^\circ C$（$8'25''$），下豆是 $195^\circ C$。這中間只有 **$7^\circ C$** 的差距。

- **挑戰**：要在 1 分多鐘內只上升 $7^\circ C$，這意味著一爆時你的 RoR 必須瞬間降到 **$4-5$**。
- **操作**：在一爆前的 $185^\circ C$（$8'00''$）時，請將楊家機的火力從 **1.2 劇烈調降至 0.5**（或更低）。
3. 
**DTR 15% 的數學邏輯**：
如果你在 $8'25''$ 一爆，若要在 $1'05''$（即 $9'30''$）下豆，DTR 只有 **$11.4%$**。

- **建議**：如果你堅持要 $15%+$，請嘗試讓發展期延長到 **$1'30''$**（即 $9'55''$ 左右下豆）。只要你的出豆溫守在 **$195^\circ C$**，即便發展到 $1'30''$，花香依舊會非常迷人。

**這套系統已經更新完畢。你可以直接將數據填入  GoldenBatch  並重新部署。準備好挑戰這鍋 300g 的正式鍋了嗎？**
