#### 📌 關鍵操作解讀 (Tank 200 實戰)

1.  **一爆前的「主動降速」 (Pre-FC Deceleration)**：

- 你在 $6'30''$ 到 $7'30''$ 之間（大約 $185^\circ C$），就要開始大幅下修火力。
- **目標**：讓 RoR 在進入一爆瞬間不是 $12$ 或 $15$，而是壓在 $7-8$。
2. 
**一爆後的「滑行模式」 (The Glide)**：

- 一爆發生時，豆體會釋放能量，此時你甚至可以暫時關小火（或利用 Tank 200 的風門拉大）。
- 讓 RoR 維持在 $3$ 到 $5$ 之間緩慢滑行。
- **重點**：因為 RoR 只有 $3-5$，你在發展期的這 $1'30''$ 內，溫度總共只會上升 $4-7^\circ C$。這就是為什麼你能達到 $15%$ 以上的 DTR，但出豆溫卻能壓在 $200^\circ C$ 左右，保住極致花香。

 function doGet() {
  return HtmlService.createHtmlOutputFromFile('index')

function recordFullData(data) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName("烘焙紀錄") || ss.insertSheet("烘焙紀錄");
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["日期", "名稱", "DTR%", "失重", "Agtron", "建議"]);
  }
  const advice = generateAdvice(data);
  sheet.appendRow([new Date(), data.name, data.dtr, data.wl, data.agtron, advice]);
  return advice;
}

function generateAdvice(d) {
  if (d.dtr < 15) return "發展不足，建議延長一爆後時間。";
  if (d.temp > 205 && d.isGesha) return "出豆溫過高，花香易流失，請嘗試軟著陸。";
  return "曲線優良，請維持！";
}

function createPDF(data, chartImg) {
  const fileName = `Report_${data.name}.pdf`;
  const html = `<h1>SCA Roast Report: ${data.name}</h1><img src="${chartImg}" style="width:100%"><p>DTR: ${data.dtr}%</p><p>Notes: ${data.advice}</p>`;
  const blob = Utilities.newBlob(html, 'text/html', 'temp.html');
  const pdf = blob.getAs('application/pdf').setName(fileName);
  return DriveApp.getRootFolder().createFile(pdf).getUrl();
}

let startTime, fcTime, chart;
let history = [];

const ctx = document.getElementById('mainChart').getContext('2d');
chart = new Chart(ctx, {
  type: 'line',
  data: { labels: [], datasets: [{ label: '豆溫', data: [], borderColor: 'red' }] }
});

function addData() {
  if(!startTime) startTime = Date.now();
  const temp = parseFloat(document.getElementById('tempInput').value);
  const time = (Date.now() - startTime) / 60000;
  
  let ror = 0;
  if(history.length > 0) {
    const prev = history[history.length - 1];
    ror = (temp - prev.y) / (time - prev.x);
  }
  
  document.getElementById('rorValue').innerText = ror.toFixed(1);
  history.push({x: time, y: temp});
  chart.data.labels.push(time.toFixed(1));
  chart.data.datasets[0].data.push(temp);
  chart.update();
  document.getElementById('tempInput').value = '';
}

function mark(ev) { if(ev === 'FC') fcTime = Date.now(); }

function finish() {
  const total = (Date.now() - startTime) / 1000;
  const dev = (Date.now() - fcTime) / 1000;
  const dtr = ((dev/total)*100).toFixed(1);
  const data = { name: "Mozart_Test", dtr: dtr, isGesha: true, temp: history[history.length-1].y };
  
  google.script.run.withSuccessHandler(adv => {

  }).recordFullData(data);
}

### 👨‍🏫 導師的最後叮嚀：關於莫札特豆

 莫札特這類豆子（葡萄乾蜜處理/厭氧），其梅納反應的速率比一般豆子快得多。

- **操作要領**：當你看到豆子顏色開始轉黃（約 $150-160^\circ C$），請提前縮小火力。
- **DTR 的意義**：如果一爆後的溫升太快，那 $15%$ 的 DTR 會變成災難（焦苦）；如果一爆後的溫升極其緩慢（軟著陸），那 $15%$ 的 DTR 就是極致的花香與平衡感。
