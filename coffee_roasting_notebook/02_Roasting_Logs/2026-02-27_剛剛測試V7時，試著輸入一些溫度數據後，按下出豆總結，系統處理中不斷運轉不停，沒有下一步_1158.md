# ☕ 烘焙日誌：剛剛測試V7時，試著輸入一些溫度數據後，按下出豆總結，系統處理中不斷運轉不停，沒有下一步

## 📊 對話烘焙紀錄數據
看到這個「無限轉圈圈」的狀況，我深感抱歉。身為導師，我判定這是程式碼中的一個**變數名稱不一致**（邏輯錯誤）加上**缺乏錯誤攔截機制**所導致的。

在 V7 版本中，前端將數據存入  window.finalData ，但呼叫後端時卻使用了  window.res ，這導致 Google 伺服器收到了空的資料而發生錯誤，進而讓「處理中」的畫面永遠無法消失。

以下是 **v7.1「防呆穩定版」**，我修正了變數邏輯，並加入了**錯誤回傳機制**，確保即使發生問題也會跳出提示，而不是死當。

let chart, startTime, fcTime, history=[], rorHistory=[], timerInt;

  const ctx = document.getElementById('mainChart').getContext('2d');

function record() {
  if (!startTime) {
    startTime = Date.now();
    timerInt = setInterval(() => {
      const sec = Math.floor((Date.now() - startTime) / 1000);
      document.getElementById('timer').innerText = Math.floor(sec/60).toString().padStart(2,'0') + ":" + (sec%60).toString().padStart(2,'0');
    }, 1000);
  }
  const t = parseFloat(document.getElementById('tempIn').value);
  const m = (Date.now() - startTime)/60000;
  let r = 0;
  if(history.length > 0) {
    const last = history[history.length-1];
    r = (t - last.y)/(m - last.x);
    if(r > 40 || r < 0) r = 0;
  }
  document.getElementById('rorNum').innerText = r.toFixed(1);
  history.push({x: m, y: t});
  rorHistory.push({x: m, y: r});
  
  chart.data.datasets[0].data = history;
  chart.data.datasets[1].data = rorHistory;
  chart.update();
  document.getElementById('tempIn').value = '';
}

function markFC() { fcTime = Date.now(); alert("一爆已標記"); }

function finish() {
  if (history.length === 0) return alert("請先輸入數據");
  clearInterval(timerInt);
  document.getElementById('loader').style.display = 'flex';
  document.getElementById('finishBtn').disabled = true;

  const total = (Date.now() - startTime)/1000;
  const dtr = fcTime ? (((Date.now() - fcTime)/1000)/total*100).toFixed(1) : 0;
  
  // 修正關鍵：變數名稱統一為 payload
  const payload = { 
    batchName: "Gotiti_300g", 
    totalTime: document.getElementById('timer').innerText, 
    dtr: dtr, 
    dropTemp: history[history.length-1].y 
  };
  window.finalReportData = payload;

  google.script.run
    .withSuccessHandler(res => {
      document.getElementById('loader').style.display = 'none';
      document.getElementById('reportArea').style.display = 'block';
      document.getElementById('aiAdvice').innerHTML = `<strong>AI 建議:</strong> ${res.advice}`;
      window.finalReportData.advice = res.advice;
    })
    .withFailureHandler(err => {
      document.getElementById('loader').style.display = 'none';
      alert("儲存失敗: " + err.message);
      document.getElementById('finishBtn').disabled = false;
    })
    .recordFullBatch(payload); // 這裡必須與上面定義的名稱一致
}

function getPDF() {
  const btn = document.getElementById('pdfBtn');
  btn.disabled = true;
  document.getElementById('linkOutput').innerText = "雲端處理中...";
  const img = document.getElementById('mainChart').toDataURL('image/png');

      document.getElementById('linkOutput').innerHTML = `<a href="${url}" target="_blank" class="btn btn-warning w-100 mt-2">📥 下載 PDF 報告</a>`;
      btn.disabled = false;
    })
    .withFailureHandler(err => {
      alert("PDF 生成失敗: " + err.message);
      btn.disabled = false;
    })
    .createRoastReport(window.finalReportData, img);
}

function recordFullBatch(data) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName("Roast_History") || ss.insertSheet("Roast_History");
    const advice = (parseFloat(data.dtr) < 15) ?   。";
    
    sheet.appendRow([new Date(), data.batchName, data.totalTime, data.dtr + "%", data.dropTemp, advice]);
    return { advice: advice };
  } catch (e) {
    throw new Error("試算表寫入失敗: " + e.message);
  }
}

function createRoastReport(data, chartBase64) {
  try {
    const html = `<html><body style="font-family: Arial;">
                  <h1>烘焙報告: ${data.batchName}</h1><hr>
                  <p>總時間: ${data.totalTime} | DTR: ${data.dtr}% | 下豆溫: ${data.dropTemp}°C</p>
                  <img src="${chartBase64}" style="width:100%"><p>建議: ${data.advice}</p>
                  </body></html>`;
    const blob = Utilities.newBlob(html, 'text/html', 'report.html');
    const pdf = blob.getAs('application/pdf').setName(`Report_${data.batchName}.pdf`);
    const file = DriveApp.getRootFolder().createFile(pdf);
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    return file.getDownloadUrl();
  } catch (e) {
    throw new Error("PDF 生成失敗: " + e.message);
  }
}

### 👨‍🏫 導師的操作建議：如何確保 RoR 曲線出現？

1. **RoR 曲線邏輯**：在 v7.1 中，我將 RoR 設定在右側 Y 軸（範圍 0-30）。當你輸入第二個點時，藍色虛線就會出現在畫面上。
2. 。我在 v7.1 加入了  finishBtn.disabled = true ，一旦按下「出豆總結」，按鈕就會鎖死直到處理完成，這能防止你因為心急多按幾次而產生的重複紀錄。
3. **PDF 按鈕**：按下「下載 PDF」後，請等待 3-5 秒，下方會出現一個黃色按鈕。**請長按該按鈕並選擇「在新分頁中開啟」**，這是目前手機端最穩定的下載方式。

 **這套 v7.1 已經修正了「無限轉圈圈」的 Bug。請務必重新部署（部署 > 管理部署 > 新版本），並在測試時確認畫面右側是否出現了藍色標尺。準備好再試一鍋果丁丁了嗎？**
