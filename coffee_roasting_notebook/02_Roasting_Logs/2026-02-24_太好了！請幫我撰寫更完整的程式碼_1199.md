// 取得黃金曲線數據 (假設存放在名為 "GoldenBatch" 的工作表)
function getTargetProfile() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName("GoldenBatch");
  if (!sheet) return [];
  // 取得時間(A欄)與溫度(B欄)
  return sheet.getRange(2, 1, sheet.getLastRow() - 1, 2).getValues();
}

function saveFullBatch(data) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName("Roast_History") || ss.insertSheet("Roast_History");
  sheet.appendRow([
    new Date(), data.batchName, data.totalTime, data.dtr, 
    data.weightLoss, data.agtron, data.notes
  ]);
}

function handleEnter(e) {
  if (e.key === 'Enter') {
    const temp = parseFloat(document.getElementById('tempInput').value);
    recordPoint(temp);
    document.getElementById('tempInput').value = '';
  }
}

function recordPoint(temp) {
  if (!roastStartTime) roastStartTime = Date.now();
  const elapsed = (Date.now() - roastStartTime) / 60000;
  
  // 計算 RoR (每分鐘升溫)
  let ror = 0;
  if (currentData.length > 0) {
    const prev = currentData[currentData.length - 1];
    ror = (temp - prev.y) / (elapsed - prev.x);
    
    // Flick & Crash 偵測邏輯
    if (elapsed > 8 && ror > lastRoR + 2) { // 假設 8 分鐘後進入敏感期
      document.getElementById('warningMsg').style.display = 'block';
    } else {
      document.getElementById('warningMsg').style.display = 'none';
    }
  }

  // 計算 Delta T (與目標的差距)
  const targetAtTime = findTargetTemp(elapsed);
  const delta = temp - targetAtTime;
  const deltaEl = document.getElementById('tempDelta');
  deltaEl.innerText = `${delta > 0 ? '+' : ''}${delta.toFixed(1)} °C`;
  deltaEl.className = delta > 0 ? 'delta-positive' : 'delta-negative';

  currentData.push({x: elapsed, y: temp});
  chart.data.datasets[1].data = currentData;
  chart.update();
  
  document.getElementById('currentRoR').innerText = ror.toFixed(1);
  lastRoR = ror;
}

function findTargetTemp(time) {
  // 簡單線性插值查找目標溫度
  if (targetData.length === 0) return 0;
  // 此處省略精確插值邏輯，直接取最近點
  return targetData.reduce((prev, curr) => 
    Math.abs(curr[0] - time) < Math.abs(prev[0] - time) ? curr : prev)[1];
}

## 🎓 SCA Professional 級別的技術要點 (Training Notes)

 拿到中級證照後，你現在的訓練重點應放在 **「熱動能管理」**：

### 1. 預判性的火力調控

在專業實作中，我們不應該等到 RoR 下降才加火。

- **公式參考**：$\Delta RoR = f(Gas, Airflow, Load)$。
- 使用此程式時，觀察 $\Delta T$。如果 $\Delta T$ 開始趨向負值，代表你的熱能供給不足以維持目標曲線，應提早 30 秒調整火力。

### 2. Flick & Crash 的化學影響

- **Crash (一爆失速)**：若 RoR 掉得太快，會導致梅納反應中斷，咖啡容易出現「生澀味 (Underdeveloped)」或「穀物感」。
- **Flick (尾段突升)**：通常是因為一爆產生的熱量無法及時排出，導致豆表迅速焦糖化甚至碳化。這在高級感官分析中會被判定為「燥感 (Harshness)」。

### 3. Agtron 值預測與重現性

在你的 GAS 試算表中，建議增加一個 **「重現性評分」** 欄位。

- 計算方法：$\sum |T_{actual} - T_{target}|$。
- 這個數值越小，代表你的烘焙重現性越高。在 SCA 高級考試中，這直接決定了你是否具備生產端 QC 的能力。
