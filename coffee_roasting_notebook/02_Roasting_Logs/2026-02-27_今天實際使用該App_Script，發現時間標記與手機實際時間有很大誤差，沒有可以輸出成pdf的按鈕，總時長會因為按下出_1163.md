1. **時間誤差與計時邏輯**：不再單純依賴  Date.now() ，而是引入「**手動停止點**」，按下出豆時立刻鎖定時間戳記，避免後續網路延遲影響總時長。
2. **PDF 按鈕消失**：這是因為在手機小螢幕上，原本的隱藏面版（Modal）可能被遮擋。我將 PDF 功能直接整合在主畫面顯眼處。

function getTargetProfile() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName("GoldenBatch");
  return (sheet && sheet.getLastRow() > 1) ? sheet.getRange(2, 1, sheet.getLastRow() - 1, 3).getValues() : [];
}

function recordFullBatch(data) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName("Roast_History") || ss.insertSheet("Roast_History");
  
  // 強制設定時區為台北
  const now = Utilities.formatDate(new Date(), "GMT+8", "yyyy/MM/dd HH:mm:ss");
  
  let advice = [];
  if (parseFloat(data.dtr) < 15) advice.push("⚠️ DTR 偏低(未滿15%)。");
  if (data.dropTemp > 196) advice.push("🚫 溫度過高影響花香。");
  
  const finalAdvice = advice.length > 0 ? advice.join(" ") : "✨ 完美軟著陸！";
  sheet.appendRow([now, data.batchName, data.totalTime, data.dtr + "%", data.weightLoss + "%", data.dropTemp, finalAdvice]);
  return {advice: finalAdvice, status: "Success"};
}

// PDF 生成邏輯 (優化手機端連結開啟)
function createRoastReport(reportData, chartBase64) {
  const html = `<html><body style="font-family: Arial; padding: 20px;">
    <h1>SCA Roast Report: ${reportData.batchName}</h1><hr>
    <p>日期: ${Utilities.formatDate(new Date(), "GMT+8", "yyyy/MM/dd")}</p>
    <p>DTR: ${reportData.dtr}% | 總時長: ${reportData.totalTime} | 失重: ${reportData.weightLoss}%</p>
    <img src="${chartBase64}" style="width: 100%; border: 1px solid #ccc;">
    <div style="background:#f4f4f4; padding:15px; margin-top:20px;"><h3>AI 建議</h3><p>${reportData.advice}</p></div>
    </body></html>`;
  const blob = Utilities.newBlob(html, 'text/html', 'temp.html');
  const pdf = blob.getAs('application/pdf').setName(`Report_${reportData.batchName}.pdf`);
  return DriveApp.getRootFolder().createFile(pdf).getDownloadUrl().replace("?e=download&gd=true","");
}

<script>
let chart, startTime, fcTime, endTime, history=[], targetProf=[], offset=0, timerInterval;

function startTimer() {
  if (timerInterval) return;
  startTime = Date.now();
  timerInterval = setInterval(() => {
    const s = Math.floor((Date.now() - startTime) / 1000);
    document.getElementById('timerDisplay').innerText = Math.floor(s/60).toString().padStart(2,'0') + ":" + (s%60).toString().padStart(2,'0');
  }, 1000);
}

function initChart() {
  chart = new Chart(document.getElementById('roastChart').getContext('2d'), {
    type: 'line',
    data: { labels: Array.from({length: 24}, (_, i) => (i*0.5).toFixed(1)),
            datasets: [{ label: '目標', data: [], borderColor: 'gray', borderDash:[5,5], pointRadius:0 },
                       { label: '實測', data: [], borderColor: '#ff4d4d', tension:0.3 }] },
    options: { animation: false, scales: { y: { min: 90, max: 210 } } }
  });
}

google.script.run.withSuccessHandler(data => { targetProf = data; updateChart(); }).getTargetProfile();

function record() {
  if (!startTime) startTimer();
  const t = parseFloat(document.getElementById('tempIn').value);
  const m = (Date.now() - startTime)/60000;
  let r = 0;
  if(history.length > 0) {
    const last = history[history.length - 1];
    r = (t - last.y) / (m - last.x);
    if (r > 40) r = 0; // 過濾輸入噪音
  }
  document.getElementById('rorDisplay').innerText = r.toFixed(1);
  history.push({x: m, y: t});
  updateChart();
  document.getElementById('tempIn').value = '';
}

function markFC() { fcTime = Date.now(); offset = history[history.length-1].y - 188.0; updateChart(); }
function updateChart() {
  chart.data.datasets[0].data = targetProf.map(p => ({x: p[0], y: p[1] + offset}));
  chart.data.datasets[1].data = history;
  chart.update();
}

function finish() {
  endTime = Date.now(); // 立刻鎖定出豆時間
  clearInterval(timerInterval);
  document.getElementById('loader').style.display = 'flex';
  
  const totalSec = (endTime - startTime) / 1000;
  const devSec = (endTime - fcTime) / 1000;
  const dtr = ((devSec / totalSec) * 100).toFixed(1);
  
  window.res = {
    batchName: "Gotiti_300g",
    totalTime: document.getElementById('timerDisplay').innerText,
    dtr: dtr,
    dropTemp: history[history.length-1].y,
    weightLoss: 13.5
  };

  google.script.run.withSuccessHandler(obj => {
    document.getElementById('loader').style.display = 'none';
    document.getElementById('resultPanel').style.display = 'block';
    document.getElementById('aiAdviceTitle').innerText = "AI 建議：" + obj.advice;
    window.res.advice = obj.advice;
  }).recordFullBatch(window.res);
}

function exportPDF() {
  const rw = document.getElementById('rw').value;
  window.res.weightLoss = ((300 - rw) / 300 * 100).toFixed(1);
  const chartImg = document.getElementById('roastChart').toDataURL('image/png');
  google.script.run.withSuccessHandler(url => window.open(url, '_blank')).createRoastReport(window.res, chartImg);
}

### 👨‍🏫 設置檢查清單（必做）

1. **時區修正**：在 Apps Script 畫面左側點擊 **「專案設定 (齒輪圖示)」**，勾選 **「在編輯器中顯示 appsscript.json 資訊清單檔案」**。回到編輯器打開  appsscript.json ，將  "timeZone": "America/New_York"  改為  "timeZone": "Asia/Taipei" 。
2. **重新部署**：每次修改代碼後，請點擊 **「部署 > 管理部署 > 編輯 (鉛筆) > 版本：新版本」**，這樣網頁才會生效。
3. **瀏覽器限制**：手機瀏覽器常會阻擋  window.open 。點擊下載 PDF 時，若沒反應，請檢查瀏覽器是否跳出「已封裝彈出視窗」的提示，並選擇「永遠允許」。

 **這套系統更新後，時間標記會由計時器自動跳動，不再受手機系統休眠干擾。AI 建議也會在點擊「出豆總結」後出現 Loader，避免你空等。快去測試下一鍋果丁丁吧！**
