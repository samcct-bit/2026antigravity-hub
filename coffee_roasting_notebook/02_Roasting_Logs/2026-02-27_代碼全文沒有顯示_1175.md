// 從試算表讀取黃金曲線
function getTargetProfile() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName("GoldenBatch");
  if (!sheet || sheet.getLastRow() < 2) return [];
  // 取得 A(時間), B(溫度), C(火力指引)
  return sheet.getRange(2, 1, sheet.getLastRow() - 1, 3).getValues();
}

// 儲存完整烘焙數據
function recordFullBatch(data) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName("Roast_History") || ss.insertSheet("Roast_History");
  
  // 如果是新工作表，寫入標題列
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["日期", "批次名稱", "總時長", "DTR%", "失重率", "環境溫度", "環境濕度", "氣壓", "下豆溫度", "AI建議"]);
  }
  
  let advice = [];
  if (parseFloat(data.dtr) < 15) advice.push("⚠️ DTR 偏低(未滿15%)，果酸可能較尖銳。");
  if (data.dropTemp > 196) advice.push("🚫 溫度過高：藝伎/果丁丁的花香極易在196°C以上轉為木質味。");
  if (data.weather && data.weather.temp > 30) advice.push("🌡️ 環境高溫：散熱較慢，下次建議提前5-10秒降火。");
  
  const finalAdvice = advice.length > 0 ? advice.join(" ") : "✨ 完美軟著陸！花香與酸質控制極佳。";
  
  sheet.appendRow([
    new Date(), 
    data.batchName, 
    data.totalTime, 
    data.dtr + "%", 
    data.weightLoss + "%", 
    data.weather.temp + "°C", 
    data.weather.humidity + "%", 
    data.weather.pressure + "hPa",
    data.dropTemp,
    finalAdvice
  ]);
  return finalAdvice;
}

// 生成 PDF 報告連結
function createRoastReport(reportData, chartBase64) {
  const fileName = `RoastReport_${reportData.batchName}_${new Date().getTime()}.pdf`;
  const html = `
    <html><body style="font-family: Arial, sans-serif; padding: 20px;">
      <h1 style="color: #d9534f; border-bottom: 2px solid #333;">SCA PRO ROAST REPORT</h1>
      <p><strong>批次:</strong> ${reportData.batchName} | <strong>日期:</strong> ${new Date().toLocaleDateString()}</p>
      <hr>
      <h3>物理數據</h3>
      <p>DTR: ${reportData.dtr}% | 失重率: ${reportData.weightLoss}% | 下豆溫: ${reportData.dropTemp}°C</p>
      <img src="${chartBase64}" style="width: 100%; border: 1px solid #ccc;" />
      <div style="margin-top:20px; padding:15px; background:#f4f4f4; border-left: 5px solid #d9534f;">
        <h3>AI 診斷與環境資訊</h3>
        <p>環境: ${reportData.weather.temp}°C / ${reportData.weather.humidity}%</p>
        <p><strong>建議:</strong> ${reportData.advice}</p>
      </div>
    </body></html>
  `;
  const blob = Utilities.newBlob(html, 'text/html', 'temp.html');
  const pdfBlob = blob.getAs('application/pdf').setName(fileName);
  const file = DriveApp.getRootFolder().createFile(pdfBlob);
  return file.getUrl();
}

<script>
let chart, startTime, fcTime, history=[], targetProf=[], offset=0, weatherData={};
let lastRoR = 0;

// *** ⚠️ 請在此填入你的 OPENWEATHER API KEY ***
const API_KEY = "你的_OPENWEATHER_API_KEY"; 
const CITY = "Taipei"; 

// 1. 初始化氣象
function fetchWeather() {
  if(!API_KEY || API_KEY.includes("你的")) {
    document.getElementById('weatherBox').innerText = "⚠️ 請在代碼中填入 OpenWeather API Key";
    return;
  }
  fetch(`https://api.openweathermap.org/data/2.5/weather?q=${CITY}&appid=${API_KEY}&units=metric`)
    .then(r => r.json()).then(d => {
      weatherData = { temp: d.main.temp, humidity: d.main.humidity, pressure: d.main.pressure };
      document.getElementById('weatherBox').innerHTML = `🌡️ 環境：${weatherData.temp}°C | 💧 濕度：${weatherData.humidity}% | 🌀 氣壓：${weatherData.pressure} hPa`;
    });
}

// 2. 初始化圖表
function initChart() {
  const ctx = document.getElementById('roastChart').getContext('2d');
  chart = new Chart(ctx, {
    type: 'line',
    data: {
      labels: Array.from({length: 24}, (_, i) => (i*0.5).toFixed(1)),
      datasets: [
        { label: '黃金目標 (300g)', data: [], borderColor: 'rgba(255,255,255,0.2)', borderDash:[5,5], pointRadius:0 },
        { label: '實測紀錄', data: [], borderColor: '#ff4d4d', tension:0.3 }
      ]
    },
    options: { scales: { y: { min: 90, max: 205 } } }
  });
}

// 從後端抓取黃金數據
google.script.run.withSuccessHandler(data => {
  targetProf = data;
  updateChart();
}).getTargetProfile();

// 3. 記錄溫度點 (平滑化 RoR 邏輯)
function recordPoint() {
  if(!startTime) { startTime = Date.now(); fetchWeather(); }
  const temp = parseFloat(document.getElementById('tempInput').value);
  if(isNaN(temp)) return;
  const elapsed = (Date.now() - startTime) / 60000;
  
  let ror = 0;
  if(history.length > 0) {
    // 平滑化：尋找 30 秒前的點作對比
    const lookbackTime = elapsed - 0.5;
    let refPoint = history.find(p => p.x >= lookbackTime) || history[0];
    ror = (temp - refPoint.y) / (elapsed - refPoint.x);
    if(ror > 40) ror = lastRoR; // 過濾輸入噪音
  }

  document.getElementById('rorDisplay').innerText = ror.toFixed(1);
  lastRoR = ror;

  // 顯示當前階段火力建議
  const step = targetProf.reduce((prev, curr) => Math.abs(curr[0]-elapsed) < Math.abs(prev[0]-elapsed) ? curr : prev, [0,0,"無"]);
  document.getElementById('fireHint').innerText = "火力建議: " + step[2];

  history.push({x: elapsed, y: temp});
  updateChart();
  document.getElementById('tempInput').value = '';
}

function markEvent(type) {
  if(type==='FC') {
    fcTime = Date.now();
    // 自動根據標竿 188 度進行曲線平移校正
    offset = history[history.length-1].y - 188.0;
    updateChart();
    alert("一爆標註成功！已根據探針差異進行曲線校正。");
  }
}

function updateChart() {
  chart.data.datasets[0].data = targetProf.map(p => ({x: p[0], y: p[1] + offset}));
  chart.data.datasets[1].data = history;
  chart.update();
}

function finishRoast() {
  const totalSec = (Date.now() - startTime) / 1000;
  const devSec = (Date.now() - fcTime) / 1000;
  const dtr = ((devSec / totalSec) * 100).toFixed(1);
  
  window.finalResults = {
    batchName: document.getElementById('batchName').value,
    totalTime: Math.floor(totalSec/60) + ":" + Math.floor(totalSec%60).toString().padStart(2,'0'),
    dtr: dtr,
    dropTemp: history[history.length-1].y,
    weather: weatherData
  };

  document.getElementById('cuppingPanel').style.display = 'block';
}

function exportPDF() {
  const rw = document.getElementById('roastedWeight').value;
  const wl = ((300 - rw) / 300 * 100).toFixed(1); // 假設生豆 300g
  const data = { ...window.finalResults, weightLoss: wl, acidity: document.getElementById('senseAcid').value };

  google.script.run.withSuccessHandler(advice => {
    document.getElementById('aiResult').innerText = "AI 診斷完成：" + advice;
    const chartImg = document.getElementById('roastChart').toDataURL('image/png');
    google.script.run.withSuccessHandler(url => window.open(url, '_blank')).createRoastReport({...data, advice: advice}, chartImg);
  }).recordFullBatch(data);
}
