 // 新增：根據數據提供診斷建議
function getAIAdvice(data) {
  let advice = [];
  let dtr = parseFloat(data.dtr);
  let wl = parseFloat(data.weightLoss);
  let acidity = parseInt(data.acidity);

  if (dtr < 15) advice.push("⚠️ 發展期過短：可能導致草本味或尖銳酸質。建議下一鍋稍微降低一爆前的火力，拉長發展時間。");
  if (dtr > 25) advice.push("⚠️ 發展期過長：可能導致風味平淡（Baked）。建議提高一爆時的動能，讓風味更明亮。");
  if (wl > 16 && dtr < 20) advice.push("🔍 失重偏高：檢查是否入豆溫過高或排風太大，導致豆表脫水過快。");
  if (acidity < 5 && dtr < 18) advice.push("💡 提示：若酸質不足且 DTR 低，可能是轉黃點拖太久，建議增加脫水期的熱能。");
  
  return advice.length > 0 ? advice.join("\n") : "✨ 曲線表現平衡！請維持此節奏，細微調整火力即可。";
}

// 修改後的儲存函數，加入感官數據
function recordFullData(data) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("烘焙總結") || ss.insertSheet("烘焙總結");
  
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["日期", "總時長", "DTR%", "失重率", "預估Agtron", "酸質", "甜感", "苦味", "AI建議"]);
  }
  
  let advice = getAIAdvice(data);
  sheet.appendRow([
    new Date(), data.totalTime, data.dtr + "%", data.weightLoss + "%", 
    data.agtron, data.acidity, data.sweetness, data.bitterness, advice
  ]);
  return advice;
}

## 📖 專業訓練講義：如何「閱讀」烘焙曲線

 這張圖標誌了 SCA 認證中最重要的幾個數據維度，請學員配合程式進行對應：

### 📈 數據解讀三部曲

1. 
**RoR 趨勢線 (The Slope)**：

- **理想狀態**：持續下降（Constantly Declining）。
- **警訊**：若在一爆前 RoR 突然拉升（Flick），杯測通常會出現強烈的燥感與苦韻。
2. 
**DTR% 發展佔比計算**：
$$DTR% = \frac{\text{發展時間 (Development Time)}}{\text{總烘焙時間 (Total Time)}} \times 100%$$

- **淺焙 (SCA Sample Roast)**：建議 $15% - 18%$。
- **中焙**：建議 $18% - 22%$。
3. 
**失重率 (Loss Rate)**：
$$Loss% = \frac{\text{生豆重} - \text{熟豆重}}{\text{生豆重}} \times 100%$$

- 如果失重率很高（>16%）但顏色很淺，代表你可能用了「低溫長烘」的脫水策略，口感會較乾、body 較薄。

## 👨‍🏫 講師的最後建議

這套系統是你的**數位副駕駛**。當你在操作烘焙機時，請觀察計時器旁的 **RoR 即時數值**：

- 如果一爆前 RoR 高於 $10^\circ C/min$，請立刻調小火力。
- 如果一爆後 RoR 掉到 $2^\circ C/min$ 以下，請小心進入「悶烘（Stall）」，這會毀掉咖啡的酸質明亮度。
