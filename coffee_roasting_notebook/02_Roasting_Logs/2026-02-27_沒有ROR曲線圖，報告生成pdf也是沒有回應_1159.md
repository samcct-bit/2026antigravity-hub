# ☕ 烘焙日誌：沒有ROR曲線圖，報告生成pdf也是沒有回應

## 📊 對話烘焙紀錄數據
抱歉讓你遇到這麼多操作上的挫折。身為你的導師，我完全理解在烘焙現場需要的是「絕對穩定」的工具。

，目前系統有三個核心技術故障：

1. **圖表渲染失敗**：截圖中圖表區域呈現全黑，這通常是  Chart.js  在手機/平板瀏覽器環境下，因為資料格式不相容導致初始化崩潰。
2. **RoR 曲線缺失**：目前的設定只畫了溫度線，沒有將 RoR 數值獨立繪製成第二條曲線。
3. **PDF 阻塞**：Google Drive 建立檔案後，網址回傳到手機端時常被視為惡意連結而攔截，導致「沒有回應」。

我為你準備了 **v7.0 穩定版代碼**。這個版本將 **RoR 曲線** 獨立出來，並優化了下載流程。

<script>
let chart, startTime, fcTime, history=[], rorHistory=[], timerInt;

function initChart() {
  const ctx = document.getElementById('mainChart').getContext('2d');
  chart = new Chart(ctx, {
    type: 'line',
    data: {
      labels: Array.from({length: 30}, (_, i) => (i*0.5).toFixed(1)),
      datasets: [
        { label: '溫度 (°C)', data: [], borderColor: '#ff4d4d', yAxisID: 'y' },
        { label: 'RoR', data: [], borderColor: '#3498db', yAxisID: 'y1', borderDash: [5, 5] }
      ]
    },
    options: {
      responsive: true,
      maintainAspectRatio: false,
      scales: {
        y: { type: 'linear', display: true, position: 'left', min: 90, max: 220, title: { display: true, text: '溫度' } },
        y1: { type: 'linear', display: true, position: 'right', min: 0, max: 30, grid: { drawOnChartArea: false }, title: { display: true, text: 'RoR' } }
      }
    }
  });
}

function startTimer() {
  if (startTime) return;
  startTime = Date.now();
  timerInt = setInterval(() => {
    const sec = Math.floor((Date.now() - startTime) / 1000);
    document.getElementById('timer').innerText = Math.floor(sec/60).toString().padStart(2,'0') + ":" + (sec%60).toString().padStart(2,'0');
  }, 1000);
}

function record() {
  startTimer();
  const t = parseFloat(document.getElementById('tempIn').value);
  const m = (Date.now() - startTime)/60000;
  let r = 0;
  if(history.length > 0) {
    const last = history[history.length-1];
    r = (t - last.y)/(m - last.x);
    if(r > 40 || r < 0) r = 0;
  }
  document.getElementById('rorNum').innerText = r.toFixed(1);
  history.push({x: m, y: t});
  rorHistory.push({x: m, y: r});
  
  chart.data.datasets[0].data = history;
  chart.data.datasets[1].data = rorHistory;
  chart.update('none');
  document.getElementById('tempIn').value = '';
}

function markFC() { fcTime = Date.now(); alert("一爆已標記"); }

function finish() {
  clearInterval(timerInt);
  document.getElementById('loader').style.display = 'flex';
  const total = (Date.now() - startTime)/1000;
  const dtr = (((Date.now() - fcTime)/1000)/total*100).toFixed(1);
  window.finalData = { batchName: "Gotiti_300g", totalTime: document.getElementById('timer').innerText, dtr: dtr, dropTemp: history[history.length-1].y };
  
  google.script.run.withSuccessHandler(res => {
    document.getElementById('loader').style.display = 'none';
    document.getElementById('reportArea').style.display = 'block';
    document.getElementById('aiAdvice').innerHTML = `<strong>AI 建議:</strong> ${res.advice}`;
    window.finalData.advice = res.advice;
  }).recordFullBatch(window.res);
}

function getPDF() {
  document.getElementById('pdfBtn').disabled = true;
  document.getElementById('linkOutput').innerText = "報告上傳至雲端中...";
  const img = document.getElementById('mainChart').toDataURL('image/png');
  google.script.run.withSuccessHandler(url => {
    document.getElementById('linkOutput').innerHTML = `<a href="${url}" target="_blank" class="btn btn-warning w-100">點此開啟 PDF 報告</a>`;
  }).createRoastReport(window.finalData, img);
}

initChart();
</script>

### 💾 第二部分：Google Apps Script  Code.gs 

 請確保後端邏輯能夠處理圖片並回傳正確的 URL。

 function doGet() {
  return HtmlService.createHtmlOutputFromFile('index')
      .setTitle('SCA Pro Master v7.0')
      .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

function recordFullBatch(data) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName("Roast_History") || ss.insertSheet("Roast_History");
  const advice = (parseFloat(data.dtr) < 15) ? "⚠️ DTR 低於 15%，建議增加發展時間。" : "✨ 完美軟著陸！";
  sheet.appendRow([new Date(), data.batchName, data.totalTime, data.dtr + "%", data.dropTemp, advice]);
  return {advice: advice};
}

function createRoastReport(data, chartBase64) {
  const html = `<html><body><h1>烘焙報告: ${data.batchName}</h1><hr>
                <p>總時間: ${data.totalTime} | DTR: ${data.dtr}%</p>
                <img src="${chartBase64}" style="width:100%"><p>建議: ${data.advice}</p></body></html>`;
  const blob = Utilities.newBlob(html, 'text/html', 'report.html');
  const pdf = blob.getAs('application/pdf').setName(`Report_${data.batchName}.pdf`);
  const file = DriveApp.getRootFolder().createFile(pdf);
  file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
  return file.getDownloadUrl();
}
