 為了確保你的系統一次到位，我幫你把 **氣象資訊 (Weather)**、**300g 黃金數據**、**平滑化 RoR** 以及 **一爆校正** 全部整合好了。請直接替換你 Google Apps Script 內的兩個檔案：

#### 1. Google Apps Script 內的  Code.gs 

 function doGet() {
  return HtmlService.createHtmlOutputFromFile('index')
      .setTitle('SCA Pro Master v5.2')
      .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

function getTargetProfile() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName("GoldenBatch");
  if (!sheet || sheet.getLastRow() < 2) return [];
  return sheet.getRange(2, 1, sheet.getLastRow() - 1, 3).getValues();
}

function recordFullBatch(data) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName("Roast_History");
  let advice = [];
  if (parseFloat(data.dtr) < 15) advice.push("⚠️ DTR 偏低，建議增加軟著陸時長。");
  if (data.dropTemp > 196) advice.push("🚫 溫度過高，藝伎花香可能流失。");
  if (data.weather.temp > 30) advice.push("🌡️ 環境熱，注意機器散熱速度。");
  
  const finalAdvice = advice.length > 0 ? advice.join(" ") : "✨ 完美軟著陸！";
  sheet.appendRow([
    new Date(), data.batchName, data.totalTime, data.dtr + "%", data.weightLoss + "%", 
    data.weather.temp + "°C", data.weather.humidity + "%", data.weather.pressure + "hPa", finalAdvice
  ]);
  return finalAdvice;
}
  

#### 2. Google Apps Script 內的  index.html  (核心 UI)

 **請記得在代碼中替換你的 OpenWeather API Key。**

 <!DOCTYPE html>
<html>
<head>
  <script src="https://cdn.jsdelivr.net/npm/chart.js"></script>
  <link href="https://cdnjs.cloudflare.com/ajax/libs/bootstrap/5.1.3/css/bootstrap.min.css" rel="stylesheet">
  <style>

    #rorDisplay { font-size: 2.8rem; color: #ff4d4d; font-weight: bold; }
    .weather-text { color: #5cb85c; font-size: 0.9rem; }
  </style>
</head>
<body>
<div class="container-fluid">
  <div class="stat-card mb-2"><span id="weatherInfo" class="weather-text">正在同步氣象...</span></div>
  <div class="row">
    <div class="col-md-8"><div class="stat-card"><canvas id="roastChart"></canvas></div></div>
    <div class="col-md-4">
      <div class="stat-card text-center"><h6>即時 RoR</h6><div id="rorDisplay">0.0</div></div>
      <div id="fireHint" class="stat-card text-warning text-center">火力建議: 等待中</div>
      <input type="number" id="tempIn" class="form-control mb-2 bg-dark text-white" placeholder="輸入豆溫" onkeypress="if(event.key==='Enter') record()">
      <button class="btn btn-outline-warning w-100 mb-2" onclick="markFC()">標註一爆 & 校正</button>
      <button class="btn btn-success w-100" onclick="finish()">出豆總結</button>
    </div>
  </div>

<script>
let chart, startTime, fcTime, history=[], targetProf=[], offset=0, weatherData={};
const API_KEY = "你的_OPENWEATHER_API_KEY"; // *** 請在此填入你的 API KEY ***

function fetchWeather() {
  fetch(`https://api.openweathermap.org/data/2.5/weather?q=Taipei&appid=${API_KEY}&units=metric`)
    .then(r => r.json()).then(d => {
      weatherData = { temp: d.main.temp, humidity: d.main.humidity, pressure: d.main.pressure };
      document.getElementById('weatherInfo').innerText = `🌡️ ${weatherData.temp}°C | 💧 ${weatherData.humidity}% | 🌀 ${weatherData.pressure} hPa`;
    });
}

function initChart() {
  chart = new Chart(document.getElementById('roastChart').getContext('2d'), {
    type: 'line',
    data: {
      labels: Array.from({length: 24}, (_, i) => (i*0.5).toFixed(1)),
      datasets: [{ label: '黃金目標', data: [], borderColor: 'gray', borderDash:[5,5], pointRadius:0 },
                 { label: '實測', data: [], borderColor: '#ff4d4d', tension:0.3 }]
    },
    options: { scales: { y: { min: 90, max: 210 } } }
  });
}

google.script.run.withSuccessHandler(data => { targetProf = data; updateChart(); }).getTargetProfile();

function record() {
  if(!startTime) { startTime = Date.now(); fetchWeather(); }
  const t = parseFloat(document.getElementById('tempIn').value);
  const m = (Date.now() - startTime)/60000;
  
  let r = 0;
  if(history.length > 0) {
    const lookback = history.find(p => p.x >= m - 0.5) || history[0];
    r = (t - lookback.y) / (m - lookback.x);
    if (r > 40) r = 0; // 過濾輸入雜訊
  }
  
  document.getElementById('rorDisplay').innerText = r.toFixed(1);
  const hint = targetProf.reduce((prev, curr) => Math.abs(curr[0]-m) < Math.abs(prev[0]-m) ? curr : prev, [0,0,"無"]);
  document.getElementById('fireHint').innerText = "火力建議: " + hint[2];
  
  history.push({x: m, y: t});
  updateChart();
  document.getElementById('tempIn').value = '';
}

function markFC() { fcTime = Date.now(); offset = history[history.length-1].y - 188.0; updateChart(); }
function updateChart() {
  chart.data.datasets[0].data = targetProf.map(p => ({x: p[0], y: p[1] + offset}));
  chart.data.datasets[1].data = history;
  chart.update();
}

function finish() {
  const total = (Date.now() - startTime)/1000;
  const dtr = (((Date.now() - fcTime)/1000) / total * 100).toFixed(1);
  google.script.run.withSuccessHandler(adv => alert("烘焙完成！AI 建議：" + adv))
    .recordFullBatch({batchName: "果丁丁_300g", totalTime: Math.floor(total/60)+":"+Math.floor(total%60), dtr: dtr, dropTemp: history[history.length-1].y, weather: weatherData, weightLoss: 13.5});
}
initChart();
</script>
</body>
</html>
