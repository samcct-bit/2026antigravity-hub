## 📋 對話理論筆記
```react
import React, { useState, useEffect, useMemo } from 'react';
import { initializeApp } from 'firebase/app';
import { getAuth, signInAnonymously, onAuthStateChanged } from 'firebase/auth';
import { getFirestore, collection, doc, onSnapshot, setDoc, updateDoc, deleteDoc, writeBatch } from 'firebase/firestore';
import { 
  Users, 
  ShoppingCart, 
  Truck, 
  Calendar, 
  CheckCircle2, 
  AlertCircle,
  Plus,
  Trash2,
  Loader2,
  XCircle,
  Database
} from 'lucide-react';

// --- Firebase 配置 ---
const firebaseConfig = JSON.parse(__firebase_config);
const app = initializeApp(firebaseConfig);
const auth = getAuth(app);
const db = getFirestore(app);
const appId = typeof __app_id !== 'undefined' ? __app_id : 'lettuce-master-v3';

const LETTUCE_PRICE = 230;
const WEEKLY_SHIPPING = 300;
const MAX_BAGS_PER_WEEK = 12;

// 訂購日期
const DEFAULT_DATES = ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"];

// 原始已知資料對照表
const INITIAL_IMPORT_DATA = {
  "金成": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "蕙敏": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "珮萱": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "淑芬": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "阿姿": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "嘉鎂": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "斯涵": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "明智": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "翠花": ["2/24", "3/3", "3/10", "3/24"],
  "政君": ["3/3", "3/17", "3/31"],
  "睿婷": ["3/3", "3/17", "3/31"],
  "亭愷": ["3/3", "3/17", "3/31"],
  "俊明": ["2/24", "3/10", "3/24"],
  "惠偵": ["2/24", "3/10", "3/24"]
};

export default function App() {
  const [user, setUser] = useState(null);
  const [orders, setOrders] = useState({});
  const [loading, setLoading] = useState(true);
  const [newMemberName, setNewMemberName] = useState("");
  const [statusMsg, setStatusMsg] = useState(null);

  // 1. 身份認證
  useEffect(() => {
    const initAuth = async () => {
      try {
        if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {
          await signInWithCustomToken(auth, __initial_auth_token);
        } else {
          await signInAnonymously(auth);
        }
      } catch (e) { console.error("Auth failed", e); }
    };
    initAuth();
    const unsubscribe = onAuthStateChanged(auth, setUser);
    return () => unsubscribe();
  }, []);

  // 2. 資料同步
  useEffect(() => {
    if (!user) return;

    const dataPath = `artifacts/${appId}/public/data/lettuce_orders`;
    const unsub = onSnapshot(collection(db, dataPath), (snapshot) => {
      const newOrders = {};
      if (snapshot.empty) {
        // 資料庫全空時，提示匯入
        setLoading(false);
      } else {
        snapshot.docs.forEach(d => {
          newOrders[d.id] = d.data().orders || {};
        });
        setOrders(newOrders);
        setLoading(false);
      }
    }, (err) => {
      console.error("Firestore error:", err);
      setLoading(false);
    });

    return () => unsub();
  }, [user]);

  // 3. 匯入功能 (Batch 寫入)
  const importInitialData = async () => {
    setLoading(true);
    try {
      const batch = writeBatch(db);
      Object.entries(INITIAL_IMPORT_DATA).forEach(([name, dates]) => {
        const docRef = doc(db, `artifacts/${appId}/public/data/lettuce_orders`, name);
        const ordersObj = {};
        dates.forEach(d => ordersObj[d] = true);
        batch.set(docRef, { orders: ordersObj });
      });
      await batch.commit();
      showToast("已成功匯入原始訂購資料！");
    } catch (e) {
      showToast("匯入失敗", "error");
    } finally {
      setLoading(false);
    }
  };

  // 4. 計算邏輯
  const stats = useMemo(() => {
    const dailyCount = {};
    const shippingPerPerson = {};
    const memberList = Object.keys(orders).sort();

    DEFAULT_DATES.forEach(date => {
      let count = 0;
      memberList.forEach(m => { if (orders[m]?.[date]) count++; });
      dailyCount[date] = count;

      if (count === 12) shippingPerPerson[date] = 25;
      else if (count === 11) shippingPerPerson[date] = 27;
      else if (count === 10) shippingPerPerson[date] = 30;
      else if (count > 0) shippingPerPerson[date] = Math.round(WEEKLY_SHIPPING / count);
      else shippingPerPerson[date] = 0;
    });

    return { dailyCount, shippingPerPerson, memberList };
  }, [orders]);

  // 5. 操作處理
  const toggleOrder = async (name, date) => {
    const isOrdered = orders[name]?.[date] || false;
    if (!isOrdered && stats.dailyCount[date] >= MAX_BAGS_PER_WEEK) {
      showToast(`${date} 已達上限 ${MAX_BAGS_PER_WEEK} 包！`, 'error');
      return;
    }
    const docRef = doc(db, `artifacts/${appId}/public/data/lettuce_orders`, name);
    try {
      await updateDoc(docRef, { [`orders.${date}`]: !isOrdered });
    } catch (e) {
      await setDoc(docRef, { orders: { [date]: !isOrdered } }, { merge: true });
    }
  };

  const addMember = async () => {
    const name = newMemberName.trim();
    if (!name) return;
    if (orders[name]) { showToast("成員已存在", "error"); return; }
    await setDoc(doc(db, `artifacts/${appId}/public/data/lettuce_orders`, name), { orders: {} });
    setNewMemberName("");
    showToast(`已新增：${name}`);
  };

  const deleteMember = async (name) => {
    if (!window.confirm(`確定要移除「${name}」？`)) return;
    await deleteDoc(doc(db, `artifacts/${appId}/public/data/lettuce_orders`, name));
  };

  const showToast = (text, type = 'success') => {
    setStatusMsg({ text, type });
    setTimeout(() => setStatusMsg(null), 3000);
  };

  if (loading && Object.keys(orders).length === 0) {
    return (
      <div className="flex flex-col items-center justify-center min-h-screen bg-green-50">
        <Loader2 className="w-10 h-10 text-green-600 animate-spin mb-4" />
        <p className="text-green-800 font-bold tracking-widest">系統準備中...</p>
      </div>
    );
  }

  return (
    <div className="min-h-screen bg-gray-50 pb-20">
      {/* 頂部導航 */}
      <nav className="bg-green-800 text-white shadow-xl sticky top-0 z-30 px-4 py-4">
        <div className="max-w-6xl mx-auto flex flex-col md:flex-row items-center justify-between gap-4">
          <div className="flex items-center gap-3">
            <ShoppingCart className="w-7 h-7" />
            <h1 className="text-xl font-black tracking-tight">生菜訂購管理系統</h1>
          </div>

          <div className="flex flex-wrap justify-center gap-2">
            <div className="flex gap-1 bg-white/10 p-1 rounded-lg">
              <input 
                className="w-32 px-3 py-1.5 rounded-md text-gray-800 text-sm outline-none"
                placeholder="新成員..."
                value={newMemberName}
                onChange={(e) => setNewMemberName(e.target.value)}
              />
              <button onClick={addMember} className="bg-green-500 hover:bg-green-400 p-1.5 rounded-md"><Plus size={18} /></button>
            </div>
            <button 
              onClick={importInitialData}
              className="bg-blue-600 hover:bg-blue-500 px-4 py-1.5 rounded-lg text-sm font-bold flex items-center gap-2 transition-all shadow-lg"
            >
              <Database size={16} /> 匯入原始名單
            </button>
          </div>
        </div>
      </nav>

      <main className="max-w-6xl mx-auto px-4 py-8 space-y-8">
        {statusMsg && (
          <div className={`fixed top-24 right-4 z-50 flex items-center gap-2 px-6 py-3 rounded-2xl shadow-2xl animate-bounce ${
            statusMsg.type === 'error' ? 'bg-red-600 text-white' : 'bg-green-600 text-white'
          }`}>
            <CheckCircle2 size={20} /> <span className="font-bold">{statusMsg.text}</span>
          </div>
        )}

        {/* 訂購矩陣 */}
        <div className="bg-white rounded-3xl shadow-xl overflow-hidden border border-gray-200">
          <div className="overflow-x-auto">
            <table className="w-full text-left">
              <thead>
                <tr className="bg-gray-100 text-gray-500 text-[10px] font-black uppercase tracking-tighter">
                  <th className="p-4 sticky left-0 bg-gray-100 z-20 border-r w-32 text-center">姓名</th>
                  {DEFAULT_DATES.map(date => (
                    <th key={date} className="p-4 text-center min-w-[90px] border-b">
                      <span className="block text-gray-400 font-normal">週二</span>
                      <span className="text-base text-gray-800">{date}</span>
                    </th>
                  ))}
                  <th className="p-4 text-right border-b bg-green-50 text-green-800">應付總額</th>
                </tr>
              </thead>
              <tbody className="divide-y divide-gray-100">
                {stats.memberList.length === 0 ? (
                  <tr>
                    <td colSpan={DEFAULT_DATES.length + 2} className="p-20 text-center text-gray-400">
                      尚未匯入資料，請點擊右上方「匯入原始名單」按鈕。
                    </td>
                  </tr>
                ) : (
                  stats.memberList.map(name => {
                    let bags = 0; let ship = 0;
                    DEFAULT_DATES.forEach(d => { if (orders[name]?.[d]) { bags++; ship += stats.shippingPerPerson[d]; } });
                    return (
                      <tr key={name} className="group hover:bg-green-50/50">
                        <td className="p-3 font-bold text-gray-700 sticky left-0 bg-white z-10 border-r flex items-center justify-between">
                          <span>{name}</span>
                          <button onClick={() => deleteMember(name)} className="opacity-0 group-hover:opacity-100 text-gray-300 hover:text-red-500 transition-all"><Trash2 size={12} /></button>
                        </td>
                        {DEFAULT_DATES.map(date => (
                          <td key={date} className="p-2 text-center">
                            <button
                              onClick={() => toggleOrder(name, date)}
                              className={`mx-auto w-10 h-10 rounded-xl flex items-center justify-center transition-all ${
                                orders[name]?.[date] ? 'bg-green-600 text-white shadow-md' : 'bg-gray-50 text-gray-200 hover:bg-gray-200'
                              }`}
                            >
                              {orders[name]?.[date] ? <CheckCircle2 size={24} /> : <Plus size={16} />}
                            </button>
                          </td>
                        ))}
                        <td className="p-3 text-right bg-green-50/50 font-black text-green-700">${(bags * LETTUCE_PRICE + ship).toLocaleString()}</td>
                      </tr>
                    );
                  })
                )}
              </tbody>
              <tfoot className="bg-gray-800 text-white text-sm">
                <tr className="font-bold">
                  <td className="p-4 sticky left-0 bg-gray-800 border-r">當週小計</td>
                  {DEFAULT_DATES.map(date => (
                    <td key={date} className="p-4 text-center">
                      <div className={`px-2 py-0.5 rounded-full inline-block ${stats.dailyCount[date] >= MAX_BAGS_PER_WEEK ? 'bg-red-500' : 'bg-green-600'}`}>
                        {stats.dailyCount[date]}
                      </div>
                    </td>
                  ))}
                  <td className="p-4 text-right">運費均分制</td>
                </tr>
              </tfoot>
            </table>
          </div>
        </div>

        {/* 成員清單 */}
        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
          {stats.memberList.map(name => {
            let bags = 0; let ship = 0;
            DEFAULT_DATES.forEach(d => { if (orders[name]?.[d]) { bags++; ship += stats.shippingPerPerson[d]; } });
            if (bags === 0) return null;
            return (
              <div key={name} className="bg-white p-5 rounded-2xl border border-gray-200 shadow-sm">
                <div className="flex justify-between items-center mb-2">
                  <span className="font-black text-lg">{name}</span>
                  <span className="text-xs text-gray-400">{bags} 包</span>
                </div>
                <div className="text-xs text-gray-500 space-y-1">
                  <div className="flex justify-between"><span>菜費:</span><span>${bags * LETTUCE_PRICE}</span></div>
                  <div className="flex justify-between border-b pb-1"><span>運費:</span><span>${ship}</span></div>
                  <div className="flex justify-between pt-1 font-black text-green-600 text-lg"><span>總計:</span><span>${(bags * LETTUCE_PRICE + ship).toLocaleString()}</span></div>
                </div>
              </div>
            );
          })}
        </div>
      </main>
    </div>
  );
}
```
