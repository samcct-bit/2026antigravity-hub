  const props = PropertiesService.getScriptProperties();

function toggleOrder(name, date) {
  const props = PropertiesService.getScriptProperties();
  let members = JSON.parse(props.getProperty('members') || "{}");
  if (!members[name]) members[name] = {};
  
  let current = members[name][date] || 0;
  members[name][date] = (current + 1) % 3;
  
  props.setProperty('members', JSON.stringify(members));
  return { success: true, members };
}

  return { success: true, members };

function deleteUser(name) {
  const props = PropertiesService.getScriptProperties();

  return { success: true, members };

  const props = PropertiesService.getScriptProperties();

  props.setProperty('members', JSON.stringify(members));
  return { success: true, members };
}

function addDate(dateStr) {
  const props = PropertiesService.getScriptProperties();
  let dates = JSON.parse(props.getProperty('dates') || "[]");
  if (!dates.includes(dateStr)) {
    dates.push(dateStr);
    dates.sort();
    props.setProperty('dates', JSON.stringify(dates));
  }
  return { success: true, dates };
}
