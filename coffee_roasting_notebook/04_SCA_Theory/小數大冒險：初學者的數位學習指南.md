        function updateCompare(id, val) {
            const v = parseFloat(val);

        }

        function getCompareResult() {
            const a = state.compareValue1, b = state.compareValue2;
            if (a > b) return `${a.toFixed(1)} > ${b.toFixed(1)}`;
            if (a < b) return `${a.toFixed(1)} < ${b.toFixed(1)}`;
            return `${a.toFixed(1)} = ${b.toFixed(1)}`;
        }
