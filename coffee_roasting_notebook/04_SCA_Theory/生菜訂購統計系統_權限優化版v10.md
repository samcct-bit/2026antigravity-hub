```react
import React, { useState, useEffect, useMemo } from 'react';
import { initializeApp } from 'firebase/app';
import { getAuth, signInAnonymously, onAuthStateChanged } from 'firebase/auth';
import { getFirestore, collection, doc, onSnapshot, setDoc, updateDoc, deleteDoc } from 'firebase/firestore';
import { 
  Users, 
  ShoppingCart, 
  Truck, 
  Calendar, 
  CheckCircle2, 
  AlertCircle,
  Plus,
  Trash2,
  Loader2,
  Database,
  Lock,
  Unlock,
  Settings,
  X,
  CreditCard,
  Wifi
} from 'lucide-react';

// --- Firebase 初始化 ---
const firebaseConfig = JSON.parse(__firebase_config);
const app = initializeApp(firebaseConfig);
const auth = getAuth(app);
const db = getFirestore(app);
const appId = typeof __app_id !== 'undefined' ? __app_id : 'lettuce-v10-final';

const LETTUCE_PRICE = 230;
const WEEKLY_SHIPPING = 300;
const MAX_BAGS_PER_WEEK = 12;

// 預設匯入資料
const INITIAL_IMPORT_DATA = {
  "金成": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "蕙敏": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "珮萱": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "淑芬": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "阿姿": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "嘉鎂": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "斯涵": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "明智": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "翠花": ["2/24", "3/3", "3/10", "3/24"],
  "政君": ["3/3", "3/17", "3/31"],
  "睿婷": ["3/3", "3/17", "3/31"],
  "亭愷": ["3/3", "3/17", "3/31"],
  "俊明": ["2/24", "3/10", "3/24"],
  "惠偵": ["2/24", "3/10", "3/24"]
};
const INITIAL_DATES = ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"];

export default function App() {
  const [user, setUser] = useState(null);
  const [orders, setOrders] = useState({});
  const [dates, setDates] = useState([]);
  const [isReady, setIsReady] = useState(false);
  const [isAdmin, setIsAdmin] = useState(false);
  const [currentMonth, setCurrentMonth] = useState("");
  const [isProcessing, setIsProcessing] = useState(false);
  const [newMemberName, setNewMemberName] = useState("");
  const [newDateInput, setNewDateInput] = useState("");
  const [statusMsg, setStatusMsg] = useState(null);
  const [showAdminPanel, setShowAdminPanel] = useState(false);

  // 1. 認證優先 (Rule 3)
  useEffect(() => {
    const initAuth = async () => {
      try {
        const result = await signInAnonymously(auth);
        setUser(result.user);
        setIsReady(true);
      } catch (e) {
        console.error("Auth Failed", e);
        setIsReady(true);
      }
    };
    initAuth();
    const unsubscribe = onAuthStateChanged(auth, setUser);
    return () => unsubscribe();
  }, []);

  // 2. 資料同步 (Rule 1: 扁平化集合路徑)
  useEffect(() => {
    if (!user) return;

    // 將日期設定存在一個專門的 collection 中
    const configDoc = doc(db, 'artifacts', appId, 'public', 'data', 'lettuce_config', 'active_dates');
    const orderCol = collection(db, 'artifacts', appId, 'public', 'data', 'lettuce_orders');

    const unsubDates = onSnapshot(configDoc, (snap) => {
      if (snap.exists()) {
        const d = snap.data().list || [];
        setDates(d);
        if (d.length > 0 && !currentMonth) {
          setCurrentMonth(d[d.length - 1].split('/')[0] + '月');
        }
      }
    });

    const unsubOrders = onSnapshot(orderCol, (snap) => {
      const o = {};
      snap.docs.forEach(d => { o[d.id] = d.data().orders || {}; });
      setOrders(o);
    });

    return () => { unsubDates(); unsubOrders(); };
  }, [user, currentMonth]);

  // 3. 核心統計邏輯
  const stats = useMemo(() => {
    const dailyCount = {};
    const shippingPerPerson = {};
    const memberList = Object.keys(orders).sort();
    const months = new Set();
    
    dates.forEach(d => {
      months.add(d.split('/')[0] + '月');
      let count = 0;
      memberList.forEach(m => { if (orders[m]?.[d]) count++; });
      dailyCount[d] = count;
      if (count >= 12) shippingPerPerson[d] = 25;
      else if (count === 11) shippingPerPerson[d] = 27;
      else if (count === 10) shippingPerPerson[d] = 30;
      else if (count > 0) shippingPerPerson[d] = Math.round(WEEKLY_SHIPPING / count);
      else shippingPerPerson[d] = 0;
    });

    return { 
      dailyCount, 
      shippingPerPerson, 
      memberList, 
      monthList: Array.from(months).sort((a,b) => parseInt(a) - parseInt(b)) 
    };
  }, [orders, dates]);

  const currentDates = useMemo(() => {
    return dates.filter(d => d.split('/')[0] + '月' === currentMonth);
  }, [dates, currentMonth]);

  // 4. 操作功能
  const handleImport = async () => {
    if (!user) { showToast("連線未就緒", "error"); return; }
    setIsProcessing(true);
    try {
      // 匯入日期列
      const configDoc = doc(db, 'artifacts', appId, 'public', 'data', 'lettuce_config', 'active_dates');
      await setDoc(configDoc, { list: INITIAL_DATES });

      // 匯入成員訂單
      for (const [name, dateArr] of Object.entries(INITIAL_IMPORT_DATA)) {
        const orderDoc = doc(db, 'artifacts', appId, 'public', 'data', 'lettuce_orders', name);
        const o = {}; dateArr.forEach(d => o[d] = true);
        await setDoc(orderDoc, { orders: o });
      }

      showToast("匯入完成，請確認表格");
      setCurrentMonth("3月");
    } catch (e) {
      console.error(e);
      showToast("寫入權限錯誤，請重新整理網頁", "error");
    } finally {
      setIsProcessing(false);
    }
  };

  const handleSettle = async (name) => {
    if (!window.confirm(`確認「${name}」已繳費並將訂單歸零？`)) return;
    setIsProcessing(true);
    try {
      const orderDoc = doc(db, 'artifacts', appId, 'public', 'data', 'lettuce_orders', name);
      await updateDoc(orderDoc, { orders: {} });
      showToast(`${name} 已結帳`);
    } catch (e) { showToast("結帳失敗", "error"); }
    finally { setIsProcessing(false); }
  };

  const toggleOrder = async (name, date) => {
    if (isAdmin && showAdminPanel) return;
    const isOrdered = orders[name]?.[date] || false;
    if (!isOrdered && stats.dailyCount[date] >= MAX_BAGS_PER_WEEK) {
      showToast("該週 12 包已滿", "error"); return;
    }
    const orderDoc = doc(db, 'artifacts', appId, 'public', 'data', 'lettuce_orders', name);
    await setDoc(orderDoc, { orders: { [date]: !isOrdered } }, { merge: true });
  };

  const showToast = (text, type = 'success') => {
    setStatusMsg({ text, type });
    setTimeout(() => setStatusMsg(null), 3000);
  };

  if (!isReady) return <div className="h-screen flex items-center justify-center bg-green-50"><Loader2 className="animate-spin text-green-600"/></div>;

  return (
    <div className="min-h-screen bg-slate-50 pb-20 font-sans text-slate-800">
      {/* 導航列 */}
      <nav className="bg-green-700 text-white shadow-xl sticky top-0 z-[60] px-4 py-3">
        <div className="max-w-7xl mx-auto flex items-center justify-between">
          <div className="flex items-center gap-2">
            <ShoppingCart className="w-5 h-5" />
            <h1 className="text-sm sm:text-lg font-black tracking-tight">生菜訂購助手 v10</h1>
          </div>
          
          <div className="flex items-center gap-3">
            <div className={`p-1 rounded-full ${user ? 'text-green-300' : 'text-red-400'}`} title={user ? '連線正常' : '連線異常'}>
              <Wifi size={16} />
            </div>
            <button 
              onClick={() => setIsAdmin(!isAdmin)}
              className={`px-3 py-1.5 rounded-full text-[10px] font-bold flex items-center gap-1 transition-all ${
                isAdmin ? 'bg-red-500 text-white shadow-lg' : 'bg-white/20'
              }`}
            >
              {isAdmin ? <Unlock size={12}/> : <Lock size={12}/>}
              {isAdmin ? '管理模式' : '檢視模式'}
            </button>
            {isAdmin && (
              <button onClick={() => setShowAdminPanel(!showAdminPanel)} className="p-1.5 bg-white/10 rounded-lg">
                <Settings size={20} />
              </button>
            )}
          </div>
        </div>
      </nav>

      <main className="max-w-7xl mx-auto px-2 sm:px-4 py-6 space-y-6">
        
        {/* 管理控制台 */}
        {isAdmin && showAdminPanel && (
          <div className="bg-white p-6 rounded-3xl shadow-2xl border-2 border-red-50 animate-in zoom-in-95">
            <div className="flex items-center justify-between mb-6 border-b pb-4">
              <h3 className="text-red-600 font-black flex items-center gap-2">
                <Settings size={20} /> 管理員控制台
              </h3>
              <button onClick={() => setShowAdminPanel(false)}><X size={20} className="text-slate-400"/></button>
            </div>
            
            <div className="bg-blue-50 p-6 rounded-2xl border border-blue-100 mb-6">
              <p className="text-xs font-black text-blue-600 uppercase mb-3 tracking-widest text-center">初始化與恢復</p>
              <button 
                onClick={handleImport}
                disabled={isProcessing}
                className="w-full bg-blue-600 hover:bg-blue-700 text-white py-4 rounded-xl font-bold flex items-center justify-center gap-2 shadow-lg disabled:opacity-50 transition-all active:scale-95"
              >
                {isProcessing ? <Loader2 className="animate-spin" size={20}/> : <Database size={20} />}
                立即匯入 14 位成員訂單 (2/24 ~ 3/31)
              </button>
            </div>

            <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
              <div className="flex gap-2">
                <input className="flex-1 border p-2 rounded-xl text-sm outline-none" placeholder="新成員姓名" value={newMemberName} onChange={e => setNewMemberName(e.target.value)}/>
                <button onClick={async () => {
                  if(!newMemberName) return;
                  await setDoc(doc(db, 'artifacts', appId, 'public', 'data', 'lettuce_orders', newMemberName), { orders: {} });
                  setNewMemberName(""); showToast("已新增成員");
                }} className="bg-green-600 text-white px-4 py-2 rounded-xl font-bold text-xs whitespace-nowrap">新增成員</button>
              </div>
              <div className="flex gap-2">
                <input className="flex-1 border p-2 rounded-xl text-sm outline-none" placeholder="新日期 (4/7)" value={newDateInput} onChange={e => setNewDateInput(e.target.value)}/>
                <button onClick={async () => {
                  if(!newDateInput.includes('/')) return;
                  const updated = [...dates, newDateInput].sort((a,b) => {
                    const [am, ad] = a.split('/').map(Number);
                    const [bm, bd] = b.split('/').map(Number);
                    return am !== bm ? am - bm : ad - bd;
                  });
                  await setDoc(doc(db, 'artifacts', appId, 'public', 'data', 'lettuce_config', 'active_dates'), { list: updated });
                  setNewDateInput(""); showToast("已新增日期");
                }} className="bg-blue-600 text-white px-4 py-2 rounded-xl font-bold text-xs whitespace-nowrap">新增日期</button>
              </div>
            </div>
          </div>
        )}

        {/* 月份切換列 */}
        <div className="flex items-center gap-2 bg-white p-2 rounded-2xl shadow-sm border border-slate-200 overflow-x-auto no-scrollbar">
            {stats.monthList.map(m => (
                <button 
                    key={m}
                    onClick={() => setCurrentMonth(m)}
                    className={`px-6 py-2 rounded-xl font-black text-xs transition-all whitespace-nowrap ${
                        currentMonth === m ? 'bg-green-600 text-white shadow-md' : 'text-slate-400 hover:bg-slate-100'
                    }`}
                >
                    {m}份
                </button>
            ))}
        </div>

        {/* 訂購表格 (固定標題與姓名) */}
        <div className="bg-white rounded-3xl shadow-xl overflow-hidden border border-slate-200">
          <div className="overflow-auto max-h-[60vh]">
            <table className="w-full text-left border-separate border-spacing-0">
              <thead className="sticky top-0 z-40 bg-slate-900 text-white shadow-md">
                <tr>
                  <th className="p-4 sticky left-0 z-50 bg-slate-900 border-r border-slate-800 text-center text-[10px] font-black w-24 sm:w-32">成員</th>
                  {currentDates.map(date => (
                    <th key={date} className="p-4 text-center min-w-[80px] border-b border-slate-800">
                      <div className="flex flex-col">
                        <span className="text-[8px] text-slate-500 font-normal">TUE</span>
                        <span className="text-sm font-black">{date}</span>
                      </div>
                    </th>
                  ))}
                  <th className="p-4 text-right border-b border-slate-800 bg-slate-950 text-green-400 text-[10px] font-black">累計應付</th>
                </tr>
              </thead>
              <tbody className="divide-y divide-slate-100">
                {stats.memberList.length === 0 ? (
                  <tr>
                    <td colSpan={currentDates.length + 2} className="p-20 text-center text-slate-400 text-sm">
                      目前沒有資料，請進入「管理模式」並點選「一鍵匯入」。
                    </td>
                  </tr>
                ) : (
                  stats.memberList.map(name => {
                    let totalBags = 0; let totalShip = 0;
                    dates.forEach(d => { if (orders[name]?.[d]) { totalBags++; totalShip += stats.shippingPerPerson[d]; } });

                    return (
                      <tr key={name} className="hover:bg-slate-50 group transition-colors">
                        <td className="p-3 font-bold text-slate-700 sticky left-0 bg-white z-20 border-r border-slate-100 flex items-center justify-between gap-1 shadow-sm">
                          <span className="truncate">{name}</span>
                          {isAdmin && (
                            <button onClick={async () => { if(window.confirm(`刪除 ${name}？`)) await deleteDoc(doc(db, 'artifacts', appId, 'public', 'data', 'lettuce_orders', name)); }} className="text-red-400 p-1 rounded hover:bg-red-50"><Trash2 size={12}/></button>
                          )}
                        </td>
                        {currentDates.map(date => (
                          <td key={date} className="p-2 text-center">
                            <button
                              onClick={() => toggleOrder(name, date)}
                              className={`mx-auto w-10 h-10 rounded-xl flex items-center justify-center transition-all ${
                                orders[name]?.[date] ? 'bg-green-600 text-white shadow-lg scale-110' : 'bg-slate-50 text-slate-200 active:scale-95'
                              }`}
                            >
                              {orders[name]?.[date] ? <CheckCircle2 size={24} /> : <Plus size={16} />}
                            </button>
                          </td>
                        ))}
                        <td className="p-4 text-right bg-slate-50/80 sticky right-0 z-10 font-mono font-black text-green-700 text-sm">
                          ${(totalBags * LETTUCE_PRICE + totalShip).toLocaleString()}
                        </td>
                      </tr>
                    );
                  })
                )}
              </tbody>
              <tfoot className="sticky bottom-0 z-40 bg-slate-900 text-white text-[10px] border-t border-slate-700 shadow-[0_-2px_10px_rgba(0,0,0,0.1)]">
                <tr className="font-bold">
                  <td className="p-4 sticky left-0 bg-slate-900 border-r border-slate-800">當週數量</td>
                  {currentDates.map(date => (
                    <td key={date} className="p-4 text-center">
                      <div className={`px-2 py-0.5 rounded-full inline-block ${stats.dailyCount[date] >= MAX_BAGS_PER_WEEK ? 'bg-red-500' : 'bg-green-600'}`}>
                        {stats.dailyCount[date]} / 12
                      </div>
                    </td>
                  ))}
                  <td className="p-4 text-right">份數</td>
                </tr>
                <tr className="bg-slate-800 text-blue-300">
                  <td className="p-2 sticky left-0 bg-slate-800 border-r border-slate-700 text-center">分擔運費</td>
                  {currentDates.map(date => (
                    <td key={date} className="p-2 text-center font-mono">
                      <div className="flex items-center justify-center gap-1 opacity-70">
                          <Truck size={10} /> ${stats.shippingPerPerson[date]}
                      </div>
                    </td>
                  ))}
                  <td className="p-2"></td>
                </tr>
              </tfoot>
            </table>
          </div>
        </div>

        {/* 結帳卡片區 */}
        <div className="space-y-4">
          <div className="flex items-center gap-2">
            <Users size={18} className="text-green-600"/>
            <h2 className="font-black text-lg text-slate-700">繳費結帳清單</h2>
          </div>
          <div className="grid grid-cols-2 lg:grid-cols-4 gap-3">
            {stats.memberList.map(name => {
              let bags = 0; let ship = 0;
              dates.forEach(d => { if (orders[name]?.[d]) { bags++; ship += stats.shippingPerPerson[d]; } });
              if (bags === 0 && !isAdmin) return null;

              return (
                <div key={name} className="bg-white p-4 rounded-3xl border border-slate-200 shadow-sm flex flex-col justify-between hover:shadow-lg transition-all relative overflow-hidden group">
                  <div className="absolute top-0 left-0 w-1.5 h-full bg-green-500"></div>
                  <div>
                    <div className="flex justify-between items-start mb-2">
                      <span className="font-black text-slate-800">{name}</span>
                      <span className="text-[10px] bg-slate-100 px-2 py-0.5 rounded text-slate-500 font-bold">{bags} 包</span>
                    </div>
                    <div className="text-[11px] text-slate-500 space-y-1 pb-3 border-b border-dashed">
                      <div className="flex justify-between"><span>生菜費用:</span><span>${bags * LETTUCE_PRICE}</span></div>
                      <div className="flex justify-between"><span>累計運費:</span><span>${ship}</span></div>
                      <div className="flex justify-between pt-1 font-black text-green-600 text-lg"><span>總計:</span><span>${(bags * LETTUCE_PRICE + ship).toLocaleString()}</span></div>
                    </div>
                  </div>
                  
                  {isAdmin && (
                    <button 
                      onClick={() => handleSettle(name)}
                      disabled={bags === 0}
                      className={`mt-4 w-full py-2 rounded-xl text-[10px] font-black flex items-center justify-center gap-1 transition-all shadow-md ${
                        bags === 0 ? 'bg-slate-50 text-slate-300' : 'bg-green-600 text-white hover:bg-green-700'
                      }`}
                    >
                      <CreditCard size={12}/> 匯款結帳歸零
                    </button>
                  )}
                </div>
              );
            })}
          </div>
        </div>
      </main>

      {/* 底部狀態通知 */}
      {statusMsg && (
        <div className={`fixed bottom-8 left-1/2 -translate-x-1/2 z-[100] flex items-center gap-2 px-8 py-3 rounded-full shadow-2xl animate-in slide-in-from-bottom-8 ${
          statusMsg.type === 'error' ? 'bg-red-600 text-white' : 'bg-green-800 text-white'
        }`}>
          {statusMsg.type === 'error' ? <AlertCircle size={18}/> : <CheckCircle2 size={18} />}
          <span className="font-bold text-sm tracking-tight">{statusMsg.text}</span>
        </div>
      )}
    </div>
  );
}
```
