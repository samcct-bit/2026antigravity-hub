  <main class="max-w-7xl mx-auto p-2 sm:p-6 space-y-6">

    <!-- Admin Panel -->
    <div id="adminPanel" class="bg-white p-6 rounded-2xl shadow-xl border-2 border-red-100 hidden animate-in slide-in-from-top-4">

        <div class="space-y-3">

  <script>
    let appData = { dates: [], members: {} };
    let currentMonth = "";
    let isAdmin = false;

    window.onload = () => fetchData();

    function fetchData() {
      showLoading(true);
      google.script.run.withSuccessHandler(data => {
        appData = data;
        if (data.dates.length > 0 && !currentMonth) currentMonth = data.dates[data.dates.length-1].split('/')[0] + '月';
        renderUI();
        showLoading(false);
      }).getAppData();
    }

    function renderUI() {
      renderMonthTabs();
      renderTable();
      renderCards();
    }

    function renderMonthTabs() {
      const months = [...new Set(appData.dates.map(d => d.split('/')[0] + '月'))].sort((a,b) => parseInt(a)-parseInt(b));
      document.getElementById('monthTabs').innerHTML = months.map(m => `
        <button onclick="currentMonth='${m}';renderUI();" class="px-6 py-2 rounded-xl font-black text-xs whitespace-nowrap ${currentMonth === m ? 'bg-green-600 text-white shadow-lg' : 'bg-white border text-slate-400'}">
          ${m}份
        </button>
      `).join('');
    }

    function calculateStats() {
      const counts = {}; const shipping = {};
      appData.dates.forEach(d => {
        let totalBags = 0;
        Object.keys(appData.members).forEach(n => totalBags += (appData.members[n][d] || 0));
        counts[d] = totalBags;
        if (totalBags >= 12) shipping[d] = 25;
        else if (totalBags === 11) shipping[d] = 27;
        else if (totalBags === 10) shipping[d] = 30;
        else if (totalBags > 0) shipping[d] = Math.round(300 / totalBags);
        else shipping[d] = 0;
      });
      return { counts, shipping };
    }

    function renderTable() {
      const datesInMonth = appData.dates.filter(d => d.split('/')[0] + '月' === currentMonth);
      const memberNames = Object.keys(appData.members).sort();
      const stats = calculateStats();

      let html = `<thead><tr><th class="p-4 text-center">成員</th>`;
      datesInMonth.forEach(d => html += `<th class="p-4 text-center min-w-[80px]"><span class="block text-[8px] opacity-50">TUE</span>${d}</th>`);
      html += `<th class="p-4 text-right bg-slate-900 text-green-400">累計</th></tr></thead><tbody class="divide-y divide-slate-100">`;

      memberNames.forEach(name => {
        let totalCost = 0;
        appData.dates.forEach(d => {
          const qty = appData.members[name][d] || 0;
          if (qty > 0) totalCost += (qty * 230) + (qty * stats.shipping[d]);
        });

        html += `<tr><td class="p-3 sticky-col font-bold flex justify-between items-center bg-white">
          <span class="truncate">${name}</span>
          ${isAdmin ? `<button onclick="runDelete('${name}')" class="text-red-400 p-1"><i class="fas fa-trash-alt text-[10px]"></i></button>` : ''}
        </td>`;
        
        datesInMonth.forEach(d => {
          const qty = appData.members[name][d] || 0;
          html += `<td class="p-2 text-center">
            <button onclick="runToggle('${name}', '${d}')" class="w-10 h-10 rounded-xl transition-all flex items-center justify-center mx-auto ${qty === 1 ? 'bg-green-600 text-white' : qty === 2 ? 'bg-blue-600 text-white' : 'bg-slate-50 text-slate-200'}">
              ${qty === 2 ? '<span class="font-black">2</span>' : qty === 1 ? '<i class="fas fa-check"></i>' : '<i class="fas fa-plus text-[10px]"></i>'}
            </button>
          </td>`;
        });

        html += `<td class="p-4 text-right sticky-total font-mono font-black text-green-700">$${totalCost.toLocaleString()}</td></tr>`;
      });

      html += `</tbody><tfoot class="sticky bottom-0 z-40 bg-slate-800 text-white text-[10px] font-bold">
        <tr><td class="p-3 sticky-col bg-slate-800 text-slate-400">每週總量</td>
        ${datesInMonth.map(d => `<td class="p-3 text-center"><span class="px-2 py-0.5 rounded-full ${stats.counts[d] >= 12 ? 'bg-red-500' : 'bg-green-600'}">${stats.counts[d]}/12</span></td>`).join('')}
        <td class="p-3 text-right sticky-total bg-slate-800 italic">Total</td></tr>
      </tfoot>`;

      document.getElementById('orderTable').innerHTML = html;
    }

    function renderCards() {
      const stats = calculateStats();
      const container = document.getElementById('summaryCards');
      container.innerHTML = Object.keys(appData.members).sort().map(name => {
        let bags = 0; let shipTotal = 0;
        appData.dates.forEach(d => {
          const qty = appData.members[name][d] || 0;
          if (qty > 0) { bags += qty; shipTotal += (qty * stats.shipping[d]); }
        });
        if (bags === 0 && !isAdmin) return '';
        return `
          <div class="bg-white p-4 rounded-2xl border border-slate-200 shadow-sm relative overflow-hidden group">
            <div class="absolute top-0 left-0 w-1.5 h-full bg-green-500"></div>
            <div class="flex justify-between mb-2"><span class="font-black text-slate-800">${name}</span><span class="text-[10px] bg-slate-100 px-2 py-0.5 rounded font-bold">${bags}包</span></div>
            <div class="text-[10px] text-slate-500 space-y-1 pb-2 border-b border-dashed">
              <div class="flex justify-between"><span>生菜:</span><span>$${bags * 230}</span></div>
              <div class="flex justify-between"><span>運費:</span><span>$${shipTotal}</span></div>
              <div class="flex justify-between pt-1 font-black text-green-600 text-sm"><span>總額:</span><span>$${(bags * 230 + shipTotal).toLocaleString()}</span></div>
            </div>
            ${isAdmin ? `<button onclick="runSettle('${name}')" class="mt-3 w-full py-1.5 rounded-lg text-[9px] font-black bg-green-600 text-white active:scale-95 shadow-md"><i class="fas fa-credit-card mr-1"></i> 已匯款 (歸零)</button>` : ''}
          </div>
        `;
      }).join('');
    }

    function runToggle(name, date) {
      google.script.run.withSuccessHandler(res => { appData.members = res.members; renderUI(); }).toggleOrder(name, date);
    }

    function runSettle(name) {
      if (!confirm(`確認「${name}」已繳費並將金額歸零？`)) return;
      showLoading(true);
      google.script.run.withSuccessHandler(res => { appData.members = res.members; renderUI(); showLoading(false); }).settleUser(name);
    }

    function runDelete(name) {
      if (!confirm(`確定刪除「${name}」？`)) return;
      showLoading(true);
      google.script.run.withSuccessHandler(res => { appData.members = res.members; renderUI(); showLoading(false); }).deleteUser(name);
    }

    function runAddUser() {
      const n = document.getElementById('newName').value.trim(); if (!n) return;
      showLoading(true);
      google.script.run.withSuccessHandler(res => { appData.members = res.members; document.getElementById('newName').value = ""; renderUI(); showLoading(false); }).addUser(n);
    }

    function runAddDate() {
      const d = document.getElementById('newDate').value.trim(); if (!d.includes('/')) return;
      showLoading(true);
      google.script.run.withSuccessHandler(res => { appData.dates = res.dates; document.getElementById('newDate').value = ""; renderUI(); showLoading(false); }).addDate(d);
    }

    function runImport() {
      if (!confirm("確定要匯入四月份 14 人名單嗎？這會覆寫現有資料。")) return;
      showLoading(true);
      google.script.run.withSuccessHandler(res => { appData = res; currentMonth = "4月"; renderUI(); showLoading(false); }).importAprilData();
    }

    function toggleAdmin() {
      isAdmin = !isAdmin;
      const btn = document.getElementById('adminBtn');
      btn.innerHTML = isAdmin ? '<i class="fas fa-unlock"></i> 管理模式' : '<i class="fas fa-lock"></i> 唯讀模式';
      btn.className = `px-3 py-1.5 rounded-full text-[10px] font-bold transition-all ${isAdmin ? 'bg-red-500 text-white' : 'bg-white/10 text-white'}`;
      renderUI();
    }

    function showLoading(show) { document.getElementById('loadingOverlay').classList.toggle('hidden', !show); }
  </script>
</body>
</html>
