        function calculate() {
            const tds = parseFloat(document.getElementById('tds').value) || 0;
            const beverage = parseFloat(document.getElementById('beverage').value) || 0;
            const grounds = parseFloat(document.getElementById('grounds').value) || 0;
            
            const display = document.getElementById('extYieldDisplay');
            const badge = document.getElementById('statusBadge');
            const dot = document.getElementById('targetDot');

            if (tds > 0 && beverage > 0 && grounds > 0) {
                // 公式：(TDS * Beverage) / Grounds
                const yieldResult = (tds * beverage) / grounds;
                display.innerText = yieldResult.toFixed(2) + ' %';

                // 判斷狀態
                let status = "";
                let color = "";
                let bgColor = "";

                const isYieldGood = yieldResult >= 18 && yieldResult <= 22;
                const isTdsGood = tds >= 1.15 && tds <= 1.45;

                if (isYieldGood && isTdsGood) {
                    status = "理想萃取 (Gold Cup)";
                    color = "text-green-700";
                    bgColor = "bg-green-100";
                } else if (yieldResult < 18) {
                    status = "萃取不足 (Under-extracted)";
                    color = "text-blue-700";
                    bgColor = "bg-blue-100";
                } else if (yieldResult > 22) {
                    status = "過度萃取 (Over-extracted)";
                    color = "text-red-700";
                    bgColor = "bg-red-100";
                } else {
                    status = "濃度偏離理想區間";
                    color = "text-orange-700";
                    bgColor = "bg-orange-100";
                }

                badge.innerText = status;
                badge.className = `inline-block px-3 py-1 rounded-full text-xs font-bold uppercase ${color} ${bgColor}`;

                // 更新圖表點
                // X 軸對應 (10% to 30%)
                let xPos = ((yieldResult - 10) / 20) * 100;
                // Y 軸對應 (0.8% to 1.8%)
                let yPos = 100 - ((tds - 0.8) / 1.0) * 100;

                // 限制範圍不超出 SVG
                xPos = Math.max(0, Math.min(100, xPos));
                yPos = Math.max(0, Math.min(100, yPos));

                dot.setAttribute('cx', xPos);
                dot.setAttribute('cy', yPos);
            } else {
                display.innerText = "-- %";
                badge.innerText = "--";
                badge.className = "inline-block px-3 py-1 rounded-full text-xs font-bold uppercase bg-gray-100 text-gray-400";
                dot.setAttribute('cx', -10);
            }
        }

        // 初始化
        window.onload = calculate;
    </script>
