  function getLayoutByPlaceholder(needsBody, needsSubtitle) {

        if (pType === SlidesApp.PlaceholderType.SUBTITLE) hasSubtitle = true;
      }
      
      // 如果這個版型符合我們的需求，就直接回傳
      if (needsBody && hasBody) return layout;       // 找「標題與內文」版型

        bodyShape = shapes[k];

           .setFontSize(36)
           .setBold(true);
    }
    
    // 填入內文
    if (bodyShape) {
      var textRange = bodyShape.getText();
      var content = bodyItems.join("\n");
      textRange.setText(content);
      textRange.getTextStyle()
           .setForegroundColor(bodyColor)
           .setFontSize(20);
      
      var paragraphStyle = textRange.getParagraphStyle();
