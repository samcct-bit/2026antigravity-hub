// 洗牌函數
const shuffleArray = (array) => {
  const shuffled = [...array];
  for (let i = shuffled.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [shuffled[i], shuffled[j]] = [shuffled[j], shuffled[i]];
  }
  return shuffled;
};

const getRandomQuestions = (pool, count) => {
  const shuffled = shuffleArray(pool);
  return shuffled.slice(0, count);
};

              } else if (selectedOptionIndex === index) {
                if (feedback === 'correct') {
                  buttonStyle = "bg-green-500 border-green-400 text-white shadow-[0_0_15px_rgba(34,197,94,0.6)] z-20 scale-105";
                } else if (feedback === 'incorrect') {
                  buttonStyle = "bg-red-500 border-red-400 text-white animate-shake";
                }
              } else if (feedback === 'correct') {
                buttonStyle = "bg-slate-800 border-slate-700 text-slate-500 opacity-50 cursor-not-allowed";
              }

              return (
                <button
                  key={index}
                  onClick={() => handleAnswer(index)}
                  disabled={feedback === 'correct' || isLocked}
                  className={`relative min-h-[64px] md:min-h-[80px] p-4 rounded-2xl border-b-4 text-lg md:text-2xl font-bold transition-all flex items-center justify-center gap-3 touch-manipulation select-none ${buttonStyle}`}
                >
                  <span className={`${isLocked ? 'line-through decoration-red-500/50 decoration-2' : ''} font-mono tracking-wider`}>
                    {option}
                  </span>
                  {selectedOptionIndex === index && feedback === 'correct' && <CheckCircle className="w-6 h-6 text-white absolute right-4" />}
                  {isLocked && <XCircle className="w-6 h-6 text-red-500/50 absolute right-4" />}
                </button>
              );
            })}
          </div>

          {/* 提示機器人區塊 */}
          <div className="mt-8 flex flex-col items-center relative z-10 min-h-[80px]">
            {!showHint ? (
              <button 
                onClick={handleShowHint}
                className="flex items-center gap-2 text-blue-300 hover:text-blue-100 bg-blue-900/40 hover:bg-blue-900/70 px-5 py-3 rounded-full transition-colors text-sm md:text-base border border-blue-800 touch-manipulation"
              >
                <HelpCircle className="w-5 h-5" />
                呼叫機器人提示 (將扣除部分分數)
              </button>
            ) : (
              <div className="bg-indigo-900/60 border border-indigo-400/50 text-indigo-100 p-4 md:p-5 rounded-2xl text-base md:text-lg animate-pop-in flex items-start gap-3 w-full shadow-inner">
                <div className="text-2xl md:text-3xl shrink-0">🤖</div>
                <div className="font-medium leading-relaxed mt-1">
                  <span className="text-yellow-300 font-bold block mb-1">【系統強制學習模式開啟】</span>
                  {currentQuestion.hint}
                </div>
              </div>
            )}
          </div>
        </div>
      </div>
    );
  };

  const renderEndScreen = () => (
    <div className="flex flex-col items-center justify-center min-h-[80vh] text-center animate-pop-in space-y-6 px-4">
      <div className="relative">
        <Star className="w-28 h-28 md:w-40 md:h-40 text-yellow-400 animate-float drop-shadow-[0_0_40px_rgba(250,204,21,0.6)]" />
        <Sparkles className="w-10 h-10 text-white absolute top-0 right-0 animate-pulse" />
      </div>
      <h1 className="text-4xl md:text-6xl font-extrabold text-white tracking-wider drop-shadow-lg">
        任務完成！
      </h1>
      
      <div className="bg-slate-800 p-8 md:p-10 rounded-[2rem] border-4 border-yellow-400/30 mt-6 w-full max-w-sm shadow-[0_0_50px_rgba(250,204,21,0.15)] relative overflow-hidden">
        <div className="absolute inset-0 bg-gradient-to-b from-yellow-400/5 to-transparent"></div>
        <div className="text-slate-400 text-sm md:text-base uppercase tracking-widest mb-2 relative z-10">最終獲得能量積分</div>
        <div className="text-6xl md:text-7xl font-black text-yellow-400 relative z-10">{score}</div>
        <div className="text-slate-400 text-sm mt-4">
          滿分 240 分
        </div>
      </div>

      {score === 240 ? (
        <p className="text-green-400 text-xl font-bold mt-4">🏆 完美過關！你完全沒有猜測！</p>
      ) : score > 180 ? (
        <p className="text-blue-300 text-xl mt-4">表現很棒！下次挑戰完美通關吧！</p>
      ) : (
        <p className="text-yellow-300 text-xl mt-4">有很多是看提示或猜出來的對吧？再多練習一次！</p>
      )}

      <button 
        onClick={startGame}
        className="mt-8 w-full max-w-xs py-4 bg-slate-700 hover:bg-slate-600 text-white text-xl font-bold rounded-2xl transition-all transform hover:scale-105 active:scale-95 flex items-center justify-center gap-3 border border-slate-500 shadow-lg touch-manipulation"
      >
        <RefreshCcw className="w-6 h-6" />
        再次挑戰新題目
      </button>
    </div>
  );

  return (
    <div className="min-h-screen bg-slate-900 bg-[radial-gradient(ellipse_at_top,_var(--tw-gradient-stops))] from-slate-900 via-indigo-950 to-slate-900 flex items-center justify-center py-8 font-sans selection:bg-blue-500/30">
      <div className="w-full max-w-4xl">
        {gameState === 'START' && renderStartScreen()}
        {gameState === 'PLAYING' && renderPlayingScreen()}
        {gameState === 'END' && renderEndScreen()}
      </div>
    </div>
  );
}
