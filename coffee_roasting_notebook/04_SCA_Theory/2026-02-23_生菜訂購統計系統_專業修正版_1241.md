```react
import React, { useState, useEffect, useMemo } from 'react';
import { initializeApp } from 'firebase/app';
import { getAuth, signInAnonymously, onAuthStateChanged, signInWithCustomToken } from 'firebase/auth';
import { getFirestore, collection, doc, onSnapshot, setDoc, updateDoc, deleteDoc, writeBatch } from 'firebase/firestore';
import { 
  Users, 
  ShoppingCart, 
  Truck, 
  Calendar, 
  CheckCircle2, 
  AlertCircle,
  Plus,
  Trash2,
  Loader2,
  Database,
  X
} from 'lucide-react';

// --- Firebase 配置 ---
const firebaseConfig = JSON.parse(__firebase_config);
const app = initializeApp(firebaseConfig);
const auth = getAuth(app);
const db = getFirestore(app);
const appId = typeof __app_id !== 'undefined' ? __app_id : 'lettuce-master-v5';

const LETTUCE_PRICE = 230;
const WEEKLY_SHIPPING = 300;
const MAX_BAGS_PER_WEEK = 12;
const DEFAULT_DATES = ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"];

const INITIAL_IMPORT_DATA = {
  "金成": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "蕙敏": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "珮萱": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "淑芬": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "阿姿": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "嘉鎂": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "斯涵": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "明智": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "翠花": ["2/24", "3/3", "3/10", "3/24"],
  "政君": ["3/3", "3/17", "3/31"],
  "睿婷": ["3/3", "3/17", "3/31"],
  "亭愷": ["3/3", "3/17", "3/31"],
  "俊明": ["2/24", "3/10", "3/24"],
  "惠偵": ["2/24", "3/10", "3/24"]
};

export default function App() {
  const [user, setUser] = useState(null);
  const [orders, setOrders] = useState({});
  const [isReady, setIsReady] = useState(false);
  const [isImporting, setIsImporting] = useState(false);
  const [newMemberName, setNewMemberName] = useState("");
  const [statusMsg, setStatusMsg] = useState(null);
  
  // 自訂刪除確認 Modal 狀態
  const [deleteConfirm, setDeleteConfirm] = useState(null);

  useEffect(() => {
    const initAuth = async () => {
      try {
        if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {
          await signInWithCustomToken(auth, __initial_auth_token);
        } else {
          await signInAnonymously(auth);
        }
      } catch (e) {
        console.error("Auth failed:", e);
        setIsReady(true);
      }
    };
    initAuth();
    const unsubscribe = onAuthStateChanged(auth, (u) => {
      setUser(u);
      if (u) setIsReady(true);
    });
    return () => unsubscribe();
  }, []);

  useEffect(() => {
    if (!user) return;
    const dataPath = `artifacts/${appId}/public/data/lettuce_orders`;
    const unsub = onSnapshot(collection(db, dataPath), (snapshot) => {
      const newOrders = {};
      snapshot.docs.forEach(d => {
        newOrders[d.id] = d.data().orders || {};
      });
      setOrders(newOrders);
      setIsReady(true);
    }, (err) => {
      console.error("Firestore error:", err);
      setIsReady(true);
    });
    return () => unsub();
  }, [user]);

  const importInitialData = async () => {
    if (!user) return;
    setIsImporting(true);
    try {
      const batch = writeBatch(db);
      Object.entries(INITIAL_IMPORT_DATA).forEach(([name, dates]) => {
        const docRef = doc(db, `artifacts/${appId}/public/data/lettuce_orders`, name);
        const ordersObj = {};
        dates.forEach(d => ordersObj[d] = true);
        batch.set(docRef, { orders: ordersObj });
      });
      await batch.commit();
      showToast("已成功匯入資料！");
    } catch (e) {
      showToast("匯入失敗", "error");
    } finally {
      setIsImporting(false);
    }
  };

  const stats = useMemo(() => {
    const dailyCount = {};
    const shippingPerPerson = {};
    const memberList = Object.keys(orders).sort();
    DEFAULT_DATES.forEach(date => {
      let count = 0;
      memberList.forEach(m => { if (orders[m]?.[date]) count++; });
      dailyCount[date] = count;
      if (count === 12) shippingPerPerson[date] = 25;
      else if (count === 11) shippingPerPerson[date] = 27;
      else if (count === 10) shippingPerPerson[date] = 30;
      else if (count > 0) shippingPerPerson[date] = Math.round(WEEKLY_SHIPPING / count);
      else shippingPerPerson[date] = 0;
    });
    return { dailyCount, shippingPerPerson, memberList };
  }, [orders]);

  const toggleOrder = async (name, date) => {
    const isOrdered = orders[name]?.[date] || false;
    if (!isOrdered && stats.dailyCount[date] >= MAX_BAGS_PER_WEEK) {
      showToast(`${date} 已達上限 ${MAX_BAGS_PER_WEEK} 包！`, 'error');
      return;
    }
    const docRef = doc(db, `artifacts/${appId}/public/data/lettuce_orders`, name);
    try {
      await updateDoc(docRef, { [`orders.${date}`]: !isOrdered });
    } catch (e) {
      await setDoc(docRef, { orders: { [date]: !isOrdered } }, { merge: true });
    }
  };

  const addMember = async () => {
    const name = newMemberName.trim();
    if (!name) return;
    if (orders[name]) { showToast("成員已存在", "error"); return; }
    await setDoc(doc(db, `artifacts/${appId}/public/data/lettuce_orders`, name), { orders: {} });
    setNewMemberName("");
    showToast(`已新增成員：${name}`);
  };

  const handleDeleteClick = (name) => {
    setDeleteConfirm(name);
  };

  const confirmDelete = async () => {
    if (!deleteConfirm) return;
    const nameToDelete = deleteConfirm;
    setDeleteConfirm(null);
    try {
      await deleteDoc(doc(db, `artifacts/${appId}/public/data/lettuce_orders`, nameToDelete));
      showToast(`已移除成員：${nameToDelete}`);
    } catch (e) {
      showToast("移除失敗", "error");
    }
  };

  const showToast = (text, type = 'success') => {
    setStatusMsg({ text, type });
    setTimeout(() => setStatusMsg(null), 3000);
  };

  if (!isReady) {
    return (
      <div className="flex flex-col items-center justify-center min-h-screen bg-green-50">
        <Loader2 className="w-10 h-10 text-green-600 animate-spin mb-4" />
        <p className="text-green-800 font-bold">資料同步中...</p>
      </div>
    );
  }

  return (
    <div className="min-h-screen bg-slate-50 pb-20 font-sans text-slate-900">
      {/* 導航列 */}
      <nav className="bg-green-700 text-white shadow-lg sticky top-0 z-50 px-4 py-3">
        <div className="max-w-6xl mx-auto flex flex-col sm:flex-row items-center justify-between gap-4">
          <div className="flex items-center gap-2">
            <ShoppingCart className="w-6 h-6" />
            <h1 className="text-lg font-black tracking-tight">生菜訂購助手 v5</h1>
          </div>
          <div className="flex w-full sm:w-auto gap-2">
            <div className="flex flex-1 bg-white/10 p-1 rounded-lg backdrop-blur-md">
              <input 
                className="w-full sm:w-32 px-3 py-1.5 rounded-md text-slate-800 text-sm outline-none"
                placeholder="新增姓名..."
                value={newMemberName}
                onChange={(e) => setNewMemberName(e.target.value)}
                onKeyDown={(e) => e.key === 'Enter' && addMember()}
              />
              <button onClick={addMember} className="bg-green-500 hover:bg-green-400 p-1.5 rounded-md ml-1"><Plus size={18} /></button>
            </div>
            <button 
              onClick={importInitialData}
              disabled={isImporting}
              className="bg-blue-600 hover:bg-blue-500 disabled:opacity-50 px-3 py-1.5 rounded-lg text-xs font-bold flex items-center gap-1 shadow-md transition-all"
            >
              {isImporting ? <Loader2 size={14} className="animate-spin" /> : <Database size={14} />}
              匯入
            </button>
          </div>
        </div>
      </nav>

      <main className="max-w-6xl mx-auto px-2 sm:px-4 py-6 space-y-6">
        {/* 狀態提示 */}
        {statusMsg && (
          <div className={`fixed bottom-6 left-1/2 -translate-x-1/2 z-[100] flex items-center gap-2 px-6 py-3 rounded-full shadow-2xl animate-in fade-in slide-in-from-bottom-4 ${
            statusMsg.type === 'error' ? 'bg-red-600 text-white' : 'bg-green-600 text-white'
          }`}>
            <AlertCircle size={18} /> <span className="font-bold text-sm">{statusMsg.text}</span>
          </div>
        )}

        {/* 刪除確認彈窗 */}
        {deleteConfirm && (
          <div className="fixed inset-0 z-[110] flex items-center justify-center p-4 bg-slate-900/60 backdrop-blur-sm animate-in fade-in">
            <div className="bg-white rounded-2xl p-6 w-full max-w-xs shadow-2xl scale-in-center">
              <h3 className="text-lg font-black mb-2 text-slate-800">確認移除？</h3>
              <p className="text-slate-500 text-sm mb-6">確定要將「<span className="text-red-600 font-bold">{deleteConfirm}</span>」從名單中移除嗎？</p>
              <div className="flex gap-3">
                <button onClick={() => setDeleteConfirm(null)} className="flex-1 py-2 bg-slate-100 text-slate-600 rounded-xl font-bold text-sm">取消</button>
                <button onClick={confirmDelete} className="flex-1 py-2 bg-red-500 text-white rounded-xl font-bold text-sm">確認刪除</button>
              </div>
            </div>
          </div>
        )}

        {/* 訂購矩陣表格 */}
        <div className="bg-white rounded-2xl shadow-xl overflow-hidden border border-slate-200">
          <div className="overflow-auto max-h-[70vh]">
            <table className="w-full text-left border-separate border-spacing-0">
              <thead className="sticky top-0 z-40">
                <tr className="bg-slate-800 text-white">
                  <th className="p-4 sticky left-0 z-50 bg-slate-800 border-r border-slate-700 text-center text-xs font-black w-24 sm:w-32">成員姓名</th>
                  {DEFAULT_DATES.map(date => (
                    <th key={date} className="p-4 text-center min-w-[80px] sm:min-w-[100px] border-b border-slate-700">
                      <div className="flex flex-col items-center">
                        <span className="text-[10px] text-slate-400 font-normal">週二</span>
                        <span className="text-sm sm:text-base font-black">{date}</span>
                      </div>
                    </th>
                  ))}
                  <th className="p-4 text-right border-b border-slate-700 bg-slate-900 text-green-400 text-xs font-black">應付總額</th>
                </tr>
              </thead>
              <tbody className="divide-y divide-slate-100">
                {stats.memberList.length === 0 ? (
                  <tr>
                    <td colSpan={DEFAULT_DATES.length + 2} className="p-16 text-center text-slate-400 text-sm">
                      尚未匯入資料，請點擊上方「匯入」按鈕。
                    </td>
                  </tr>
                ) : (
                  stats.memberList.map(name => {
                    let bags = 0; let ship = 0;
                    DEFAULT_DATES.forEach(d => { if (orders[name]?.[d]) { bags++; ship += stats.shippingPerPerson[d]; } });
                    return (
                      <tr key={name} className="hover:bg-green-50/40 transition-colors">
                        <td className="p-3 font-bold text-slate-700 sticky left-0 bg-white z-20 border-r border-slate-100 flex items-center justify-between gap-2 shadow-[2px_0_5px_rgba(0,0,0,0.02)]">
                          <span className="truncate">{name}</span>
                          <button 
                            onClick={() => handleDeleteClick(name)} 
                            className="text-slate-300 hover:text-red-500 transition-colors p-1"
                          >
                            <Trash2 size={14} />
                          </button>
                        </td>
                        {DEFAULT_DATES.map(date => (
                          <td key={date} className="p-2 text-center">
                            <button
                              onClick={() => toggleOrder(name, date)}
                              className={`mx-auto w-10 h-10 sm:w-12 sm:h-12 rounded-xl flex items-center justify-center transition-all ${
                                orders[name]?.[date] 
                                  ? 'bg-green-600 text-white shadow-md scale-105' 
                                  : 'bg-slate-50 text-slate-200 hover:bg-slate-200'
                              }`}
                            >
                              {orders[name]?.[date] ? <CheckCircle2 size={24} strokeWidth={3} /> : <Plus size={18} />}
                            </button>
                          </td>
                        ))}
                        <td className="p-3 text-right bg-slate-50/80 sticky right-0 z-10 font-mono font-black text-green-700 text-sm sm:text-base border-l border-slate-100">
                          ${(bags * LETTUCE_PRICE + ship).toLocaleString()}
                        </td>
                      </tr>
                    );
                  })
                )}
              </tbody>
              <tfoot className="sticky bottom-0 z-40 bg-slate-800 text-white text-[10px] sm:text-xs">
                <tr className="font-bold">
                  <td className="p-4 sticky left-0 bg-slate-800 border-r border-slate-700">當週小計</td>
                  {DEFAULT_DATES.map(date => (
                    <td key={date} className="p-4 text-center">
                      <div className={`px-2 py-0.5 rounded-full inline-block ${stats.dailyCount[date] >= MAX_BAGS_PER_WEEK ? 'bg-red-500' : 'bg-green-600'}`}>
                        {stats.dailyCount[date]} / 12
                      </div>
                    </td>
                  ))}
                  <td className="p-4 text-right bg-slate-900 text-gray-400">總計: {Object.values(stats.dailyCount).reduce((a,b)=>a+b,0)}包</td>
                </tr>
                <tr className="bg-slate-700 text-slate-300">
                  <td className="p-2 sticky left-0 bg-slate-700 border-r border-slate-600 text-center">分擔運費</td>
                  {DEFAULT_DATES.map(date => (
                    <td key={date} className="p-2 text-center font-mono">
                      ${stats.shippingPerPerson[date]}
                    </td>
                  ))}
                  <td className="p-2 text-right">均分制</td>
                </tr>
              </tfoot>
            </table>
          </div>
        </div>

        {/* 費用總覽卡片 */}
        <div className="space-y-4">
          <div className="flex items-center gap-2 px-1">
            <Users size={18} className="text-green-600" />
            <h2 className="font-black text-lg text-slate-700">各成員繳費明細</h2>
          </div>
          <div className="grid grid-cols-2 sm:grid-cols-2 lg:grid-cols-4 gap-3 sm:gap-4">
            {stats.memberList.map(name => {
              let bags = 0; let ship = 0;
              DEFAULT_DATES.forEach(d => { if (orders[name]?.[d]) { bags++; ship += stats.shippingPerPerson[d]; } });
              if (bags === 0) return null;
              return (
                <div key={name} className="bg-white p-4 rounded-2xl border border-slate-200 shadow-sm hover:shadow-md transition-all relative overflow-hidden group">
                  <div className="absolute top-0 left-0 w-1 h-full bg-green-500"></div>
                  <div className="flex justify-between items-start mb-2">
                    <span className="font-black text-slate-800">{name}</span>
                    <span className="text-[10px] bg-slate-100 px-2 py-0.5 rounded text-slate-500">{bags}包</span>
                  </div>
                  <div className="text-[11px] text-slate-500 space-y-1">
                    <div className="flex justify-between"><span>菜費:</span><span>${bags * LETTUCE_PRICE}</span></div>
                    <div className="flex justify-between border-b border-dashed pb-1"><span>運費:</span><span>${ship}</span></div>
                    <div className="flex justify-between pt-1 font-black text-green-600 text-base"><span>總計:</span><span>${(bags * LETTUCE_PRICE + ship).toLocaleString()}</span></div>
                  </div>
                </div>
              );
            })}
          </div>
        </div>
      </main>

      <footer className="text-center py-12 text-slate-400 text-[10px] leading-relaxed">
        <p>生菜團購助手 v5 · 互動式網頁管理系統</p>
        <p>資料即時存儲於雲端，不同裝置開啟將同步顯示最新結果。</p>
      </footer>
    </div>
  );
}
```
