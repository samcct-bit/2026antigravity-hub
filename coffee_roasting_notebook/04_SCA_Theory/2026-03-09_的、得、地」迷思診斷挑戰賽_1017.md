        <div id="quiz-screen" class="p-8 hidden">

            </div>

                    的
                </button>
                <button onclick="handleAnswer('得')" class="option-btn w-full py-4 px-6 border-2 border-slate-200 hover:border-indigo-400 hover:bg-indigo-50 rounded-2xl text-xl font-black text-slate-700 transition-all text-center">

                <button onclick="handleAnswer('地')" class="option-btn w-full py-4 px-6 border-2 border-slate-200 hover:border-indigo-400 hover:bg-indigo-50 rounded-2xl text-xl font-black text-slate-700 transition-all text-center">

                    <p id="explanation" class="text-sm font-medium leading-relaxed"></p>
                </div>

                    下一題
                </button>
            </div>
        </div>

        <!-- Result Screen -->
        <div id="result-screen" class="p-8 hidden">
            <div class="text-center mb-6">
                <h2 class="text-3xl font-black text-slate-800 mb-2">測驗完成！</h2>
                <div class="w-32 h-32 rounded-full border-8 border-indigo-500 flex items-center justify-center mx-auto my-6 bg-indigo-50 shadow-inner">
                    <span id="final-score" class="text-5xl font-black text-indigo-600">0</span>
                </div>
                <p id="final-msg" class="text-slate-600 font-bold text-lg mb-6"></p>
            </div>

            <!-- 迷思概念診斷區 -->
            <div class="bg-white border-2 border-slate-100 rounded-2xl p-6 mb-8 shadow-sm">
                <h3 class="text-lg font-black text-slate-800 mb-4 flex items-center">
                    <span class="bg-amber-100 text-amber-600 p-2 rounded-lg mr-2">📊</span> 迷思概念診斷
                </h3>
                <div id="diagnosis-box" class="space-y-4">
                    <!-- 診斷結果會透過 JS 插入在這裡 -->
                </div>
            </div>

            <button onclick="startGame()" class="w-full py-4 bg-indigo-600 text-white rounded-2xl font-bold text-lg hover:bg-indigo-700 transition-all shadow-lg shadow-indigo-200">
                再測驗一次 (隨機換題)
            </button>
        </div>

    </div>

    <script>
        // 題庫 (帶有觀念標籤 type 供診斷使用)
        const allQuestions = [
            { q: "王獻之寫____字，跟人一樣高。", a: "的", type: "de_noun", e: "「字」是名詞，修飾名詞要用白部的「的」。" },
            { q: "毛筆被獻之牢牢____抓在手裡。", a: "地", type: "de_adv", e: "「抓」是動詞，描述動作的狀態要用土部的「地」。" },
            { q: "你寫字時，筆握____牢。", a: "得", type: "de_comp", e: "「握」是動詞，「牢」是動作的結果，中間要用赤部的「得」來補充說明。" },
            { q: "大家讚美他寫的字和父親一樣____好。", a: "得", type: "de_comp", e: "「好」是補充說明前面「寫」這個動作的程度，故用「得」。" },
            { q: "美麗____風景就在鹿谷。", a: "的", type: "de_noun", e: "「風景」是名詞，前面連接形容詞時用「的」。" },
            { q: "他興奮____跳了起來。", a: "地", type: "de_adv", e: "「跳」是動作，說明如何跳（興奮）要用「地」。" },
            { q: "天黑____看不見路。", a: "得", type: "de_comp", e: "形容「黑」到了一種程度（看不見路），故用「得」。" },
            { q: "這是一本很有趣____書。", a: "的", type: "de_noun", e: "「書」是具體的名詞物品，前面用「的」。" },
            { q: "老師大聲____責備頑皮的學生。", a: "地", type: "de_adv", e: "「責備」是動詞，修飾動詞的副詞後面要用「地」。" },
            { q: "這件事辦____非常出色。", a: "得", type: "de_comp", e: "「辦」是動詞，「出色」是辦事結果的補充，中間用「得」。" }
        ];

        let currentQuestions = [];
        let currentIndex = 0;
        let score = 0;
        let answered = false;
        const MAX_QUESTIONS = 5; // 每次測驗5題
        
        // 紀錄錯誤類型，用於迷思診斷
        let errorCounts = {
            de_noun: 0, // 錯「的」
            de_comp: 0, // 錯「得」
            de_adv: 0   // 錯「地」
        };

        function startGame() {
            // 隨機抽選 5 題
            currentQuestions = [...allQuestions].sort(() => Math.random() - 0.5).slice(0, MAX_QUESTIONS);
            currentIndex = 0;
            score = 0;
            errorCounts = { de_noun: 0, de_comp: 0, de_adv: 0 };
            
            document.getElementById('start-screen').classList.add('hidden');
            document.getElementById('result-screen').classList.add('hidden');
            document.getElementById('quiz-screen').classList.remove('hidden');
            
            updateUI();
        }

        function updateUI() {
            answered = false;
            const q = currentQuestions[currentIndex];
            
            // 題目挖空加上底線樣式
            const formattedQ = q.q.replace("____", "<span class='inline-block border-b-4 border-slate-300 w-12 mx-1'></span>");
            document.getElementById('q-text').innerHTML = formattedQ;
            
            document.getElementById('q-count').innerText = `第 ${currentIndex + 1} / ${MAX_QUESTIONS} 題`;
            document.getElementById('score-count').innerText = `得分: ${score}`;
            
            // 更新進度條 (因為是 5 題，每一題佔 20%)
            document.getElementById('progress').style.width = `${((currentIndex + 1) / MAX_QUESTIONS) * 100}%`;
            
            document.getElementById('feedback').classList.add('hidden');
            
            // 恢復按鈕狀態
            const btns = document.querySelectorAll('.option-btn');
            btns.forEach(b => {
                b.disabled = false;
                b.className = "option-btn w-full py-4 px-6 border-2 border-slate-200 hover:border-indigo-400 hover:bg-indigo-50 rounded-2xl text-xl font-black text-slate-700 transition-all text-center";
            });
        }

        function handleAnswer(choice) {
            if (answered) return;
            answered = true;
            
            const q = currentQuestions[currentIndex];
            const feedback = document.getElementById('feedback');
            const feedbackBox = document.getElementById('feedback-box');
            const feedbackMsg = document.getElementById('feedback-msg');
            const explanation = document.getElementById('explanation');
            const nextBtn = document.getElementById('next-btn');
            
            const btns = document.querySelectorAll('.option-btn');
            
            // 判斷對錯
            if (choice === q.a) {
                score += (100 / MAX_QUESTIONS); // 每題 20 分
                feedbackMsg.innerHTML = "✨ 答對了！";
                feedbackBox.className = "p-5 rounded-2xl mb-4 border-2 border-emerald-200 bg-emerald-50 text-emerald-800";
                explanation.className = "text-sm font-medium leading-relaxed text-emerald-700";
                
                btns.forEach(b => {
                    b.disabled = true;
                    if (b.innerText === q.a) {
                        b.classList.add('bg-emerald-500', 'text-white', 'border-emerald-500');
                        b.classList.remove('text-slate-700');
                    } else {
                        b.classList.add('opacity-40');
                    }
                });
            } else {
                // 答錯，記錄迷思類型
                errorCounts[q.type]++;
                
                feedbackMsg.innerHTML = "💡 再想一下喔！";
                feedbackBox.className = "p-5 rounded-2xl mb-4 border-2 border-rose-200 bg-rose-50 text-rose-800";
                explanation.className = "text-sm font-medium leading-relaxed text-rose-700";
                
                btns.forEach(b => {
                    b.disabled = true;
                    if (b.innerText === choice) {
                        b.classList.add('bg-rose-100', 'border-rose-400', 'text-rose-700', 'shake');
                    } else if (b.innerText === q.a) {
                        // 標示出正確答案
                        b.classList.add('border-emerald-500', 'text-emerald-600', 'bg-emerald-50');
                    } else {
                        b.classList.add('opacity-40');
                    }
                });
            }

            // 填入正確答案到題目空格中
            document.getElementById('q-text').innerHTML = q.q.replace("____", `<span class='inline-block border-b-4 ${choice === q.a ? 'border-emerald-500 text-emerald-600' : 'border-rose-500 text-rose-600'} w-12 mx-1 text-center pb-1'>${q.a}</span>`);
            
            explanation.innerText = q.e;
            
            // 處理「下一題」按鈕的文字與樣式變化
            if (currentIndex === currentQuestions.length - 1) {
                nextBtn.innerText = "查看診斷結果";
                nextBtn.className = "w-full py-4 bg-indigo-600 text-white rounded-2xl font-bold hover:bg-indigo-700 transition-all text-lg shadow-lg shadow-indigo-200 animate-pulse";
            } else {
                nextBtn.innerText = "下一題";
                nextBtn.className = "w-full py-4 bg-slate-800 text-white rounded-2xl font-bold hover:bg-slate-700 transition-all text-lg shadow-lg";
            }
            
            feedback.classList.remove('hidden');
            document.getElementById('score-count').innerText = `得分: ${score}`;
        }

        function nextQuestion() {
            if (currentIndex < currentQuestions.length - 1) {
                currentIndex++;
                updateUI();
            } else {
                showResult();
            }
        }

        function showResult() {
            document.getElementById('quiz-screen').classList.add('hidden');
            document.getElementById('result-screen').classList.remove('hidden');
            
            document.getElementById('final-score').innerText = Math.round(score);
            
            const msg = document.getElementById('final-msg');
            if (score === 100) {
                msg.innerText = "太厲害了！你已經完全掌握了！";
            } else if (score >= 60) {
                msg.innerText = "表現不錯！看看下方的診斷可以更進步喔！";
            } else {
                msg.innerText = "多練習幾次，把下方的口訣背起來就沒問題！";
            }

            // 產生迷思診斷報告
            generateDiagnosis();
        }

        function generateDiagnosis() {
            const diagnosisBox = document.getElementById('diagnosis-box');
            diagnosisBox.innerHTML = ''; 
            
            // 如果全部答對
            if (errorCounts.de_noun === 0 && errorCounts.de_comp === 0 && errorCounts.de_adv === 0) {
                diagnosisBox.innerHTML = `
                    <div class="bg-emerald-50 p-4 rounded-xl border border-emerald-100 flex items-start">
                        <span class="text-2xl mr-3">🎉</span>
                        <div>
                            <p class="font-bold text-emerald-800 mb-1">完美無瑕！沒有任何迷思！</p>
                            <p class="text-sm text-emerald-600">你非常清楚「的、得、地」的差異，繼續保持！</p>
                        </div>
                    </div>`;
                return;
            }

            // 有答錯的情況，動態生成建議
            let html = '';
            
            if (errorCounts.de_noun > 0) {
                html += `
                    <div class="bg-blue-50 p-4 rounded-xl border border-blue-100">
                        <p class="font-black text-blue-800 mb-2 flex items-center">
                            <span class="bg-blue-200 text-blue-800 text-xs px-2 py-1 rounded mr-2">迷思 1</span> 搞混了「的」
                        </p>
                        <p class="text-sm text-slate-600 mb-2">你似乎在<strong class="text-slate-800">修飾「名詞（物品/人）」</strong>時選錯了字。</p>
                        <div class="bg-white p-3 rounded-lg text-sm border border-blue-50">
                            <span class="text-blue-600 font-bold">💡 必記口訣：</span>「白部的」後面要接<span class="font-bold underline text-slate-800">名詞</span>。<br>
                            <span class="text-slate-400">例句：</span>美麗<span class="text-blue-500 font-bold">的</span><strong class="text-slate-700">風景</strong>。
                        </div>
                    </div>`;
            }
            
            if (errorCounts.de_comp > 0) {
                html += `
                    <div class="bg-rose-50 p-4 rounded-xl border border-rose-100">
                        <p class="font-black text-rose-800 mb-2 flex items-center">
                            <span class="bg-rose-200 text-rose-800 text-xs px-2 py-1 rounded mr-2">迷思 2</span> 忽略了「得」
                        </p>
                        <p class="text-sm text-slate-600 mb-2">你常忘記在<strong class="text-slate-800">補充說明動作程度或結果</strong>時使用這個字。</p>
                        <div class="bg-white p-3 rounded-lg text-sm border border-rose-50">
                            <span class="text-rose-600 font-bold">💡 必記口訣：</span>動詞 +「得」+ <span class="font-bold underline text-slate-800">程度/結果</span>。<br>
                            <span class="text-slate-400">例句：</span>跑<span class="text-rose-500 font-bold">得</span><strong class="text-slate-700">很快</strong>、辦<span class="text-rose-500 font-bold">得</span><strong class="text-slate-700">很出色</strong>。
                        </div>
                    </div>`;
            }
            
            if (errorCounts.de_adv > 0) {
                html += `
                    <div class="bg-purple-50 p-4 rounded-xl border border-purple-100">
                        <p class="font-black text-purple-800 mb-2 flex items-center">
                            <span class="bg-purple-200 text-purple-800 text-xs px-2 py-1 rounded mr-2">迷思 3</span> 忘記了「地」
                        </p>
                        <p class="text-sm text-slate-600 mb-2">你在描述<strong class="text-slate-800">「如何做某個動作」</strong>時，容易選成其他字。</p>
                        <div class="bg-white p-3 rounded-lg text-sm border border-purple-50">
                            <span class="text-purple-600 font-bold">💡 必記口訣：</span>「土部地」後面要接<span class="font-bold underline text-slate-800">動詞</span>。<br>
                            <span class="text-slate-400">例句：</span>慢慢<span class="text-purple-500 font-bold">地</span><strong class="text-slate-700">走</strong>、興奮<span class="text-purple-500 font-bold">地</span><strong class="text-slate-700">跳</strong>。
                        </div>
                    </div>`;
            }
            
            diagnosisBox.innerHTML = html;
        }
    </script>
</body>
</html>
