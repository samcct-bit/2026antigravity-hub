```react
import React, { useState } from 'react';
import { Upload, Printer, Maximize2 } from 'lucide-react';

export default function App() {
    // 預設卡片資料設定
    const defaultCards = [
        { id: 0, name: "小八貓 (引導者)", src: "小八貓.jpg", defaultScale: 1.6 },
        { id: 1, name: "吉伊卡哇 (行動派)", src: "吉伊卡哇.jpg", defaultScale: 1.8 },
        { id: 2, name: "兔兔 (記錄員)", src: "兔兔.jpg", defaultScale: 1.65 },
        { id: 3, name: "飛鼠 (評量員)", src: "飛鼠.jpg", defaultScale: 1.65 },
        { id: 4, name: "鎧甲先生 (管理者)", src: "鎧甲先生.jpg", defaultScale: 1.85 },
    ];

    const [cards, setCards] = useState(defaultCards);

    // 處理圖片上傳預覽
    const handleImageUpload = (index, event) => {
        const file = event.target.files[0];
        if (file) {
            const url = URL.createObjectURL(file);
            const newCards = [...cards];
            newCards[index].src = url;
            setCards(newCards);
        }
    };

    // 處理圖片縮放（裁切背景）
    const handleScaleChange = (index, value) => {
        const newCards = [...cards];
        newCards[index].defaultScale = parseFloat(value);
        setCards(newCards);
    };

    return (
        <div className="min-h-screen bg-neutral-100 p-4 md:p-8 print:p-0 print:bg-white flex flex-col items-center font-sans text-neutral-800">
            {/* 控制面板 - 列印時會自動隱藏 */}
            <div className="print:hidden w-full max-w-5xl bg-white p-6 rounded-2xl shadow-sm border border-neutral-200 mb-8">
                <div className="flex flex-col md:flex-row justify-between items-start md:items-center mb-6 gap-4">
                    <div>
                        <h1 className="text-2xl font-bold mb-2 flex items-center gap-2 text-neutral-800">
                            🏫 班級幹部卡片 - A4 列印排版工具
                        </h1>
                        <p className="text-neutral-600 text-sm leading-relaxed">
                            系統已預設好 <b>A4 尺寸</b> 與 <b>裁切比例</b>。<br/>
                            請點擊「更換圖片」上傳您的圖檔，並透過滑桿微調放大比例，確認虛線框內只留下中央卡片內容後，點擊「列印」即可存成 PDF 或直接印出。
                        </p>
                    </div>
                    <button
                        onClick={() => window.print()}
                        className="flex items-center gap-2 bg-blue-600 text-white px-6 py-3 rounded-xl font-bold hover:bg-blue-700 transition shadow-sm whitespace-nowrap"
                    >
                        <Printer size={20} />
                        列印 / 存為 PDF
                    </button>
                </div>

                <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-5 gap-4">
                    {cards.map((card, i) => (
                        <div key={card.id} className="bg-neutral-50 p-4 rounded-xl border border-neutral-200 flex flex-col">
                            <h3 className="font-bold text-neutral-700 mb-3 text-sm">{card.name}</h3>
                            
                            <label className="flex items-center justify-center gap-2 bg-white border border-neutral-300 px-3 py-2 rounded-lg cursor-pointer hover:bg-neutral-50 transition mb-4 text-sm font-medium">
                                <Upload size={16} className="text-blue-600" />
                                <span className="text-neutral-700">選擇圖片</span>
                                <input
                                    type="file"
                                    accept="image/*"
                                    onChange={(e) => handleImageUpload(i, e)}
                                    className="hidden"
                                />
                            </label>

                            <div className="mt-auto">
                                <div className="flex justify-between text-xs text-neutral-500 mb-1 font-medium">
                                    <span className="flex items-center gap-1"><Maximize2 size={12}/> 放大裁切</span>
                                    <span>{card.defaultScale.toFixed(2)}x</span>
                                </div>
                                <input
                                    type="range"
                                    min="1" max="3" step="0.05"
                                    value={card.defaultScale}
                                    onChange={(e) => handleScaleChange(i, e.target.value)}
                                    className="w-full accent-blue-600"
                                />
                            </div>
                        </div>
                    ))}
                </div>
            </div>

            {/* A4 列印預覽區域 */}
            <div
                className="bg-white shadow-2xl print:shadow-none relative print:m-0"
                style={{
                    width: '210mm',
                    minHeight: '297mm',
                    padding: '15mm',
                    boxSizing: 'border-box'
                }}
            >
                <div className="grid grid-cols-2 gap-x-[15mm] gap-y-[15mm] justify-items-center align-content-start h-full">
                    {cards.map((card) => (
                        <div
                            key={card.id}
                            className="relative overflow-hidden flex justify-center items-center rounded-sm bg-neutral-50"
                            style={{
                                width: '85mm',      // 設定固定卡片寬度
                                height: '70mm',     // 設定固定卡片高度
                                border: '1.5px dashed #a3a3a3' // 裁切用的虛線邊框
                            }}
                        >
                            <img
                                src={card.src}
                                alt={card.name}
                                className="w-full h-full object-cover transition-transform duration-200"
                                style={{
                                    transform: `scale(${card.defaultScale})`,
                                    transformOrigin: 'center center'
                                }}
                                onError={(e) => {
                                    // 若圖片載入失敗，隱藏破圖並顯示提示文字
                                    e.target.style.opacity = '0';
                                    e.target.nextElementSibling.style.display = 'flex';
                                }}
                                onLoad={(e) => {
                                    // 圖片載入成功，隱藏提示文字
                                    e.target.style.opacity = '1';
                                    if(e.target.nextElementSibling) e.target.nextElementSibling.style.display = 'none';
                                }}
                            />
                            {/* 圖片未上傳或未載入時的提示畫面 */}
                            <div className="absolute inset-0 flex flex-col items-center justify-center text-neutral-400 p-4 text-center print:hidden" style={{display: 'none'}}>
                                <span className="mb-2 font-bold text-neutral-300">圖片未載入</span>
                                <span className="text-xs text-neutral-400">請點擊上方按鈕<br/>上傳【{card.name.split(' ')[0]}】圖片</span>
                            </div>
                        </div>
                    ))}
                    {/* 第6個留空欄位，維持版面平衡 */}
                    <div
                        className="relative flex justify-center items-center"
                        style={{ width: '85mm', height: '70mm' }}
                    >
                        <span className="text-neutral-300 print:hidden text-sm font-medium tracking-wider">（排版留空區）</span>
                    </div>
                </div>
            </div>
        </div>
    );
}
```
