# 📚 SCA 考官理論：快樂學除法

## 📋 對話理論筆記
```react
import React, { useState, useEffect } from 'react';
import { Play, RotateCcw, CheckCircle, HelpCircle, ArrowRight, BookOpen, Star } from 'lucide-react';

const DivisionLearning = () => {
  const [activeTab, setActiveTab] = useState('partitive'); // 'partitive' (等分) or 'quotitive' (包含) or 'quiz' (測驗)
  
  return (
    <div className="min-h-screen bg-sky-50 font-sans selection:bg-yellow-200">
      {/* Header */}
      <header className="bg-white shadow-md border-b-4 border-sky-200 p-4 sticky top-0 z-10">
        <div className="max-w-4xl mx-auto flex items-center justify-between">
          <div className="flex items-center gap-3">
            <div className="bg-yellow-400 p-2 rounded-xl rotate-3 shadow-sm">
              <Star className="text-white w-6 h-6 fill-current" />
            </div>
            <h1 className="text-2xl font-bold text-sky-800 tracking-wide">快樂學除法</h1>
          </div>
          <div className="text-sm font-medium text-sky-600 bg-sky-100 px-3 py-1 rounded-full hidden sm:block">
            適合年級：國小三年級
          </div>
        </div>
      </header>

      {/* Main Content */}
      <main className="max-w-4xl mx-auto p-4 md:p-6">
        {/* Navigation Tabs */}
        <div className="flex flex-wrap gap-2 mb-8 justify-center sm:justify-start">
          <TabButton 
            active={activeTab === 'partitive'} 
            onClick={() => setActiveTab('partitive')}
            color="blue"

          >

          </TabButton>
          <TabButton 
            active={activeTab === 'quotitive'} 
            onClick={() => setActiveTab('quotitive')}
            color="green"
            icon={<BookOpen size={20} />}
          >
            包含除 (裝裝看)
          </TabButton>
          <TabButton 
            active={activeTab === 'quiz'} 
            onClick={() => setActiveTab('quiz')}
            color="purple"
            icon={<HelpCircle size={20} />}
          >
            小挑戰 (測驗)
          </TabButton>
        </div>

        {/* Content Area */}
        <div className="bg-white rounded-3xl shadow-xl border-2 border-slate-100 overflow-hidden min-h-[500px]">
          {activeTab === 'partitive' && <PartitiveView />}
          {activeTab === 'quotitive' && <QuotitiveView />}
          {activeTab === 'quiz' && <QuizView />}
        </div>
      </main>
    </div>
  );
};

// --- Components ---

const TabButton = ({ active, onClick, children, color, icon }) => {
  const colorStyles = {
    blue: active ? 'bg-blue-500 text-white shadow-blue-200' : 'bg-white text-blue-500 hover:bg-blue-50',
    green: active ? 'bg-emerald-500 text-white shadow-emerald-200' : 'bg-white text-emerald-500 hover:bg-emerald-50',
    purple: active ? 'bg-purple-500 text-white shadow-purple-200' : 'bg-white text-purple-500 hover:bg-purple-50',
  };

  return (
    <button
      onClick={onClick}
      className={`
        px-6 py-3 rounded-2xl font-bold text-lg transition-all duration-300 flex items-center gap-2 shadow-lg
        ${active ? 'transform -translate-y-1' : 'opacity-80 hover:opacity-100'}
        ${colorStyles[color]}
      `}
    >
      {icon}
      {children}
    </button>
  );
};

// --- Partitive Division (等分除) View ---
const PartitiveView = () => {
  const totalItems = 12;
  const groups = 3; // 3 plates
  const [distributedCount, setDistributedCount] = useState(0);
  const [isAnimating, setIsAnimating] = useState(false);
  const [showMath, setShowMath] = useState(false);

  useEffect(() => {
    let interval;
    if (isAnimating && distributedCount < totalItems) {
      interval = setInterval(() => {
        setDistributedCount(prev => prev + 1);
      }, 500);
    } else if (distributedCount >= totalItems) {
      setIsAnimating(false);
      setShowMath(true);
    }
    return () => clearInterval(interval);
  }, [isAnimating, distributedCount]);

  const startAnimation = () => {
    setDistributedCount(0);
    setShowMath(false);
    setIsAnimating(true);
  };

  const reset = () => {
    setDistributedCount(0);
    setIsAnimating(false);
    setShowMath(false);
  };

  return (
    <div className="p-6 md:p-8">
      <div className="bg-blue-50 rounded-2xl p-6 mb-6 border border-blue-100">
        <h2 className="text-2xl font-bold text-blue-800 mb-2 flex items-center gap-2">
          <span className="bg-blue-200 p-1 rounded text-2xl">🍎</span> 情境：大家平分
        </h2>
        <p className="text-lg text-blue-700">
          有 <strong className="text-2xl text-red-500">{totalItems}</strong> 顆蘋果，
          <span className="border-b-4 border-blue-300 mx-1">平分</span>到 
          <strong className="text-2xl text-blue-600 mx-1"> {groups} </strong> 個盤子上。
          <br/>
          <span className="text-slate-500 text-base mt-2 block">問題：每個盤子可以分到幾顆？</span>
        </p>
      </div>

      {/* Visual Area */}
      <div className="relative h-64 bg-slate-50 rounded-xl border-2 border-dashed border-slate-300 mb-6 p-4 flex flex-col items-center justify-end">
        {/* Source Pile (Invisible logic, just shows decreasing count) */}
        <div className="absolute top-4 left-4 bg-white px-3 py-1 rounded-lg shadow text-sm font-bold text-slate-500">
          還沒分的蘋果：{totalItems - distributedCount}
        </div>

        <div className="flex justify-around w-full items-end gap-4">
          {[...Array(groups)].map((_, plateIndex) => (
            <div key={plateIndex} className="relative w-32 h-32 flex flex-col items-center justify-end">
              {/* Plate */}
              <div className="w-full h-12 bg-gray-200 rounded-[50%] border-4 border-gray-300 shadow-inner absolute bottom-0 z-0"></div>
              {/* Apples on plate */}
              <div className="mb-4 z-10 flex flex-wrap justify-center content-end w-24 gap-1 transform translate-y-2">
                {[...Array(distributedCount)].map((_, i) => {
                  if (i % groups === plateIndex) {
                    return (
                      <div key={i} className="w-8 h-8 bg-red-500 rounded-full border-b-4 border-red-700 shadow-sm animate-bounce-short">
                        <div className="w-2 h-2 bg-white opacity-40 rounded-full ml-1 mt-1"></div>
                      </div>
                    );
                  }
                  return null;
                })}
              </div>
              <div className="absolute -bottom-8 font-bold text-gray-500">盤子 {plateIndex + 1}</div>
            </div>
          ))}
        </div>
      </div>

      {/* Controls */}
      <div className="flex gap-4 justify-center mb-8">
        <button 
          onClick={startAnimation} 
          disabled={isAnimating || showMath}
          className="bg-blue-600 hover:bg-blue-700 text-white px-8 py-3 rounded-xl font-bold text-lg shadow-lg shadow-blue-200 disabled:opacity-50 disabled:cursor-not-allowed flex items-center gap-2"
        >
          <Play size={20} fill="currentColor" /> 開始平分
        </button>
        <button 
          onClick={reset}
          className="bg-white border-2 border-slate-200 text-slate-600 hover:bg-slate-50 px-6 py-3 rounded-xl font-bold text-lg flex items-center gap-2"
        >
          <RotateCcw size={20} /> 重來
        </button>
      </div>

      {/* Math Explanation */}
      {showMath && (
        <div className="animate-fade-in bg-yellow-50 border-2 border-yellow-200 rounded-2xl p-6 text-center relative overflow-hidden">
          <div className="absolute top-0 left-0 w-full h-2 bg-yellow-300"></div>
          
          <div className="grid md:grid-cols-2 gap-8 items-center">
            <div>
              <h3 className="text-gray-500 font-bold mb-2">除法算式</h3>
              <div className="text-3xl font-bold text-slate-800 tracking-wider">
                {totalItems} <span className="text-slate-400">÷</span> <span className="text-blue-600">{groups}</span> = <span className="text-red-600 text-4xl">4</span>
              </div>
              <p className="text-sm text-slate-500 mt-2">總數 ÷ 盤子數 = 每盤幾顆</p>
            </div>

            <div className="bg-white p-4 rounded-xl shadow-sm border border-yellow-100">
              <h3 className="text-blue-600 font-bold mb-2 flex items-center justify-center gap-2">
                 對應的乘法 <ArrowRight size={16} />
              </h3>
              <div className="text-3xl font-bold text-slate-800 tracking-wider py-2">
                ( <span className="text-red-600 animate-pulse">4</span> ) <span className="text-slate-400">×</span> <span className="text-blue-600">{groups}</span> = {totalItems}
              </div>
              <div className="text-sm text-slate-500 mt-1 space-y-1">
                <p>我們不知道「每盤幾顆」</p>
                <p>所以<span className="text-red-500 font-bold">括號 (未知數)</span> 放在前面！</p>
              </div>
            </div>
          </div>
        </div>
      )}
    </div>
  );
};

// --- Quotitive Division (包含除) View ---
const QuotitiveView = () => {
  const totalItems = 12;
  const perGroup = 4; // 4 apples per bag
  const [groupedCount, setGroupedCount] = useState(0); // How many groups formed
  const [isAnimating, setIsAnimating] = useState(false);
  const [showMath, setShowMath] = useState(false);

  useEffect(() => {
    let interval;
    const totalGroups = totalItems / perGroup;
    
    if (isAnimating && groupedCount < totalGroups) {
      interval = setInterval(() => {
        setGroupedCount(prev => prev + 1);
      }, 800);
    } else if (groupedCount >= totalGroups) {
      setIsAnimating(false);
      setShowMath(true);
    }
    return () => clearInterval(interval);
  }, [isAnimating, groupedCount]);

  const startAnimation = () => {
    setGroupedCount(0);
    setShowMath(false);
    setIsAnimating(true);
  };

  const reset = () => {
    setGroupedCount(0);
    setIsAnimating(false);
    setShowMath(false);
  };

  return (
    <div className="p-6 md:p-8">
      <div className="bg-emerald-50 rounded-2xl p-6 mb-6 border border-emerald-100">
        <h2 className="text-2xl font-bold text-emerald-800 mb-2 flex items-center gap-2">
          <span className="bg-emerald-200 p-1 rounded text-2xl">🛍️</span> 情境：裝進袋子
        </h2>
        <p className="text-lg text-emerald-700">
          有 <strong className="text-2xl text-red-500">{totalItems}</strong> 顆蘋果，
          <span className="border-b-4 border-emerald-300 mx-1">每 {perGroup} 顆</span>裝一袋。
          <br/>
          <span className="text-slate-500 text-base mt-2 block">問題：總共可以裝成幾袋？</span>
        </p>
      </div>

      {/* Visual Area */}
      <div className="relative h-64 bg-slate-50 rounded-xl border-2 border-dashed border-slate-300 mb-6 p-4 flex flex-col justify-center items-center">
        
        {/* Apples Grid */}
        <div className="flex flex-wrap justify-center gap-4 w-full max-w-2xl">
           {/* We render 3 potential groups. We show them based on animation state */}
           {[...Array(totalItems / perGroup)].map((_, groupIndex) => (
             <div 
               key={groupIndex} 
               className={`
                 relative p-2 rounded-xl transition-all duration-500
                 ${groupIndex < groupedCount ? 'bg-emerald-100 border-4 border-emerald-400 scale-105 shadow-md' : 'bg-transparent border-4 border-transparent'}
               `}
             >
                {/* Bag Label */}
                <div className={`
                  absolute -top-4 left-1/2 transform -translate-x-1/2 bg-emerald-500 text-white text-xs px-2 py-1 rounded-full transition-opacity duration-300
                  ${groupIndex < groupedCount ? 'opacity-100' : 'opacity-0'}
                `}>
                  第 {groupIndex + 1} 袋
                </div>

               <div className="grid grid-cols-2 gap-1 w-20 h-20">
                 {[...Array(perGroup)].map((_, i) => (
                   <div key={i} className="w-8 h-8 bg-red-500 rounded-full border-b-4 border-red-700 shadow-sm">
                      <div className="w-2 h-2 bg-white opacity-40 rounded-full ml-1 mt-1"></div>
                   </div>
                 ))}
               </div>
             </div>
           ))}
        </div>
      </div>

      {/* Controls */}
      <div className="flex gap-4 justify-center mb-8">
        <button 
          onClick={startAnimation} 
          disabled={isAnimating || showMath}
          className="bg-emerald-600 hover:bg-emerald-700 text-white px-8 py-3 rounded-xl font-bold text-lg shadow-lg shadow-emerald-200 disabled:opacity-50 disabled:cursor-not-allowed flex items-center gap-2"
        >
          <Play size={20} fill="currentColor" /> 開始裝袋
        </button>
        <button 
          onClick={reset}
          className="bg-white border-2 border-slate-200 text-slate-600 hover:bg-slate-50 px-6 py-3 rounded-xl font-bold text-lg flex items-center gap-2"
        >
          <RotateCcw size={20} /> 重來
        </button>
      </div>

      {/* Math Explanation */}
      {showMath && (
        <div className="animate-fade-in bg-yellow-50 border-2 border-yellow-200 rounded-2xl p-6 text-center relative overflow-hidden">
          <div className="absolute top-0 left-0 w-full h-2 bg-yellow-300"></div>
          
          <div className="grid md:grid-cols-2 gap-8 items-center">
            <div>
              <h3 className="text-gray-500 font-bold mb-2">除法算式</h3>
              <div className="text-3xl font-bold text-slate-800 tracking-wider">
                {totalItems} <span className="text-slate-400">÷</span> <span className="text-emerald-600">{perGroup}</span> = <span className="text-red-600 text-4xl">{totalItems/perGroup}</span>
              </div>
              <p className="text-sm text-slate-500 mt-2">總數 ÷ 每袋幾顆 = 幾袋</p>
            </div>

            <div className="bg-white p-4 rounded-xl shadow-sm border border-yellow-100">
              <h3 className="text-emerald-600 font-bold mb-2 flex items-center justify-center gap-2">
                 對應的乘法 <ArrowRight size={16} />
              </h3>
              <div className="text-3xl font-bold text-slate-800 tracking-wider py-2">
                <span className="text-emerald-600">{perGroup}</span> <span className="text-slate-400">×</span> ( <span className="text-red-600 animate-pulse">{totalItems/perGroup}</span> ) = {totalItems}
              </div>
              <div className="text-sm text-slate-500 mt-1 space-y-1">
                <p>我們不知道「有幾袋」</p>
                <p>所以<span className="text-red-500 font-bold">括號 (未知數)</span> 放在後面！</p>
              </div>
            </div>
          </div>
        </div>
      )}
    </div>
  );
};

// --- Quiz View ---
const QuizView = () => {
  const [question, setQuestion] = useState(null);
  const [selectedAnswer, setSelectedAnswer] = useState(null);
  const [feedback, setFeedback] = useState(null); // 'correct' or 'wrong'

  const questions = [
    {
      text: "有 15 個布丁，平分給 3 個人，每個人可以吃幾個？",
      type: "partitive",
      total: 15,
      known: 3,
      correctEquation: "( ? ) × 3 = 15",
      wrongEquation: "3 × ( ? ) = 15"
    },
    {
      text: "有 24 顆糖果，每 6 顆裝一盒，可以裝成幾盒？",
      type: "quotitive",
      total: 24,
      known: 6,
      correctEquation: "6 × ( ? ) = 24",
      wrongEquation: "( ? ) × 6 = 24"
    },
    {
      text: "老師有 18 枝筆，平分給 6 位小朋友，每位小朋友拿幾枝？",
      type: "partitive",
      total: 18,
      known: 6,
      correctEquation: "( ? ) × 6 = 18",
      wrongEquation: "6 × ( ? ) = 18"
    },
    {
      text: "一條繩子長 20 公分，每 5 公分剪一段，可以剪成幾段？",
      type: "quotitive",
      total: 20,
      known: 5,
      correctEquation: "5 × ( ? ) = 20",
      wrongEquation: "( ? ) × 5 = 20"
    }
  ];

  const generateQuestion = () => {
    const randomQ = questions[Math.floor(Math.random() * questions.length)];
    setQuestion(randomQ);
    setSelectedAnswer(null);
    setFeedback(null);
  };

  useEffect(() => {
    generateQuestion();
  }, []);

  const handleAnswer = (isCorrectEquation) => {
    setSelectedAnswer(isCorrectEquation ? 'correct' : 'wrong');
    setFeedback(isCorrectEquation ? 'correct' : 'wrong');
  };

  if (!question) return <div>載入中...</div>;

  return (
    <div className="p-6 md:p-12 flex flex-col items-center max-w-2xl mx-auto">
      <div className="bg-purple-100 p-4 rounded-full mb-6">
        <HelpCircle size={40} className="text-purple-600" />
      </div>
      
      <h2 className="text-2xl md:text-3xl font-bold text-center text-slate-800 mb-8 leading-relaxed">
        {question.text}
      </h2>

      <p className="text-slate-500 mb-6 font-bold">請問乘法算式該怎麼寫？</p>

      <div className="grid grid-cols-1 md:grid-cols-2 gap-4 w-full mb-8">
        {/* We randomize button order visually but logic keeps track */}
        <QuizOption 
          equation={question.correctEquation} 
          onClick={() => handleAnswer(true)} 
          disabled={feedback !== null}
          state={feedback === 'correct' && selectedAnswer === 'correct' ? 'correct' : (feedback === 'wrong' && selectedAnswer === 'wrong' ? 'dimmed' : null)}
        />
        <QuizOption 
          equation={question.wrongEquation} 
          onClick={() => handleAnswer(false)}
          disabled={feedback !== null}
          state={feedback === 'wrong' && selectedAnswer === 'wrong' ? 'wrong' : (feedback === 'correct' ? 'dimmed' : null)}
        />
      </div>

      {feedback === 'correct' && (
        <div className="animate-bounce-in bg-green-100 text-green-700 px-6 py-4 rounded-xl flex items-center gap-3 mb-6">
          <CheckCircle size={24} />
          <div>
            <span className="font-bold text-lg">答對了！</span>
            <p className="text-sm">
              {question.type === 'partitive' 
                ? '因為是「平分」，我們不知道「每份有多少」，所以括號在前面！' 
                : '因為是「分裝」，我們不知道「有幾份」，所以括號在後面！'}
            </p>
          </div>
        </div>
      )}

      {feedback === 'wrong' && (
        <div className="bg-red-100 text-red-700 px-6 py-4 rounded-xl flex items-center gap-3 mb-6">
          <div className="font-bold text-lg">再想一下...</div>
          <p className="text-sm">
             {question.type === 'partitive' ? '這是平分題目，要算的是「每個人拿幾個」喔。' : '這是包裝題目，要算的是「有幾包」喔。'}
          </p>
        </div>
      )}

      {feedback && (
        <button 
          onClick={generateQuestion}
          className="bg-purple-600 text-white px-8 py-3 rounded-full font-bold text-lg shadow-lg hover:bg-purple-700 transition-colors"
        >
          下一題
        </button>
      )}
    </div>
  );
};

const QuizOption = ({ equation, onClick, disabled, state }) => {
  let bgClass = "bg-white border-2 border-slate-200 hover:border-purple-400 hover:bg-purple-50 text-slate-700";
  
  if (state === 'correct') bgClass = "bg-green-500 text-white border-green-600 shadow-green-200 shadow-lg scale-105";
  if (state === 'wrong') bgClass = "bg-red-500 text-white border-red-600 opacity-50";
  if (state === 'dimmed') bgClass = "bg-slate-100 text-slate-400 border-slate-100 opacity-50";

  return (
    <button
      onClick={onClick}
      disabled={disabled}
      className={`
        py-6 px-4 rounded-2xl text-2xl font-bold transition-all duration-300
        ${bgClass}
      `}
    >
      {equation}
    </button>
  );
};

export default DivisionLearning;
```
