## 📋 對話理論筆記
```react
import React, { useState, useEffect, useMemo } from 'react';
import { initializeApp } from 'firebase/app';
import { getAuth, signInAnonymously, onAuthStateChanged } from 'firebase/auth';
import { getFirestore, collection, doc, onSnapshot, setDoc, updateDoc, deleteDoc } from 'firebase/firestore';
import { 
  Users, 
  ShoppingCart, 
  Truck, 
  Calendar, 
  CheckCircle2, 
  AlertCircle,
  Plus,
  Trash2,
  Loader2,
  XCircle
} from 'lucide-react';

// --- Firebase 配置 ---
const firebaseConfig = JSON.parse(__firebase_config);
const app = initializeApp(firebaseConfig);
const auth = getAuth(app);
const db = getFirestore(app);
const appId = typeof __app_id !== 'undefined' ? __app_id : 'lettuce-app-v2';

const LETTUCE_PRICE = 230;
const WEEKLY_SHIPPING = 300;
const MAX_BAGS_PER_WEEK = 12;

// 訂購日期清單
const DEFAULT_DATES = ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"];

// 系統預設成員 (用於第一次初始化)
const INITIAL_USERS = [
  "金成", "蕙敏", "珮萱", "淑芬", "阿姿", "嘉鎂", "斯涵", "明智",
  "翠花", "政君", "睿婷", "亭愷", "俊明", "惠偵"
];

export default function App() {
  const [user, setUser] = useState(null);
  const [orders, setOrders] = useState({}); // 格式: { [姓名]: { [日期]: boolean } }
  const [loading, setLoading] = useState(true);
  const [newMemberName, setNewMemberName] = useState("");
  const [statusMsg, setStatusMsg] = useState(null);

  // 1. 身份認證 (匿名登入)
  useEffect(() => {
    const initAuth = async () => {
      try {
        await signInAnonymously(auth);
      } catch (e) { console.error("Auth failed", e); }
    };
    initAuth();
    const unsubscribe = onAuthStateChanged(auth, setUser);
    return () => unsubscribe();
  }, []);

  // 2. 即時資料同步 (Snapshot)
  useEffect(() => {
    if (!user) return;

    const dataPath = `artifacts/${appId}/public/data/lettuce_orders`;
    const unsub = onSnapshot(collection(db, dataPath), (snapshot) => {
      const newOrders = {};
      if (snapshot.empty) {
        // 如果完全沒資料，執行初始化
        initializeData();
      } else {
        snapshot.docs.forEach(d => {
          newOrders[d.id] = d.data().orders || {};
        });
        setOrders(newOrders);
        setLoading(false);
      }
    }, (err) => {
      console.error("Firestore error:", err);
      setLoading(false);
    });

    return () => unsub();
  }, [user]);

  // 3. 初始化資料庫
  const initializeData = async () => {
    for (const name of INITIAL_USERS) {
      const initialOrders = {};
      DEFAULT_DATES.forEach(date => {
        // 根據使用者提供的規則預填資料
        if (["金成", "蕙敏", "珮萱", "淑芬", "阿姿", "嘉鎂", "斯涵", "明智"].includes(name)) initialOrders[date] = true;
        if (name === "翠花" && ["2/24", "3/3", "3/10", "3/24"].includes(date)) initialOrders[date] = true;
        if (["政君", "睿婷", "亭愷"].includes(name) && ["3/3", "3/17", "3/31"].includes(date)) initialOrders[date] = true;
        if (["俊明", "惠偵"].includes(name) && ["2/24", "3/10", "3/24"].includes(date)) initialOrders[date] = true;
      });
      await setDoc(doc(db, `artifacts/${appId}/public/data/lettuce_orders`, name), { orders: initialOrders });
    }
  };

  // 4. 計算邏輯 (Memoized)
  const stats = useMemo(() => {
    const dailyCount = {};
    const shippingPerPerson = {};
    const memberList = Object.keys(orders).sort();

    DEFAULT_DATES.forEach(date => {
      let count = 0;
      memberList.forEach(m => { if (orders[m]?.[date]) count++; });
      dailyCount[date] = count;

      // 運費規則
      if (count === 12) shippingPerPerson[date] = 25;
      else if (count === 11) shippingPerPerson[date] = 27;
      else if (count === 10) shippingPerPerson[date] = 30;
      else if (count > 0) shippingPerPerson[date] = Math.round(WEEKLY_SHIPPING / count);
      else shippingPerPerson[date] = 0;
    });

    return { dailyCount, shippingPerPerson, memberList };
  }, [orders]);

  // 5. 操作處理
  const toggleOrder = async (name, date) => {
    const isOrdered = orders[name]?.[date] || false;
    
    // 檢查上限
    if (!isOrdered && stats.dailyCount[date] >= MAX_BAGS_PER_WEEK) {
      showToast(`${date} 已達上限 ${MAX_BAGS_PER_WEEK} 包！`, 'error');
      return;
    }

    const docRef = doc(db, `artifacts/${appId}/public/data/lettuce_orders`, name);
    try {
      await updateDoc(docRef, { [`orders.${date}`]: !isOrdered });
    } catch (e) {
      // 若文件不存在則創建
      await setDoc(docRef, { orders: { [date]: !isOrdered } }, { merge: true });
    }
  };

  const addMember = async () => {
    const name = newMemberName.trim();
    if (!name) return;
    if (orders[name]) {
      showToast("成員已存在", "error");
      return;
    }
    setLoading(true);
    try {
      await setDoc(doc(db, `artifacts/${appId}/public/data/lettuce_orders`, name), { orders: {} });
      setNewMemberName("");
      showToast(`成功新增：${name}`);
    } catch (e) {
      showToast("新增失敗", "error");
    } finally {
      setLoading(false);
    }
  };

  const deleteMember = async (name) => {
    if (!window.confirm(`確定要移除「${name}」嗎？所有的訂購紀錄將消失。`)) return;
    await deleteDoc(doc(db, `artifacts/${appId}/public/data/lettuce_orders`, name));
    showToast(`已移除：${name}`);
  };

  const showToast = (text, type = 'success') => {
    setStatusMsg({ text, type });
    setTimeout(() => setStatusMsg(null), 3000);
  };

  if (loading && Object.keys(orders).length === 0) {
    return (
      <div className="flex flex-col items-center justify-center min-h-screen bg-green-50">
        <Loader2 className="w-10 h-10 text-green-600 animate-spin mb-4" />
        <p className="text-green-800 font-bold">連線中，請稍候...</p>
      </div>
    );
  }

  return (
    <div className="min-h-screen bg-gray-50 pb-12">
      {/* Header */}
      <nav className="bg-green-700 text-white shadow-xl sticky top-0 z-30 px-4 py-3">
        <div className="max-w-6xl mx-auto flex flex-col sm:flex-row items-center justify-between gap-4">
          <div className="flex items-center gap-3">
            <div className="bg-white p-2 rounded-full">
              <ShoppingCart className="text-green-700 w-6 h-6" />
            </div>
            <div>
              <h1 className="text-xl font-black">生菜訂購助手</h1>
              <p className="text-xs text-green-100 opacity-80">每週二到貨 · 每人上限 12 包</p>
            </div>
          </div>

          <div className="flex w-full sm:w-auto gap-2">
            <input 
              className="flex-1 sm:w-48 px-4 py-2 rounded-lg text-gray-800 text-sm focus:ring-2 focus:ring-green-400 outline-none"
              placeholder="輸入新成員姓名"
              value={newMemberName}
              onChange={(e) => setNewMemberName(e.target.value)}
              onKeyPress={(e) => e.key === 'Enter' && addMember()}
            />
            <button 
              onClick={addMember}
              className="bg-green-500 hover:bg-green-400 px-4 py-2 rounded-lg font-bold flex items-center gap-1 transition-all"
            >
              <Plus size={18} /> 新增
            </button>
          </div>
        </div>
      </nav>

      <main className="max-w-6xl mx-auto px-4 py-8 space-y-8">
        {/* Toast */}
        {statusMsg && (
          <div className={`fixed top-20 right-4 z-50 flex items-center gap-2 px-6 py-3 rounded-2xl shadow-2xl animate-in fade-in slide-in-from-top-4 ${
            statusMsg.type === 'error' ? 'bg-red-600 text-white' : 'bg-green-600 text-white'
          }`}>
            {statusMsg.type === 'error' ? <XCircle size={20} /> : <CheckCircle2 size={20} />}
            <span className="font-bold">{statusMsg.text}</span>
          </div>
        )}

        {/* 訂購矩陣表格 */}
        <div className="bg-white rounded-3xl shadow-2xl overflow-hidden border border-gray-200">
          <div className="overflow-x-auto">
            <table className="w-full text-left border-collapse">
              <thead>
                <tr className="bg-gray-100 text-gray-600 text-xs font-black uppercase tracking-widest">
                  <th className="p-5 sticky left-0 bg-gray-100 z-20 border-r w-32 text-center">姓名</th>
                  {DEFAULT_DATES.map(date => (
                    <th key={date} className="p-5 text-center min-w-[100px] border-b">
                      <div className="flex flex-col items-center gap-1">
                        <span className="text-gray-400 font-medium">週二</span>
                        <span className="text-lg text-gray-800">{date}</span>
                      </div>
                    </th>
                  ))}
                  <th className="p-5 text-right border-b bg-green-50 text-green-800">應付總額</th>
                </tr>
              </thead>
              <tbody className="divide-y divide-gray-100">
                {stats.memberList.map(name => {
                  let totalBags = 0;
                  let totalShip = 0;
                  DEFAULT_DATES.forEach(d => {
                    if (orders[name]?.[d]) {
                      totalBags++;
                      totalShip += stats.shippingPerPerson[d];
                    }
                  });
                  const totalAmt = (totalBags * LETTUCE_PRICE) + totalShip;

                  return (
                    <tr key={name} className="group hover:bg-green-50/30 transition-all">
                      <td className="p-4 font-black text-gray-700 sticky left-0 bg-white z-10 border-r flex items-center justify-between gap-2">
                        <span>{name}</span>
                        <button 
                          onClick={() => deleteMember(name)}
                          className="opacity-0 group-hover:opacity-100 text-gray-300 hover:text-red-500 p-1 rounded transition-all"
                        >
                          <Trash2 size={14} />
                        </button>
                      </td>
                      {DEFAULT_DATES.map(date => (
                        <td key={date} className="p-3 text-center">
                          <button
                            onClick={() => toggleOrder(name, date)}
                            className={`mx-auto w-12 h-12 rounded-2xl flex items-center justify-center transition-all ${
                              orders[name]?.[date] 
                                ? 'bg-green-600 text-white shadow-lg shadow-green-200 scale-110' 
                                : 'bg-gray-50 text-gray-200 hover:bg-gray-200 hover:text-gray-400'
                            }`}
                          >
                            {orders[name]?.[date] ? <CheckCircle2 size={28} strokeWidth={3} /> : <Plus size={20} />}
                          </button>
                        </td>
                      ))}
                      <td className="p-4 text-right bg-green-50/50">
                        <span className="font-mono font-black text-green-700 text-lg">
                          ${totalAmt.toLocaleString()}
                        </span>
                      </td>
                    </tr>
                  );
                })}
              </tbody>
              <tfoot className="bg-gray-800 text-white font-bold">
                <tr>
                  <td className="p-5 sticky left-0 bg-gray-800 border-r">當週總量</td>
                  {DEFAULT_DATES.map(date => (
                    <td key={date} className="p-5 text-center">
                      <div className={`px-2 py-1 rounded-lg text-sm ${
                        stats.dailyCount[date] >= MAX_BAGS_PER_WEEK ? 'bg-red-500' : 'bg-green-600'
                      }`}>
                        {stats.dailyCount[date]} / {MAX_BAGS_PER_WEEK}
                      </div>
                    </td>
                  ))}
                  <td className="p-5 text-right text-gray-400 text-xs">共 {Object.values(stats.dailyCount).reduce((a,b)=>a+b, 0)} 包</td>
                </tr>
                <tr>
                  <td className="p-5 sticky left-0 bg-gray-800 border-r text-xs text-gray-400">分擔運費</td>
                  {DEFAULT_DATES.map(date => (
                    <td key={date} className="p-5 text-center text-blue-300 font-mono">
                      <div className="flex items-center justify-center gap-1">
                        <Truck size={12} /> ${stats.shippingPerPerson[date]}
                      </div>
                    </td>
                  ))}
                  <td className="p-5 text-right text-sm">運費總計: $1,800</td>
                </tr>
              </tfoot>
            </table>
          </div>
        </div>

        {/* 費用明細卡片區 */}
        <div className="space-y-4">
          <div className="flex items-center gap-2">
            <div className="bg-blue-600 p-2 rounded-lg">
              <Users className="text-white w-5 h-5" />
            </div>
            <h2 className="text-xl font-black text-gray-800">繳費明細統計</h2>
          </div>
          
          <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
            {stats.memberList.map(name => {
              let bags = 0;
              let shipTotal = 0;
              DEFAULT_DATES.forEach(d => {
                if (orders[name]?.[d]) {
                  bags++;
                  shipTotal += stats.shippingPerPerson[d];
                }
              });
              if (bags === 0) return null;

              return (
                <div key={name} className="bg-white p-5 rounded-2xl border border-gray-200 shadow-sm hover:shadow-xl hover:border-green-300 transition-all">
                  <div className="flex justify-between items-center mb-4">
                    <span className="font-black text-xl text-gray-800">{name}</span>
                    <span className="bg-green-100 text-green-700 px-3 py-1 rounded-full text-xs font-black">
                      共 {bags} 包
                    </span>
                  </div>
                  <div className="space-y-2 text-sm">
                    <div className="flex justify-between text-gray-500">
                      <span>生菜費 ({bags}×230)</span>
                      <span>${bags * LETTUCE_PRICE}</span>
                    </div>
                    <div className="flex justify-between text-gray-500">
                      <span>累計運費</span>
                      <span>${shipTotal}</span>
                    </div>
                    <div className="h-px bg-gray-100 my-2" />
                    <div className="flex justify-between items-end">
                      <span className="font-bold text-gray-400">應付金額</span>
                      <span className="text-2xl font-black text-green-600 font-mono">
                        ${(bags * LETTUCE_PRICE + shipTotal).toLocaleString()}
                      </span>
                    </div>
                  </div>
                </div>
              );
            })}
          </div>
        </div>

        {/* 底部說明 */}
        <div className="grid grid-cols-1 md:grid-cols-2 gap-6 opacity-60">
          <div className="bg-white p-6 rounded-2xl border border-gray-200 flex items-start gap-4">
            <AlertCircle className="text-orange-500 shrink-0" />
            <div className="text-sm text-gray-600 leading-relaxed">
              <p className="font-bold text-gray-800 mb-1">規則提醒：</p>
              <li>每週二到貨，每週上限 12 包生菜。</li>
              <li>運費固定每週 $300，由當週訂購者均分。</li>
              <li>10人份：$30/人；11人份：$27/人；12人份：$25/人。</li>
            </div>
          </div>
          <div className="bg-white p-6 rounded-2xl border border-gray-200 flex items-start gap-4">
            <Calendar className="text-blue-500 shrink-0" />
            <div className="text-sm text-gray-600 leading-relaxed">
              <p className="font-bold text-gray-800 mb-1">使用方式：</p>
              <li>點擊上方表格方塊可新增/取消訂購。</li>
              <li>新增名單會同步至所有裝置。</li>
              <li>若名單有誤，可將滑鼠移至姓名處點擊垃圾桶移除。</li>
            </div>
          </div>
        </div>
      </main>
    </div>
  );
}
```
