## 📋 對話理論筆記
```react
import React, { useState, useEffect, useMemo } from 'react';
import { initializeApp } from 'firebase/app';
import { getAuth, signInAnonymously, onAuthStateChanged } from 'firebase/auth';
import { getFirestore, collection, doc, onSnapshot, setDoc, updateDoc, deleteDoc, writeBatch } from 'firebase/firestore';
import { 
  Users, 
  ShoppingCart, 
  Truck, 
  Calendar, 
  CheckCircle2, 
  AlertCircle,
  Plus,
  Trash2,
  Loader2,
  Database,
  Lock,
  Unlock,
  Settings,
  X,
  CreditCard,
  Check
} from 'lucide-react';

// --- Firebase 配置 ---
const firebaseConfig = JSON.parse(__firebase_config);
const app = initializeApp(firebaseConfig);
const auth = getAuth(app);
const db = getFirestore(app);
const appId = typeof __app_id !== 'undefined' ? __app_id : 'lettuce-master-v8';

const LETTUCE_PRICE = 230;
const WEEKLY_SHIPPING = 300;
const MAX_BAGS_PER_WEEK = 12;

// 原始已知資料 (匯入用)
const INITIAL_IMPORT_DATA = {
  "金成": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "蕙敏": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "珮萱": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "淑芬": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "阿姿": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "嘉鎂": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "斯涵": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "明智": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "翠花": ["2/24", "3/3", "3/10", "3/24"],
  "政君": ["3/3", "3/17", "3/31"],
  "睿婷": ["3/3", "3/17", "3/31"],
  "亭愷": ["3/3", "3/17", "3/31"],
  "俊明": ["2/24", "3/10", "3/24"],
  "惠偵": ["2/24", "3/10", "3/24"]
};

const INITIAL_DATES = ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"];

export default function App() {
  const [user, setUser] = useState(null);
  const [orders, setOrders] = useState({});
  const [dates, setDates] = useState([]);
  const [isReady, setIsReady] = useState(false);
  const [isAdmin, setIsAdmin] = useState(false);
  const [currentMonth, setCurrentMonth] = useState("");
  const [isProcessing, setIsProcessing] = useState(false);
  const [newMemberName, setNewMemberName] = useState("");
  const [newDateInput, setNewDateInput] = useState("");
  const [statusMsg, setStatusMsg] = useState(null);
  const [showAdminPanel, setShowAdminPanel] = useState(false);

  // 1. 初始化連線
  useEffect(() => {
    const initAuth = async () => {
      try {
        await signInAnonymously(auth);
      } catch (e) { 
        console.error("Auth Error", e);
        setIsReady(true); 
      }
    };
    initAuth();
    onAuthStateChanged(auth, u => { 
      setUser(u); 
      if(u) setIsReady(true); 
    });
  }, []);

  // 2. 同步資料 (Firestore 監聽)
  useEffect(() => {
    if (!user) return;
    
    // 設定路徑
    const configRef = doc(db, 'artifacts', appId, 'public', 'data', 'config', 'settings');
    const orderCol = collection(db, 'artifacts', appId, 'public', 'data', 'lettuce_orders');

    const unsubDates = onSnapshot(configRef, (docSnap) => {
      if (docSnap.exists()) {
        const d = docSnap.data().activeDates || [];
        setDates(d);
        if (d.length > 0 && !currentMonth) {
          const defaultMonth = d[d.length - 1].split('/')[0] + '月';
          setCurrentMonth(defaultMonth);
        }
      } else {
        // 如果沒有設定檔，自動建立初始日期列
        setDoc(configRef, { activeDates: INITIAL_DATES });
      }
    }, (err) => console.error("Dates Sync Error", err));

    const unsubOrders = onSnapshot(orderCol, (snapshot) => {
      const newOrders = {};
      snapshot.docs.forEach(d => { newOrders[d.id] = d.data().orders || {}; });
      setOrders(newOrders);
    }, (err) => console.error("Orders Sync Error", err));

    return () => { unsubDates(); unsubOrders(); };
  }, [user, currentMonth]);

  // 3. 計算統計邏輯
  const monthList = useMemo(() => {
    const months = new Set();
    dates.forEach(d => months.add(d.split('/')[0] + '月'));
    return Array.from(months).sort((a, b) => parseInt(a) - parseInt(b));
  }, [dates]);

  const currentDates = useMemo(() => {
    return dates.filter(d => d.split('/')[0] + '月' === currentMonth);
  }, [dates, currentMonth]);

  const stats = useMemo(() => {
    const dailyCount = {};
    const shippingPerPerson = {};
    const memberList = Object.keys(orders).sort();
    
    dates.forEach(date => {
      let count = 0;
      memberList.forEach(m => { if (orders[m]?.[date]) count++; });
      dailyCount[date] = count;
      
      if (count >= 12) shippingPerPerson[date] = 25;
      else if (count === 11) shippingPerPerson[date] = 27;
      else if (count === 10) shippingPerPerson[date] = 30;
      else if (count > 0) shippingPerPerson[date] = Math.round(WEEKLY_SHIPPING / count);
      else shippingPerPerson[date] = 0;
    });
    return { dailyCount, shippingPerPerson, memberList };
  }, [orders, dates]);

  // 4. 管理功能按鈕
  const handleImport = async () => {
    if (!user) return;
    setIsProcessing(true);
    try {
      const batch = writeBatch(db);
      
      // 1. 先寫入日期設定
      const configRef = doc(db, 'artifacts', appId, 'public', 'data', 'config', 'settings');
      batch.set(configRef, { activeDates: INITIAL_DATES });

      // 2. 寫入 14 位成員的訂單資料
      Object.entries(INITIAL_IMPORT_DATA).forEach(([name, datesArr]) => {
        const docRef = doc(db, 'artifacts', appId, 'public', 'data', 'lettuce_orders', name);
        const o = {}; 
        datesArr.forEach(d => o[d] = true);
        batch.set(docRef, { orders: o });
      });

      await batch.commit();
      showToast("原始資料與日期列匯入成功！");
      setCurrentMonth("3月"); // 強制跳轉到 3 月視圖
    } catch (e) { 
      console.error("Import Error", e);
      showToast("匯入失敗，請確認網路連線", "error"); 
    } finally { 
      setIsProcessing(false); 
    }
  };

  const handleSettle = async (name) => {
    if (!window.confirm(`確認「${name}」已匯款？結帳後所有勾選將清除歸零。`)) return;
    setIsProcessing(true);
    try {
      const docRef = doc(db, 'artifacts', appId, 'public', 'data', 'lettuce_orders', name);
      await updateDoc(docRef, { orders: {} });
      showToast(`${name} 結帳成功！`);
    } catch (e) { showToast("結帳失敗", "error"); }
    finally { setIsProcessing(false); }
  };

  const addDate = async () => {
    if (!newDateInput.includes('/')) { showToast("格式需為 4/1", "error"); return; }
    const updatedDates = [...dates, newDateInput].sort((a,b) => {
      const [am, ad] = a.split('/').map(Number);
      const [bm, bd] = b.split('/').map(Number);
      return am !== bm ? am - bm : ad - bd;
    });
    const configRef = doc(db, 'artifacts', appId, 'public', 'data', 'config', 'settings');
    await updateDoc(configRef, { activeDates: updatedDates });
    setNewDateInput("");
    showToast(`已新增日期：${newDateInput}`);
  };

  const toggleOrder = async (name, date) => {
    if (isAdmin && showAdminPanel) return; 
    const isOrdered = orders[name]?.[date] || false;
    if (!isOrdered && stats.dailyCount[date] >= MAX_BAGS_PER_WEEK) {
      showToast(`${date} 已達上限 12 包！`, 'error');
      return;
    }
    const docRef = doc(db, 'artifacts', appId, 'public', 'data', 'lettuce_orders', name);
    await setDoc(docRef, { orders: { [date]: !isOrdered } }, { merge: true });
  };

  const showToast = (text, type = 'success') => {
    setStatusMsg({ text, type });
    setTimeout(() => setStatusMsg(null), 3000);
  };

  if (!isReady) return (
    <div className="h-screen flex flex-col items-center justify-center bg-green-50">
      <Loader2 className="animate-spin text-green-600 w-10 h-10 mb-4"/>
      <p className="text-green-800 font-bold">系統連線中...</p>
    </div>
  );

  return (
    <div className="min-h-screen bg-slate-50 pb-24 font-sans text-slate-900 overflow-x-hidden">
      {/* Navbar */}
      <nav className="bg-green-700 text-white shadow-xl sticky top-0 z-[60] px-4 py-3">
        <div className="max-w-7xl mx-auto flex items-center justify-between">
          <div className="flex items-center gap-2">
            <ShoppingCart className="w-5 h-5" />
            <h1 className="text-sm sm:text-lg font-black tracking-tight">生菜訂購助手 v8</h1>
          </div>
          
          <div className="flex items-center gap-2">
            <button 
              onClick={() => setIsAdmin(!isAdmin)}
              className={`flex items-center gap-1 px-4 py-1.5 rounded-full text-xs font-bold transition-all ${
                isAdmin ? 'bg-red-500 text-white shadow-lg' : 'bg-white/20 text-green-50'
              }`}
            >
              {isAdmin ? <Unlock size={14} /> : <Lock size={14} />}
              {isAdmin ? '管理模式' : '檢視模式'}
            </button>
            {isAdmin && (
              <button onClick={() => setShowAdminPanel(!showAdminPanel)} className="p-2 bg-white/10 rounded-xl hover:bg-white/20">
                <Settings size={20} />
              </button>
            )}
          </div>
        </div>
      </nav>

      <main className="max-w-7xl mx-auto px-2 sm:px-4 py-6 space-y-6">
        
        {/* Admin Dashboard */}
        {isAdmin && showAdminPanel && (
          <div className="bg-white p-6 rounded-3xl shadow-2xl border-2 border-red-50 animate-in zoom-in-95">
            <div className="flex items-center justify-between mb-6 border-b pb-4">
              <h3 className="text-red-600 font-black flex items-center gap-2 text-lg">
                <Settings size={22} /> 管理員控制台
              </h3>
              <button onClick={() => setShowAdminPanel(false)} className="text-slate-400"><X size={24}/></button>
            </div>
            
            <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
              <div className="bg-blue-50 p-5 rounded-2xl space-y-3">
                <p className="text-xs font-bold text-blue-600 uppercase tracking-widest">資料恢復區</p>
                <button 
                  onClick={handleImport}
                  disabled={isProcessing}
                  className="w-full bg-blue-600 hover:bg-blue-700 text-white py-3 rounded-xl font-bold flex items-center justify-center gap-2 shadow-lg disabled:opacity-50"
                >
                  {isProcessing ? <Loader2 className="animate-spin" size={18}/> : <Database size={18} />}
                  立即匯入 14 位成員訂單
                </button>
                <p className="text-[10px] text-blue-400 leading-relaxed italic">
                  ※ 此操作將同時重建 2/24 至 3/31 的日期列與訂單紀錄。
                </p>
              </div>

              <div className="space-y-4">
                <div className="space-y-1">
                  <label className="text-[10px] font-bold text-slate-400">新增團購成員</label>
                  <div className="flex gap-2">
                    <input className="flex-1 border-2 border-slate-100 p-2 rounded-xl outline-none focus:border-green-400" placeholder="成員姓名..." value={newMemberName} onChange={e => setNewMemberName(e.target.value)}/>
                    <button onClick={async () => {
                      if(!newMemberName) return;
                      await setDoc(doc(db, 'artifacts', appId, 'public', 'data', 'lettuce_orders', newMemberName), { orders: {} });
                      setNewMemberName(""); showToast(`已新增：${newMemberName}`);
                    }} className="bg-green-600 text-white px-5 py-2 rounded-xl font-bold">新增</button>
                  </div>
                </div>
                <div className="space-y-1">
                  <label className="text-[10px] font-bold text-slate-400">新增配送日期 (例: 4/7)</label>
                  <div className="flex gap-2">
                    <input className="flex-1 border-2 border-slate-100 p-2 rounded-xl outline-none focus:border-blue-400" placeholder="月/日..." value={newDateInput} onChange={e => setNewDateInput(e.target.value)}/>
                    <button onClick={addDate} className="bg-blue-600 text-white px-5 py-2 rounded-xl font-bold">新增</button>
                  </div>
                </div>
              </div>
            </div>
          </div>
        )}

        {/* Month Selector */}
        <div className="flex items-center gap-2 bg-white p-2 rounded-2xl shadow-sm border border-slate-200 overflow-x-auto scrollbar-hide">
            {monthList.length === 0 ? (
                <div className="px-4 py-2 text-slate-400 text-sm italic">尚未設定任何日期...</div>
            ) : (
                monthList.map(m => (
                    <button 
                        key={m}
                        onClick={() => setCurrentMonth(m)}
                        className={`px-6 py-2 rounded-xl font-black text-xs sm:text-sm transition-all whitespace-nowrap ${
                            currentMonth === m ? 'bg-green-600 text-white shadow-md' : 'text-slate-400 hover:bg-slate-100'
                        }`}
                    >
                        {m}份
                    </button>
                ))
            )}
        </div>

        {/* Order Matrix */}
        <div className="bg-white rounded-3xl shadow-xl overflow-hidden border border-slate-200">
          <div className="overflow-auto max-h-[65vh]">
            <table className="w-full text-left border-separate border-spacing-0">
              <thead className="sticky top-0 z-40 bg-slate-900 text-white">
                <tr>
                  <th className="p-4 sticky left-0 z-50 bg-slate-900 border-r border-slate-800 text-center text-[10px] font-black w-24 sm:w-32 shadow-md">姓名</th>
                  {currentDates.map(date => (
                    <th key={date} className="p-4 text-center min-w-[90px] border-b border-slate-800">
                      <div className="flex flex-col">
                        <span className="text-[8px] text-slate-500 font-normal">週二</span>
                        <span className="text-sm sm:text-base font-black">{date}</span>
                      </div>
                    </th>
                  ))}
                  <th className="p-4 text-right border-b border-slate-800 bg-slate-950 text-green-400 text-[10px] font-black">累計應付</th>
                </tr>
              </thead>
              <tbody className="divide-y divide-slate-100">
                {stats.memberList.length === 0 ? (
                  <tr>
                    <td colSpan={currentDates.length + 2} className="p-20 text-center text-slate-400 text-sm">
                      <Database className="mx-auto mb-2 opacity-20" size={48} />
                      <p>尚未匯入任何成員，請開啟管理模式進行匯入。</p>
                    </td>
                  </tr>
                ) : (
                  stats.memberList.map(name => {
                    let totalBags = 0; let totalShip = 0;
                    dates.forEach(d => { if (orders[name]?.[d]) { totalBags++; totalShip += stats.shippingPerPerson[d]; } });

                    return (
                      <tr key={name} className="hover:bg-green-50/20 transition-colors group">
                        <td className="p-3 font-bold text-slate-700 sticky left-0 bg-white z-20 border-r border-slate-100 flex items-center justify-between gap-1 shadow-sm">
                          <span className="truncate">{name}</span>
                          {isAdmin && (
                            <div className="flex gap-1 opacity-0 group-hover:opacity-100 transition-opacity">
                              <button onClick={() => handleSettle(name)} className="text-green-500 hover:bg-green-50 p-1 rounded" title="匯款結帳"><Check size={14}/></button>
                              <button onClick={async () => { if(window.confirm(`移除成員 ${name}？`)) await deleteDoc(doc(db, 'artifacts', appId, 'public', 'data', 'lettuce_orders', name)); }} className="text-red-400 p-1 rounded"><Trash2 size={14}/></button>
                            </div>
                          )}
                        </td>
                        {currentDates.map(date => (
                          <td key={date} className="p-2 text-center">
                            <button
                              onClick={() => toggleOrder(name, date)}
                              disabled={isAdmin && showAdminPanel}
                              className={`mx-auto w-10 h-10 sm:w-12 sm:h-12 rounded-2xl flex items-center justify-center transition-all ${
                                orders[name]?.[date] 
                                  ? 'bg-green-600 text-white shadow-lg scale-105' 
                                  : 'bg-slate-50 text-slate-200 hover:bg-slate-200'
                              }`}
                            >
                              {orders[name]?.[date] ? <CheckCircle2 size={24} strokeWidth={3} /> : <Plus size={18} />}
                            </button>
                          </td>
                        ))}
                        <td className="p-4 text-right bg-slate-50/80 sticky right-0 z-10 font-mono font-black text-green-700 text-sm">
                          ${(totalBags * LETTUCE_PRICE + totalShip).toLocaleString()}
                        </td>
                      </tr>
                    );
                  })
                )}
              </tbody>
              <tfoot className="sticky bottom-0 z-40 bg-slate-900 text-white text-[10px]">
                <tr className="font-bold">
                  <td className="p-4 sticky left-0 bg-slate-900 border-r border-slate-800">當週數量</td>
                  {currentDates.map(date => (
                    <td key={date} className="p-4 text-center">
                      <div className={`px-2 py-0.5 rounded-full inline-block ${stats.dailyCount[date] >= MAX_BAGS_PER_WEEK ? 'bg-red-500' : 'bg-green-600'}`}>
                        {stats.dailyCount[date]} / 12
                      </div>
                    </td>
                  ))}
                  <td className="p-4 text-right text-gray-400">BAGS</td>
                </tr>
                <tr className="bg-slate-800 text-blue-300">
                  <td className="p-2 sticky left-0 bg-slate-800 border-r border-slate-700 text-center">分擔運費</td>
                  {currentDates.map(date => (
                    <td key={date} className="p-2 text-center font-mono">
                      <div className="flex items-center justify-center gap-1">
                          <Truck size={10} /> ${stats.shippingPerPerson[date]}
                      </div>
                    </td>
                  ))}
                  <td className="p-2"></td>
                </tr>
              </tfoot>
            </table>
          </div>
        </div>

        {/* Member Cards */}
        <div className="space-y-4">
          <div className="flex items-center gap-2">
            <Users size={20} className="text-green-600"/>
            <h2 className="font-black text-lg text-slate-700">繳費結帳清單</h2>
          </div>
          <div className="grid grid-cols-2 sm:grid-cols-2 lg:grid-cols-4 gap-3">
            {stats.memberList.map(name => {
              let bags = 0; let ship = 0;
              dates.forEach(d => { if (orders[name]?.[d]) { bags++; ship += stats.shippingPerPerson[d]; } });
              if (bags === 0 && !isAdmin) return null;

              return (
                <div key={name} className="bg-white p-4 rounded-3xl border border-slate-200 shadow-sm relative overflow-hidden flex flex-col group">
                  <div className="absolute top-0 left-0 w-1.5 h-full bg-green-500"></div>
                  <div className="flex justify-between items-start mb-3">
                    <span className="font-black text-slate-800 text-base">{name}</span>
                    <span className="text-[10px] bg-slate-100 px-2 py-0.5 rounded text-slate-500 font-bold">{bags} 包</span>
                  </div>
                  <div className="text-[11px] text-slate-500 space-y-1 border-b border-dashed pb-3">
                    <div className="flex justify-between"><span>生菜費用:</span><span>${bags * LETTUCE_PRICE}</span></div>
                    <div className="flex justify-between"><span>累計運費:</span><span>${ship}</span></div>
                    <div className="flex justify-between pt-2 font-black text-green-600 text-lg"><span>總額:</span><span>${(bags * LETTUCE_PRICE + ship).toLocaleString()}</span></div>
                  </div>
                  
                  {isAdmin && (
                    <button 
                      onClick={() => handleSettle(name)}
                      disabled={bags === 0}
                      className={`mt-3 w-full py-2 rounded-xl text-[10px] font-black flex items-center justify-center gap-1 transition-all shadow-sm ${
                        bags === 0 ? 'bg-slate-50 text-slate-300 cursor-not-allowed' : 'bg-green-600 text-white hover:bg-green-700 active:scale-95'
                      }`}
                    >
                      <CreditCard size={12}/> 確認已匯款結帳
                    </button>
                  )}
                </div>
              );
            })}
          </div>
        </div>
      </main>

      {/* Toast Notification */}
      {statusMsg && (
        <div className={`fixed bottom-8 left-1/2 -translate-x-1/2 z-[100] flex items-center gap-2 px-8 py-3 rounded-full shadow-2xl animate-in slide-in-from-bottom-6 duration-300 ${
          statusMsg.type === 'error' ? 'bg-red-600 text-white' : 'bg-green-800 text-white'
        }`}>
          {statusMsg.type === 'error' ? <AlertCircle size={18}/> : <CheckCircle2 size={18} />}
          <span className="font-bold text-sm">{statusMsg.text}</span>
        </div>
      )}
    </div>
  );
}
```
