# 📚 SCA 考官理論：對話語音生成器

## 📋 對話理論筆記
```react
import React, { useState, useRef, useEffect } from 'react';
import { 
  Play, Plus, Trash2, Volume2, Settings, Loader2, MessageSquare, Mic, FileText, Globe
} from 'lucide-react';

// API Configuration
const apiKey = ""; // The execution environment provides the key at runtime

// Voice mappings categorized by typical gender/tone for user convenience
const VOICE_CATEGORIES = {
  "推薦女聲": ["Kore", "Aoede", "Leda", "Callirrhoe", "Zephyr"],
  "推薦男聲": ["Puck", "Charon", "Fenrir", "Orus", "Enceladus"],
  "其他音色": [
    "Autonoe", "Iapetus", "Umbriel", "Algieba", "Despina", "Erinome", 
    "Algenib", "Rasalgethi", "Laomedeia", "Achernar", "Alnilam", 
    "Schedar", "Gacrux", "Pulcherrima", "Achird", "Zubenelgenubi", 
    "Vindemiatrix", "Sadachbia", "Sadaltager", "Sulafat"
  ]
};

const ALL_VOICES = Object.values(VOICE_CATEGORIES).flat();
const SUPPORTED_LANGUAGES = [
  { value: '中文', label: '中文' },
  { value: '英文', label: 'English' },
  { value: '日文', label: '日本語' },
  { value: '德文', label: 'Deutsch' },
  { value: '西班牙語', label: 'Español' }
];

export default function App() {
  // Mode Selection: 'dialogue' | 'text'
  const [activeTab, setActiveTab] = useState('dialogue');

  // ================= STATE FOR DIALOGUE MODE =================
  const [dialogueLanguage, setDialogueLanguage] = useState('中文');
  const [speakers, setSpeakers] = useState([
    { id: 'speaker_1', name: '小美', voice: 'Kore' },
    { id: 'speaker_2', name: '小明', voice: 'Puck' }
  ]);

  const [dialogue, setDialogue] = useState([
    { id: 1, speakerId: 'speaker_1', text: '早安！今天天氣真好，要一起去喝杯咖啡嗎？' },
    { id: 2, speakerId: 'speaker_2', text: '好啊！我知道附近有一家新開的店，聽說他們的拿鐵很不錯。' }
  ]);

  // ================= STATE FOR TEXT MODE =================
  const [textConfig, setTextConfig] = useState({
    text: '',
    voice: 'Kore',
    language: '中文'
  });

  // ================= SHARED STATE =================
  const [isGenerating, setIsGenerating] = useState(false);
  const [audioUrl, setAudioUrl] = useState(null);
  const [error, setError] = useState(null);
  const audioRef = useRef(null);
  const bottomRef = useRef(null);

  // Auto-scroll to bottom of dialogue when adding a new line
  useEffect(() => {
    if (activeTab === 'dialogue' && bottomRef.current) {
      bottomRef.current.scrollIntoView({ behavior: 'smooth' });
    }
  }, [dialogue.length, activeTab]);

  // Handle Speaker operations (Dialogue Mode)
  const addSpeaker = () => {
    const newId = `speaker_${Date.now()}`;
    const newName = `角色 ${speakers.length + 1}`;
    const defaultVoice = ALL_VOICES[speakers.length % ALL_VOICES.length];
    
    setSpeakers([...speakers, { id: newId, name: newName, voice: defaultVoice }]);
  };

  const updateSpeaker = (id, field, value) => {
    setSpeakers(speakers.map(s => s.id === id ? { ...s, [field]: value } : s));
  };

  const removeSpeaker = (id) => {
    if (speakers.length <= 1) return; // Keep at least one speaker
    setSpeakers(speakers.filter(s => s.id !== id));
    const firstAvailableSpeaker = speakers.find(s => s.id !== id);
    setDialogue(dialogue.map(d => d.speakerId === id ? { ...d, speakerId: firstAvailableSpeaker.id } : d));
  };

  // Handle Dialogue operations
  const addDialogueLine = () => {
    const lastSpeakerId = dialogue.length > 0 ? dialogue[dialogue.length - 1].speakerId : speakers[0].id;
    const otherSpeaker = speakers.find(s => s.id !== lastSpeakerId) || speakers[0];
    
    setDialogue([...dialogue, { 
      id: Date.now(), 
      speakerId: otherSpeaker.id, 
      text: '' 
    }]);
  };

  const updateDialogueLine = (id, field, value) => {
    setDialogue(dialogue.map(d => d.id === id ? { ...d, [field]: value } : d));
  };

  const removeDialogueLine = (id) => {
    setDialogue(dialogue.filter(d => d.id !== id));
  };

  // Utility to convert Base64 PCM data to a playable WAV Blob
  const pcmToWav = (base64Data, sampleRate = 24000, numChannels = 1) => {
    const binaryString = atob(base64Data);
    const len = binaryString.length;
    const bytes = new Uint8Array(len);
    for (let i = 0; i < len; i++) {
      bytes[i] = binaryString.charCodeAt(i);
    }

    const buffer = new ArrayBuffer(44 + bytes.length);
    const view = new DataView(buffer);

    writeString(view, 0, 'RIFF');
    view.setUint32(4, 36 + bytes.length, true);
    writeString(view, 8, 'WAVE');
    writeString(view, 12, 'fmt ');
    view.setUint32(16, 16, true);
    view.setUint16(20, 1, true);
    view.setUint16(22, numChannels, true);
    view.setUint32(24, sampleRate, true);
    view.setUint32(28, sampleRate * numChannels * 2, true);
    view.setUint16(32, numChannels * 2, true);
    view.setUint16(34, 16, true);
    writeString(view, 36, 'data');
    view.setUint32(40, bytes.length, true);

    const pcmData = new Uint8Array(buffer, 44);
    pcmData.set(bytes);

    return new Blob([buffer], { type: 'audio/wav' });
  };

  const writeString = (view, offset, string) => {
    for (let i = 0; i < string.length; i++) {
      view.setUint8(offset + i, string.charCodeAt(i));
    }
  };

  // Fetch with exponential backoff for API robustness
  const fetchWithRetry = async (url, options, maxRetries = 5) => {
    const delays = [1000, 2000, 4000, 8000, 16000];
    for (let i = 0; i < maxRetries; i++) {
      try {
        const response = await fetch(url, options);
        if (!response.ok) {
          const errData = await response.json();
          throw new Error(errData.error?.message || `HTTP 錯誤 ${response.status}`);
        }
        return await response.json();
      } catch (err) {
        if (i === maxRetries - 1) throw err;
        await new Promise(res => setTimeout(res, delays[i]));
      }
    }
  };

  // Generate Audio
  const handleGenerate = async () => {
    setIsGenerating(true);
    setError(null);
    setAudioUrl(null);

    try {
      let payload;

      if (activeTab === 'dialogue') {
        // Validation for Dialogue Mode
        const validLines = dialogue.filter(d => d.text.trim() !== '');
        if (validLines.length === 0) {
          throw new Error("請至少輸入一句對話內容！");
        }

        const textPrompt = validLines.map(line => {
          const speaker = speakers.find(s => s.id === line.speakerId);
          return `${speaker.name}: ${line.text}`;
        }).join('\n');

        const speakerVoiceConfigs = speakers.map(speaker => ({
          speaker: speaker.name,
          voiceConfig: { prebuiltVoiceConfig: { voiceName: speaker.voice } }
        }));

        const instructionPrompt = `請用${dialogueLanguage}朗讀以下對話：\n\n${textPrompt}`;

        payload = {
          contents: [{ parts: [{ text: instructionPrompt }] }],
          generationConfig: {
            responseModalities: ["AUDIO"],
            speechConfig: {
              multiSpeakerVoiceConfig: {
                speakerVoiceConfigs: speakerVoiceConfigs
              }
            }
          },
          model: "gemini-2.5-flash-preview-tts"
        };
      } else {
        // Validation for Text Mode
        if (!textConfig.text.trim()) {
          throw new Error("請貼上或輸入您要轉換的文本內容！");
        }

        // Add instruction to prompt for specific language guidance
        let instructionPrompt = `請用${textConfig.language}朗讀以下內容：\n\n${textConfig.text}`;

        payload = {
          contents: [{ parts: [{ text: instructionPrompt }] }],
          generationConfig: {
            responseModalities: ["AUDIO"],
            speechConfig: {
              voiceConfig: {
                prebuiltVoiceConfig: { voiceName: textConfig.voice }
              }
            }
          },
          model: "gemini-2.5-flash-preview-tts"
        };
      }

      const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-tts:generateContent?key=${apiKey}`;

      const result = await fetchWithRetry(url, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(payload)
      });

      // 尋找回傳資料中包含 inlineData (音檔) 的部分，因為有時 API 會同時回傳文字和音檔
      const parts = result.candidates?.[0]?.content?.parts;
      const audioPart = parts?.find(p => p.inlineData && p.inlineData.data);

      if (!audioPart) {
        console.error("API 回傳結果異常或缺少語音：", JSON.stringify(result, null, 2));
        
        let errorMsg = "無法從伺服器取得語音資料。";
        if (result.promptFeedback?.blockReason) {
          errorMsg = `內容因安全政策被阻擋 (${result.promptFeedback.blockReason})。`;
        } else if (result.candidates?.[0]?.finishReason && result.candidates[0].finishReason !== 'STOP') {
          errorMsg = `生成中斷，原因：${result.candidates[0].finishReason}。`;
        } else if (parts && parts.length > 0 && parts[0].text) {
          errorMsg = `伺服器僅回傳了文字而沒有語音，請嘗試調整文本內容。`;
        }
        throw new Error(errorMsg);
      }

      const inlineData = audioPart.inlineData;
      const base64Data = inlineData.data;
      const mimeType = inlineData.mimeType || "audio/pcm; rate=24000";
      
      const rateMatch = mimeType.match(/rate=(\d+)/);
      const sampleRate = rateMatch ? parseInt(rateMatch[1], 10) : 24000;

      const wavBlob = pcmToWav(base64Data, sampleRate);
      const newAudioUrl = URL.createObjectURL(wavBlob);
      
      setAudioUrl(newAudioUrl);

      setTimeout(() => {
        if (audioRef.current) {
          audioRef.current.play().catch(e => console.log("Auto-play prevented", e));
        }
      }, 100);

    } catch (err) {
      console.error(err);
      setError(err.message || "生成語音時發生錯誤，請稍後再試。");
    } finally {
      setIsGenerating(false);
    }
  };

  return (
    <div className="min-h-screen bg-slate-50 text-slate-800 p-4 md:p-8 font-sans">
      <div className="max-w-6xl mx-auto space-y-6">
        
        {/* Header & Tabs */}
        <header className="mb-6">
          <div className="flex items-center gap-3 mb-6">
            <div className="bg-indigo-600 p-3 rounded-xl text-white shadow-lg shadow-indigo-200">
              <Mic size={28} />
            </div>
            <div>
              <h1 className="text-3xl font-bold text-slate-900 tracking-tight">AI 語音生成器</h1>
              <p className="text-slate-500 mt-1">支援多角色對話劇本，或單一長文本的語音合成。</p>
            </div>
          </div>

          <div className="flex bg-slate-200/60 p-1.5 rounded-xl w-fit shadow-inner">
            <button 
              onClick={() => { setActiveTab('dialogue'); setAudioUrl(null); }}
              className={`flex items-center gap-2 px-6 py-2.5 rounded-lg font-medium transition-all duration-200 ${
                activeTab === 'dialogue' 
                  ? 'bg-white text-indigo-700 shadow-sm' 
                  : 'text-slate-500 hover:text-slate-800 hover:bg-slate-200'
              }`}
            >
              <MessageSquare size={18} />
              對話模式
            </button>
            <button 
              onClick={() => { setActiveTab('text'); setAudioUrl(null); }}
              className={`flex items-center gap-2 px-6 py-2.5 rounded-lg font-medium transition-all duration-200 ${
                activeTab === 'text' 
                  ? 'bg-white text-indigo-700 shadow-sm' 
                  : 'text-slate-500 hover:text-slate-800 hover:bg-slate-200'
              }`}
            >
              <FileText size={18} />
              文本模式
            </button>
          </div>
        </header>

        <div className="grid grid-cols-1 lg:grid-cols-12 gap-8">
          
          {/* ================= LEFT PANEL (Settings) ================= */}
          <div className="lg:col-span-4 space-y-4">
            
            {/* Setting for Dialogue Mode */}
            {activeTab === 'dialogue' && (
              <div className="bg-white rounded-2xl shadow-sm border border-slate-200 overflow-hidden">
                <div className="bg-slate-100/50 p-4 border-b border-slate-200 flex justify-between items-center">
                  <h2 className="font-semibold text-slate-800 flex items-center gap-2">
                    <Settings size={18} className="text-indigo-500"/>
                    對話設定
                  </h2>
                  <button 
                    onClick={addSpeaker}
                    className="text-xs font-medium bg-white px-3 py-1.5 rounded-lg border border-slate-200 hover:bg-slate-50 transition-colors flex items-center gap-1 text-indigo-600"
                  >
                    <Plus size={14} /> 新增角色
                  </button>
                </div>
                
                <div className="p-4 space-y-4">
                  {/* Dialogue Language Selection */}
                  <div>
                    <label className="text-sm font-medium text-slate-700 mb-2 flex items-center gap-2">
                      <Globe size={16} className="text-slate-400"/> 發音語言
                    </label>
                    <div className="grid grid-cols-2 gap-2 mt-2">
                      {SUPPORTED_LANGUAGES.map(lang => (
                        <button
                          key={`dialogue-lang-${lang.value}`}
                          onClick={() => setDialogueLanguage(lang.value)}
                          className={`py-2 px-3 rounded-lg text-sm font-medium border transition-all ${
                            dialogueLanguage === lang.value 
                              ? 'bg-indigo-50 border-indigo-200 text-indigo-700 ring-1 ring-indigo-500' 
                              : 'bg-white border-slate-200 text-slate-600 hover:bg-slate-50'
                          }`}
                        >
                          {lang.label}
                        </button>
                      ))}
                    </div>
                    <p className="text-xs text-amber-600 mt-3 bg-amber-50 p-2.5 rounded-lg border border-amber-100 leading-relaxed">
                      💡 <span className="font-semibold">提示：</span>系統不會自動翻譯。請輸入與選項相符的語言文字（例如：選擇 Deutsch 無法將中文翻成德文語音，需直接輸入德文）。
                    </p>
                  </div>

                  <hr className="border-slate-100" />

                  <label className="text-sm font-medium text-slate-700 block mt-2">角色音色設定</label>
                  {speakers.map((speaker, index) => (
                    <div key={speaker.id} className="p-3 bg-slate-50 rounded-xl border border-slate-100 relative group transition-all">
                      <div className="flex items-center gap-3 mb-3">
                        <div className="w-8 h-8 rounded-full bg-indigo-100 text-indigo-700 flex items-center justify-center font-bold text-sm shrink-0">
                          {speaker.name.charAt(0)}
                        </div>
                        <input 
                          type="text" 
                          value={speaker.name}
                          onChange={(e) => updateSpeaker(speaker.id, 'name', e.target.value)}
                          className="w-full bg-transparent border-b border-transparent hover:border-slate-300 focus:border-indigo-500 focus:outline-none transition-colors font-medium text-slate-700"
                          placeholder="角色名稱"
                        />
                        {speakers.length > 1 && (
                          <button 
                            onClick={() => removeSpeaker(speaker.id)}
                            className="text-slate-400 hover:text-red-500 opacity-0 group-hover:opacity-100 transition-opacity p-1"
                            title="刪除角色"
                          >
                            <Trash2 size={16} />
                          </button>
                        )}
                      </div>
                      <div>
                        <label className="text-xs text-slate-500 mb-1 block">選擇音色</label>
                        <select 
                          value={speaker.voice}
                          onChange={(e) => updateSpeaker(speaker.id, 'voice', e.target.value)}
                          className="w-full text-sm p-2 rounded-lg border border-slate-200 bg-white focus:ring-2 focus:ring-indigo-500/20 focus:border-indigo-500 outline-none appearance-none"
                        >
                          {Object.entries(VOICE_CATEGORIES).map(([category, voices]) => (
                            <optgroup label={category} key={category}>
                              {voices.map(v => <option value={v} key={v}>{v}</option>)}
                            </optgroup>
                          ))}
                        </select>
                      </div>
                    </div>
                  ))}
                </div>
              </div>
            )}

            {/* Setting for Text Mode */}
            {activeTab === 'text' && (
              <div className="bg-white rounded-2xl shadow-sm border border-slate-200 overflow-hidden">
                <div className="bg-slate-100/50 p-4 border-b border-slate-200">
                  <h2 className="font-semibold text-slate-800 flex items-center gap-2">
                    <Settings size={18} className="text-indigo-500"/>
                    文本設定
                  </h2>
                </div>
                
                <div className="p-5 space-y-6">
                  {/* Language Selection */}
                  <div>
                    <label className="text-sm font-medium text-slate-700 mb-2 flex items-center gap-2">
                      <Globe size={16} className="text-slate-400"/> 發音語言
                    </label>
                    <div className="grid grid-cols-2 gap-2 mt-2">
                      {SUPPORTED_LANGUAGES.map(lang => (
                        <button
                          key={`text-lang-${lang.value}`}
                          onClick={() => setTextConfig({...textConfig, language: lang.value})}
                          className={`py-2 px-3 rounded-lg text-sm font-medium border transition-all ${
                            textConfig.language === lang.value 
                              ? 'bg-indigo-50 border-indigo-200 text-indigo-700 ring-1 ring-indigo-500' 
                              : 'bg-white border-slate-200 text-slate-600 hover:bg-slate-50'
                          }`}
                        >
                          {lang.label}
                        </button>
                      ))}
                    </div>
                    <p className="text-xs text-amber-600 mt-3 bg-amber-50 p-2.5 rounded-lg border border-amber-100 leading-relaxed">
                      💡 <span className="font-semibold">提示：</span>系統不會自動翻譯。請輸入與選項相符的語言文字（例如：選擇 Deutsch 無法將中文翻成德文語音，需直接輸入德文）。
                    </p>
                  </div>

                  <hr className="border-slate-100" />

                  {/* Voice Selection */}
                  <div>
                    <label className="text-sm font-medium text-slate-700 mb-2 block">選擇音色 (男女聲可選)</label>
                    <select 
                      value={textConfig.voice}
                      onChange={(e) => setTextConfig({...textConfig, voice: e.target.value})}
                      className="w-full text-sm p-3 rounded-xl border border-slate-200 bg-slate-50 focus:bg-white focus:ring-2 focus:ring-indigo-500/20 focus:border-indigo-500 outline-none cursor-pointer"
                    >
                      {Object.entries(VOICE_CATEGORIES).map(([category, voices]) => (
                        <optgroup label={category} key={category}>
                          {voices.map(v => <option value={v} key={v}>{v}</option>)}
                        </optgroup>
                      ))}
                    </select>
                  </div>
                </div>
              </div>
            )}
          </div>

          {/* ================= RIGHT PANEL (Editor & Generator) ================= */}
          <div className="lg:col-span-8 flex flex-col h-[calc(100vh-16rem)] min-h-[500px]">
            <div className="bg-white rounded-2xl shadow-sm border border-slate-200 flex flex-col flex-grow overflow-hidden">
              
              {/* Header of Right Panel */}
              <div className="bg-slate-100/50 p-4 border-b border-slate-200 flex justify-between items-center shrink-0">
                <h2 className="font-semibold text-slate-800 flex items-center gap-2">
                  {activeTab === 'dialogue' ? (
                    <><MessageSquare size={18} className="text-indigo-500"/> 劇本與對話內容</>
                  ) : (
                    <><FileText size={18} className="text-indigo-500"/> 詳細文本內容</>
                  )}
                </h2>
                {activeTab === 'dialogue' && (
                  <button 
                    onClick={addDialogueLine}
                    className="text-sm font-medium bg-indigo-50 text-indigo-700 px-4 py-2 rounded-lg hover:bg-indigo-100 transition-colors flex items-center gap-2"
                  >
                    <Plus size={16} /> 新增對話
                  </button>
                )}
              </div>

              {/* Editor Area */}
              <div className="flex-grow overflow-y-auto p-4 bg-slate-50/50 relative">
                
                {/* Dialogue Editor */}
                {activeTab === 'dialogue' && (
                  <div className="space-y-4">
                    {dialogue.map((line, index) => (
                      <div key={line.id} className="flex gap-3 group">
                        <div className="w-32 shrink-0">
                          <select 
                            value={line.speakerId}
                            onChange={(e) => updateDialogueLine(line.id, 'speakerId', e.target.value)}
                            className="w-full p-3 rounded-xl border font-medium text-sm outline-none transition-colors cursor-pointer appearance-none bg-white"
                            style={{
                              backgroundColor: `hsl(${(speakers.findIndex(s => s.id === line.speakerId) * 137.5) % 360}, 70%, 95%)`,
                              borderColor: `hsl(${(speakers.findIndex(s => s.id === line.speakerId) * 137.5) % 360}, 60%, 85%)`,
                              color: `hsl(${(speakers.findIndex(s => s.id === line.speakerId) * 137.5) % 360}, 70%, 30%)`
                            }}
                          >
                            {speakers.map(s => <option value={s.id} key={s.id}>{s.name}</option>)}
                          </select>
                        </div>
                        <div className="flex-grow relative">
                          <textarea
                            value={line.text}
                            onChange={(e) => updateDialogueLine(line.id, 'text', e.target.value)}
                            placeholder={`輸入 ${speakers.find(s => s.id === line.speakerId)?.name || '角色'} 的台詞...`}
                            className="w-full p-3 rounded-xl border border-slate-200 bg-white focus:ring-2 focus:ring-indigo-500/20 focus:border-indigo-500 outline-none resize-none min-h-[52px] leading-relaxed shadow-sm transition-shadow"
                            rows={1}
                            onInput={(e) => { e.target.style.height = 'auto'; e.target.style.height = e.target.scrollHeight + 'px'; }}
                          />
                        </div>
                        <div className="shrink-0 pt-2 opacity-0 group-hover:opacity-100 transition-opacity">
                          <button 
                            onClick={() => removeDialogueLine(line.id)}
                            className="p-2 text-slate-400 hover:text-red-500 hover:bg-red-50 rounded-lg transition-colors"
                            title="刪除這行對話"
                          >
                            <Trash2 size={18} />
                          </button>
                        </div>
                      </div>
                    ))}
                    <div ref={bottomRef} />
                  </div>
                )}

                {/* Text Editor */}
                {activeTab === 'text' && (
                  <textarea
                    value={textConfig.text}
                    onChange={(e) => setTextConfig({...textConfig, text: e.target.value})}
                    placeholder="請在此貼上或輸入您要轉換的詳細文本..."
                    className="w-full h-full min-h-[300px] p-4 rounded-xl border border-slate-200 bg-white focus:ring-2 focus:ring-indigo-500/20 focus:border-indigo-500 outline-none resize-none leading-relaxed shadow-sm transition-shadow text-base text-slate-700"
                  />
                )}
              </div>

              {/* Footer / Generate Bar */}
              <div className="p-4 bg-white border-t border-slate-200 shrink-0">
                {error && (
                  <div className="mb-4 p-3 bg-red-50 text-red-600 rounded-lg text-sm flex items-center gap-2 border border-red-100">
                    <span className="font-bold">發生錯誤：</span> {error}
                  </div>
                )}
                
                <div className="flex flex-col md:flex-row gap-4 items-center justify-between">
                  {audioUrl ? (
                    <div className="flex-grow w-full bg-slate-50 p-2 rounded-xl border border-slate-200 flex items-center gap-3">
                      <div className="bg-indigo-100 p-2 rounded-lg text-indigo-600 shrink-0">
                        <Volume2 size={20} />
                      </div>
                      <audio ref={audioRef} controls src={audioUrl} className="w-full h-10 outline-none" />
                    </div>
                  ) : (
                    <div className="flex-grow text-sm text-slate-500 flex items-center gap-2">
                      準備就緒，點擊按鈕即可生成{activeTab === 'dialogue' ? '對話' : '文本'}語音。
                    </div>
                  )}

                  <button 
                    onClick={handleGenerate}
                    disabled={isGenerating}
                    className="w-full md:w-auto px-8 py-3 bg-indigo-600 text-white rounded-xl font-medium shadow-lg shadow-indigo-200 hover:bg-indigo-700 focus:ring-4 focus:ring-indigo-500/30 transition-all disabled:opacity-70 disabled:cursor-not-allowed flex items-center justify-center gap-2 shrink-0"
                  >
                    {isGenerating ? (
                      <><Loader2 size={20} className="animate-spin" /> 正在生成語音...</>
                    ) : (
                      <><Play size={20} className="fill-current" /> 一鍵生成語音</>
                    )}
                  </button>
                </div>
              </div>

            </div>
          </div>
        </div>

        {/* Footer */}
        <footer className="text-center text-sm text-slate-500 mt-12 pb-4 pt-6 border-t border-slate-200/60">
          Open Sourced by <a href="https://www.facebook.com/Journal.of.Digital.Narrative" target="_blank" rel="noopener noreferrer" className="text-indigo-600 hover:text-indigo-800 hover:underline font-medium transition-colors">數位敘事力期刊</a>，歡迎複製Canvas並改版
        </footer>

      </div>
    </div>
  );
}
```
