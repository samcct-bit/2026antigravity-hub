## 📋 對話理論筆記
```react
import React, { useState, useEffect, useMemo } from 'react';
import { initializeApp } from 'firebase/app';
import { getAuth, signInAnonymously, onAuthStateChanged, signInWithCustomToken } from 'firebase/auth';
import { getFirestore, collection, doc, onSnapshot, setDoc, deleteDoc } from 'firebase/firestore';
import { 
  Users, ShoppingCart, Truck, Calendar, CheckCircle2, AlertCircle, Plus, Trash2, 
  Loader2, Database, Lock, Unlock, Settings, X, CreditCard, Wifi, Check 
} from 'lucide-react';

// --- Firebase 初始化 ---
// 這裡嚴格遵循系統提供的環境變數
const firebaseConfig = JSON.parse(__firebase_config);
const app = initializeApp(firebaseConfig);
const auth = getAuth(app);
const db = getFirestore(app);
const appId = typeof __app_id !== 'undefined' ? __app_id : 'lettuce-v21-final';

const LETTUCE_PRICE = 230;
const WEEKLY_SHIPPING = 300;
const MAX_BAGS_PER_WEEK = 12;

// 原始匯入資料備份
const INITIAL_IMPORT_DATA = {
  "金成": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "蕙敏": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "珮萱": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "淑芬": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "阿姿": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "嘉鎂": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "斯涵": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "明智": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "翠花": ["2/24", "3/3", "3/10", "3/24"],
  "政君": ["3/3", "3/17", "3/31"],
  "睿婷": ["3/3", "3/17", "3/31"],
  "亭愷": ["3/3", "3/17", "3/31"],
  "俊明": ["2/24", "3/10", "3/24"],
  "惠偵": ["2/24", "3/10", "3/24"]
};
const INITIAL_DATES = ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"];

export default function App() {
  const [user, setUser] = useState(null);
  const [orders, setOrders] = useState({});
  const [dates, setDates] = useState([]);
  const [isReady, setIsReady] = useState(false);
  const [isAdmin, setIsAdmin] = useState(false);
  const [currentMonth, setCurrentMonth] = useState("");
  const [isProcessing, setIsProcessing] = useState(false);
  const [newMemberName, setNewMemberName] = useState("");
  const [newDateInput, setNewDateInput] = useState("");
  const [statusMsg, setStatusMsg] = useState(null);
  const [showAdminPanel, setShowAdminPanel] = useState(false);

  // 1. 核心身分驗證 (Rule 3)
  useEffect(() => {
    const initAuth = async () => {
      try {
        if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {
          await signInWithCustomToken(auth, __initial_auth_token);
        } else {
          await signInAnonymously(auth);
        }
      } catch (e) {
        console.error("Auth Failed", e);
      }
    };
    initAuth();
    const unsubscribe = onAuthStateChanged(auth, (u) => {
      setUser(u);
      if (u) setIsReady(true);
    });
    return () => unsubscribe();
  }, []);

  // 2. 資料同步 (嚴格路徑 Rule 1)
  useEffect(() => {
    if (!user) return;

    // 集合名稱必須直接在 artifacts/{appId}/public/data/ 之下
    const configRef = doc(db, 'artifacts', appId, 'public', 'data', 'v21_settings', 'dates');
    const ordersCol = collection(db, 'artifacts', appId, 'public', 'data', 'v21_members');

    const unsubDates = onSnapshot(configRef, (snap) => {
      if (snap.exists()) {
        const d = snap.data().list || [];
        setDates(d);
        if (d.length > 0 && !currentMonth) setCurrentMonth(d[0].split('/')[0] + '月');
      } else {
        setDoc(configRef, { list: INITIAL_DATES });
      }
    }, (err) => console.error("Snapshot Dates Error", err));

    const unsubOrders = onSnapshot(ordersCol, (snap) => {
      const o = {};
      snap.docs.forEach(d => { o[d.id] = d.data().orders || {}; });
      setOrders(o);
    }, (err) => console.error("Snapshot Orders Error", err));

    return () => { unsubDates(); unsubOrders(); };
  }, [user, currentMonth]);

  // 3. 計算邏輯
  const stats = useMemo(() => {
    const dailyCount = {};
    const shippingPerPerson = {};
    const memberList = Object.keys(orders).sort();
    const months = new Set();
    
    dates.forEach(d => {
      months.add(d.split('/')[0] + '月');
      let count = 0;
      memberList.forEach(m => { if (orders[m]?.[d]) count++; });
      dailyCount[d] = count;
      if (count >= 12) shippingPerPerson[d] = 25;
      else if (count === 11) shippingPerPerson[d] = 27;
      else if (count === 10) shippingPerPerson[d] = 30;
      else if (count > 0) shippingPerPerson[d] = Math.round(WEEKLY_SHIPPING / count);
      else shippingPerPerson[d] = 0;
    });

    return { dailyCount, shippingPerPerson, memberList, monthList: Array.from(months).sort((a,b) => parseInt(a) - parseInt(b)) };
  }, [orders, dates]);

  const currentDates = useMemo(() => dates.filter(d => d.split('/')[0] + '月' === currentMonth), [dates, currentMonth]);

  // 4. 功能操作 (加上身分檢查守衛)
  const showToast = (text, type = 'success') => {
    setStatusMsg({ text, type });
    setTimeout(() => setStatusMsg(null), 3000);
  };

  const handleImport = async () => {
    if (!user || isProcessing) return;
    setIsProcessing(true);
    try {
      await setDoc(doc(db, 'artifacts', appId, 'public', 'data', 'v21_settings', 'dates'), { list: INITIAL_DATES });
      for (const [name, dateArr] of Object.entries(INITIAL_IMPORT_DATA)) {
        const o = {}; dateArr.forEach(d => o[d] = true);
        await setDoc(doc(db, 'artifacts', appId, 'public', 'data', 'v21_members', name), { orders: o });
      }
      showToast("匯入成功！");
      setCurrentMonth("3月");
    } catch (e) { 
      showToast(`寫入錯誤: ${e.code || 'Unauthorized'}`, "error"); 
    } finally { setIsProcessing(false); }
  };

  const handleReset = async (name) => {
    if (!user || isProcessing) return;
    if (!window.confirm(`確認「${name}」結帳歸零？`)) return;
    setIsProcessing(true);
    try {
      // 物理覆寫法：直接發送空物件
      const memberRef = doc(db, 'artifacts', appId, 'public', 'data', 'v21_members', name);
      await setDoc(memberRef, { orders: {} });
      showToast(`${name} 已結帳歸零`);
    } catch (e) { 
      showToast(`操作失敗: ${e.code || 'WriteBlocked'}`, "error"); 
    } finally { setIsProcessing(false); }
  };

  const handleDelete = async (name) => {
    if (!user || isProcessing) return;
    if (!window.confirm(`確定移除「${name}」？`)) return;
    setIsProcessing(true);
    try {
      const memberRef = doc(db, 'artifacts', appId, 'public', 'data', 'v21_members', name);
      await deleteDoc(memberRef);
      showToast(`已移除：${name}`);
    } catch (e) { 
      showToast(`刪除失敗: ${e.code || 'Denied'}`, "error"); 
    } finally { setIsProcessing(false); }
  };

  const toggleOrder = async (name, date) => {
    if (isAdmin && showAdminPanel) return;
    const currentOrders = orders[name] || {};
    const isOrdered = !!currentOrders[date];
    if (!isOrdered && stats.dailyCount[date] >= MAX_BAGS_PER_WEEK) { showToast("上限 12 包已滿", "error"); return; }

    try {
      const memberRef = doc(db, 'artifacts', appId, 'public', 'data', 'v21_members', name);
      const updated = { ...currentOrders, [date]: !isOrdered };
      await setDoc(memberRef, { orders: updated });
    } catch (e) { showToast("連線受阻", "error"); }
  };

  if (!isReady) return <div className="h-screen flex items-center justify-center bg-green-50"><Loader2 className="animate-spin text-green-600" size={32}/></div>;

  return (
    <div className="min-h-screen bg-slate-50 pb-20 font-sans text-slate-800 antialiased overflow-x-hidden">
      {/* 導航列 */}
      <nav className="bg-green-700 text-white shadow-xl sticky top-0 z-[100] px-4 py-3">
        <div className="max-w-7xl mx-auto flex items-center justify-between">
          <div className="flex items-center gap-2">
            <ShoppingCart className="w-5 h-5 flex-shrink-0" />
            <h1 className="text-base font-black">生菜管理系統 v21</h1>
          </div>
          <div className="flex items-center gap-2">
            <Wifi className={`w-4 h-4 ${user ? 'text-green-300' : 'text-red-400'}`} />
            <button onClick={() => setIsAdmin(!isAdmin)} className={`px-3 py-1.5 rounded-full text-[10px] font-bold transition-all ${isAdmin ? 'bg-red-500 text-white' : 'bg-white/10'}`}>
              {isAdmin ? <Unlock size={12} className="inline mr-1"/> : <Lock size={12} className="inline mr-1"/>}
              {isAdmin ? '管理模式' : '檢視模式'}
            </button>
            {isAdmin && <button onClick={() => setShowAdminPanel(!showAdminPanel)} className="p-1.5 bg-white/10 rounded-lg"><Settings size={18} /></button>}
          </div>
        </div>
      </nav>

      <main className="max-w-7xl mx-auto px-2 sm:px-4 py-6 space-y-6">
        {/* 管理面板 */}
        {isAdmin && showAdminPanel && (
          <div className="bg-white p-6 rounded-3xl shadow-2xl border-2 border-red-50 animate-in slide-in-from-top-4">
            <div className="flex items-center justify-between mb-4 border-b pb-2">
              <h3 className="text-red-600 font-black flex items-center gap-2"><Settings size={20} /> 管理中心</h3>
              <button onClick={() => setShowAdminPanel(false)}><X size={20} className="text-slate-400"/></button>
            </div>
            <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
              <div className="bg-blue-50 p-4 rounded-xl text-center">
                <button onClick={handleImport} disabled={isProcessing} className="w-full bg-blue-600 hover:bg-blue-700 text-white py-3 rounded-lg font-bold flex items-center justify-center gap-2 disabled:opacity-50">
                   {isProcessing ? <Loader2 className="animate-spin" size={16}/> : <Database size={16} />} 恢復 14 位成員資料
                </button>
              </div>
              <div className="space-y-3">
                <div className="flex gap-2">
                  <input className="flex-1 border p-2 rounded-lg text-sm outline-none" placeholder="新增人名" value={newMemberName} onChange={e => setNewMemberName(e.target.value)}/>
                  <button onClick={async () => { if(!newMemberName) return; await setDoc(doc(db, 'artifacts', appId, 'public', 'data', 'v21_members', newMemberName), { orders: {} }); setNewMemberName(""); showToast("已新增"); }} className="bg-green-600 text-white px-4 rounded-lg font-bold text-xs">新增</button>
                </div>
                <div className="flex gap-2">
                  <input className="flex-1 border p-2 rounded-lg text-sm outline-none" placeholder="新日期 (4/7)" value={newDateInput} onChange={e => setNewDateInput(e.target.value)}/>
                  <button onClick={async () => { if(!newDateInput.includes('/')) return; const updated = [...dates, newDateInput].sort(); await setDoc(doc(db, 'artifacts', appId, 'public', 'data', 'v21_settings', 'dates'), { list: updated }); setNewDateInput(""); showToast("已新增"); }} className="bg-blue-600 text-white px-4 rounded-lg font-bold text-xs">新增日期</button>
                </div>
              </div>
            </div>
          </div>
        )}

        {/* 月份選單 */}
        <div className="flex items-center gap-2 bg-white p-2 rounded-2xl shadow-sm border border-slate-200 overflow-x-auto">
            {stats.monthList.map(m => (
                <button key={m} onClick={() => setCurrentMonth(m)} className={`px-6 py-2 rounded-xl font-black text-xs transition-all whitespace-nowrap ${currentMonth === m ? 'bg-green-600 text-white shadow-md' : 'text-slate-400 hover:bg-slate-100'}`}>
                    {m}份
                </button>
            ))}
        </div>

        {/* 訂購表格 (手機版雙向固定) */}
        <div className="bg-white rounded-3xl shadow-xl overflow-hidden border border-slate-200 relative">
          <div className="overflow-auto max-h-[60vh] relative">
            <table className="w-full text-left border-separate border-spacing-0">
              <thead className="sticky top-0 z-[60] bg-slate-900 text-white">
                <tr>
                  <th className="p-4 sticky left-0 z-[70] bg-slate-900 border-r border-slate-800 text-center text-[10px] font-black w-24 sm:w-32 shadow-md">成員姓名</th>
                  {currentDates.map(date => (
                    <th key={date} className="p-4 text-center min-w-[85px] border-b border-slate-800">
                      <div className="flex flex-col">
                        <span className="text-[8px] text-slate-500 font-normal">週二到貨</span>
                        <span className="text-sm font-black tracking-tighter">{date}</span>
                      </div>
                    </th>
                  ))}
                  <th className="p-4 text-right border-b border-slate-800 bg-slate-950 text-green-400 text-[10px] font-black min-w-[100px] sticky right-0 z-[70] shadow-md">應付總額</th>
                </tr>
              </thead>
              <tbody className="divide-y divide-slate-100">
                {stats.memberList.map(name => {
                  let totalBags = 0; let totalShip = 0;
                  dates.forEach(d => { if (orders[name]?.[d]) { totalBags++; totalShip += stats.shippingPerPerson[d]; } });
                  return (
                    <tr key={name} className="hover:bg-green-50 transition-colors group">
                      <td className="p-3 font-bold text-slate-700 sticky left-0 z-[40] bg-white border-r border-slate-100 flex items-center justify-between gap-1 shadow-sm">
                        <span className="truncate">{name}</span>
                        {isAdmin && <button onClick={() => handleDelete(name)} className="text-red-400 p-1 hover:bg-red-50 rounded transition-all"><Trash2 size={12}/></button>}
                      </td>
                      {currentDates.map(date => (
                        <td key={date} className="p-2 text-center">
                          <button onClick={() => toggleOrder(name, date)} className={`mx-auto w-10 h-10 rounded-xl flex items-center justify-center transition-all ${orders[name]?.[date] ? 'bg-green-600 text-white shadow-lg scale-110' : 'bg-slate-50 text-slate-200'}`}>
                            {orders[name]?.[date] ? <CheckCircle2 size={24} strokeWidth={3} /> : <Plus size={18} />}
                          </button>
                        </td>
                      ))}
                      <td className="p-4 text-right bg-slate-50 sticky right-0 z-[30] font-mono font-black text-green-700 text-sm border-l border-slate-100 shadow-sm">
                        ${(totalBags * LETTUCE_PRICE + totalShip).toLocaleString()}
                      </td>
                    </tr>
                  );
                })}
              </tbody>
            </table>
          </div>
        </div>

        {/* 結帳統計卡片 */}
        <div className="space-y-4">
          <div className="flex items-center gap-2 px-1">
            <Users size={18} className="text-green-600"/><h2 className="font-black text-lg text-slate-700 tracking-tighter">繳費與結帳清單</h2>
          </div>
          <div className="grid grid-cols-2 lg:grid-cols-4 gap-3">
            {stats.memberList.map(name => {
              let bags = 0; let ship = 0;
              dates.forEach(d => { if (orders[name]?.[d]) { bags++; ship += stats.shippingPerPerson[d]; } });
              if (bags === 0 && !isAdmin) return null;
              return (
                <div key={name} className="bg-white p-4 rounded-3xl border border-slate-200 shadow-sm flex flex-col justify-between hover:shadow-lg transition-all relative overflow-hidden group">
                  <div className="absolute top-0 left-0 w-1.5 h-full bg-green-500"></div>
                  <div>
                    <div className="flex justify-between items-start mb-2"><span className="font-black text-slate-800 text-base">{name}</span><span className="text-[10px] bg-slate-50 px-2 py-0.5 rounded text-slate-500 font-bold">{bags} 包</span></div>
                    <div className="text-[11px] text-slate-500 space-y-1 pb-3 border-b border-dashed border-slate-100">
                      <div className="flex justify-between"><span>生菜總費:</span><span>${bags * LETTUCE_PRICE}</span></div>
                      <div className="flex justify-between"><span>運費分擔:</span><span>${ship}</span></div>
                      <div className="flex justify-between pt-1 font-black text-green-600 text-lg tracking-tight"><span>結帳總額:</span><span>${(bags * LETTUCE_PRICE + ship).toLocaleString()}</span></div>
                    </div>
                  </div>
                  {isAdmin && (
                    <button onClick={() => handleReset(name)} disabled={bags === 0 || isProcessing} className={`mt-4 w-full py-2.5 rounded-xl text-[10px] font-black flex items-center justify-center gap-1 transition-all shadow-md ${bags === 0 ? 'bg-slate-50 text-slate-300' : 'bg-green-600 text-white hover:bg-green-700 active:scale-95'}`}>
                      <Check size={14}/> 確認匯款歸零
                    </button>
                  )}
                </div>
              );
            })}
          </div>
        </div>
      </main>
      
      {/* 浮動提示狀態 */}
      {statusMsg && (
        <div className={`fixed bottom-10 left-1/2 -translate-x-1/2 z-[200] flex items-center gap-2 px-10 py-4 rounded-full shadow-2xl animate-in slide-in-from-bottom-10 ${statusMsg.type === 'error' ? 'bg-red-600 text-white' : 'bg-green-800 text-white'}`}>
          {statusMsg.type === 'error' ? <AlertCircle size={20}/> : <CheckCircle2 size={20} />}<span className="font-black text-sm">{statusMsg.text}</span>
        </div>
      )}
    </div>
  );
}
```
