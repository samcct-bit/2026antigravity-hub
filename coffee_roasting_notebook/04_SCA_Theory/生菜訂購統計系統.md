import React, { useState, useEffect, useMemo } from 'react';
import { initializeApp } from 'firebase/app';
import { getAuth, signInAnonymously, onAuthStateChanged } from 'firebase/auth';
import { getFirestore, collection, doc, onSnapshot, setDoc, updateDoc } from 'firebase/firestore';
import { 
  Users, 
  ShoppingCart, 
  Truck, 
  Calendar, 
  CheckCircle2, 
  AlertCircle,
  Plus,
  Trash2,
  Save,
  Loader2
} from 'lucide-react';

// --- Firebase Configuration ---
const firebaseConfig = JSON.parse(__firebase_config);
const app = initializeApp(firebaseConfig);
const auth = getAuth(app);
const db = getFirestore(app);
const appId = typeof __app_id !== 'undefined' ? __app_id : 'lettuce-app-001';

const LETTUCE_PRICE = 230;
const WEEKLY_SHIPPING = 300;
const MAX_BAGS_PER_WEEK = 12;

// 預設日期清單
const DEFAULT_DATES = ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"];

// 初始成員清單
const INITIAL_USERS = [
  "金成", "蕙敏", "珮萱", "淑芬", "阿姿", "嘉鎂", "斯涵", "明智",
  "翠花", "政君", "睿婷", "亭愷", "俊明", "惠偵"
];

export default function App() {
  const [user, setUser] = useState(null);
  const [members, setMembers] = useState([]);
  const [orders, setOrders] = useState({}); // { memberName: { date: boolean } }
  const [loading, setLoading] = useState(true);
  const [newMemberName, setNewMemberName] = useState("");
  const [message, setMessage] = useState(null);

  // 1. Authentication
  useEffect(() => {
    const initAuth = async () => {
      try {
        await signInAnonymously(auth);
      } catch (error) {
        console.error("Auth error:", error);
      }
    };
    initAuth();
    const unsubscribe = onAuthStateChanged(auth, (currentUser) => {
      setUser(currentUser);
    });
    return () => unsubscribe();
  }, []);

  // 2. Data Fetching (Real-time Sync)
  useEffect(() => {
    if (!user) return;

    const dataPath = `/artifacts/${appId}/public/data/lettuce_orders`;
    const unsub = onSnapshot(collection(db, dataPath), (snapshot) => {
      const docs = snapshot.docs.map(d => ({ id: d.id, ...d.data() }));
      
      // 如果資料庫是空的，初始化成員
      if (docs.length === 0) {
        initializeData();
      } else {
        const orderMap = {};
        const memberList = [];
        docs.forEach(d => {
          memberList.push(d.id);
          orderMap[d.id] = d.orders || {};
        });
        setMembers(memberList.sort());
        setOrders(orderMap);
      }
      setLoading(false);
    }, (err) => {
      console.error("Firestore error:", err);
      setLoading(false);
    });

    return () => unsub();
  }, [user]);

  const initializeData = async () => {
    setLoading(true);
    for (const name of INITIAL_USERS) {
      const docRef = doc(db, `/artifacts/${appId}/public/data/lettuce_orders`, name);
      // 根據使用者提供的初始資料預填
      const initialOrders = {};
      DEFAULT_DATES.forEach(date => {
        // 全勤者
        if (["金成", "蕙敏", "珮萱", "淑芬", "阿姿", "嘉鎂", "斯涵", "明智"].includes(name)) {
          initialOrders[date] = true;
        }
        // 隔週者邏輯...
        if (name === "翠花" && ["2/24", "3/3", "3/10", "3/24"].includes(date)) initialOrders[date] = true;
        if (["政君", "睿婷", "亭愷"].includes(name) && ["3/3", "3/17", "3/31"].includes(date)) initialOrders[date] = true;
        if (["俊明", "惠偵"].includes(name) && ["2/24", "3/10", "3/24"].includes(date)) initialOrders[date] = true;
      });
      await setDoc(docRef, { orders: initialOrders });
    }
  };

  // 3. Logic & Calculations
  const stats = useMemo(() => {
    const dailyCount = {};
    DEFAULT_DATES.forEach(date => {
      dailyCount[date] = members.reduce((sum, member) => sum + (orders[member]?.[date] ? 1 : 0), 0);
    });

    const shippingPerPerson = {};
    DEFAULT_DATES.forEach(date => {
      const count = dailyCount[date];
      if (count === 10) shippingPerPerson[date] = 30;
      else if (count === 11) shippingPerPerson[date] = 27;
      else if (count === 12) shippingPerPerson[date] = 25;
      else if (count > 0) shippingPerPerson[date] = Math.round(WEEKLY_SHIPPING / count);
      else shippingPerPerson[date] = 0;
    });

    return { dailyCount, shippingPerPerson };
  }, [members, orders]);

  // 4. Handlers
  const toggleOrder = async (memberName, date) => {
    const currentStatus = orders[memberName]?.[date] || false;
    
    // 檢查上限
    if (!currentStatus && stats.dailyCount[date] >= MAX_BAGS_PER_WEEK) {
      showMessage(`抱歉，${date} 的訂購量已達上限 ${MAX_BAGS_PER_WEEK} 包！`, 'error');
      return;
    }

    const docRef = doc(db, `/artifacts/${appId}/public/data/lettuce_orders`, memberName);
    const updatedOrders = { ...orders[memberName], [date]: !currentStatus };
    
    try {
      await updateDoc(docRef, { orders: updatedOrders });
    } catch (err) {
      console.error("Update error:", err);
    }
  };

  const addMember = async () => {
    if (!newMemberName.trim()) return;
    if (members.includes(newMemberName)) {
      showMessage("該成員已存在", "error");
      return;
    }
    const docRef = doc(db, `/artifacts/${appId}/public/data/lettuce_orders`, newMemberName);
    await setDoc(docRef, { orders: {} });
    setNewMemberName("");
    showMessage(`已新增成員：${newMemberName}`);
  };

  const showMessage = (text, type = 'success') => {
    setMessage({ text, type });
    setTimeout(() => setMessage(null), 3000);
  };

  if (loading) {
    return (
      <div className="flex flex-col items-center justify-center min-h-screen bg-green-50">
        <Loader2 className="w-12 h-12 text-green-600 animate-spin mb-4" />
        <p className="text-green-800 font-medium">載入生菜資料中...</p>
      </div>
    );
  }

  return (
    <div className="min-h-screen bg-slate-50 pb-20">
      {/* Header */}
      <header className="bg-green-600 text-white p-6 shadow-lg sticky top-0 z-20">
        <div className="max-w-6xl mx-auto flex flex-col md:flex-row md:items-center justify-between gap-4">
          <div className="flex items-center gap-3">
            <ShoppingCart className="w-8 h-8" />
            <div>
              <h1 className="text-2xl font-bold">生菜訂購互動調查表</h1>
              <p className="text-green-100 text-sm">每週二到貨 · 每包 $230 · 運費分擔制</p>
            </div>
          </div>
          <div className="flex items-center gap-2 bg-white/10 p-2 rounded-lg backdrop-blur-sm">
            <input 
              type="text" 
              placeholder="新增姓名"
              className="bg-white text-slate-800 px-3 py-1.5 rounded-md text-sm outline-none w-32 md:w-40"
              value={newMemberName}
              onChange={(e) => setNewMemberName(e.target.value)}
            />
            <button 
              onClick={addMember}
              className="bg-green-500 hover:bg-green-400 p-1.5 rounded-md transition-colors"
            >
              <Plus size={20} />
            </button>
          </div>
        </div>
      </header>

      <main className="max-w-6xl mx-auto p-4 md:p-6 space-y-8">
        {/* Message Toast */}
        {message && (
          <div className={`fixed top-24 right-4 z-50 flex items-center gap-2 px-4 py-3 rounded-xl shadow-2xl transition-all animate-bounce ${
            message.type === 'error' ? 'bg-red-500 text-white' : 'bg-green-500 text-white'
          }`}>
            {message.type === 'error' ? <AlertCircle size={20} /> : <CheckCircle2 size={20} />}
            <span className="font-medium">{message.text}</span>
          </div>
        )}

        {/* Desktop Table View */}
        <div className="bg-white rounded-2xl shadow-xl overflow-hidden border border-slate-200">
          <div className="overflow-x-auto">
            <table className="w-full text-left border-collapse">
              <thead className="bg-slate-100 text-slate-600 text-sm uppercase tracking-wider">
                <tr>
                  <th className="p-4 border-b sticky left-0 bg-slate-100 z-10 w-32">成員姓名</th>
                  {DEFAULT_DATES.map(date => (
                    <th key={date} className="p-4 border-b text-center min-w-[100px]">
                      <div className="flex flex-col items-center">
                        <Calendar size={16} className="mb-1 text-green-600" />
                        <span>{date}</span>
                      </div>
                    </th>
                  ))}
                  <th className="p-4 border-b text-right bg-green-50 text-green-700 font-bold">應付總額</th>
                </tr>
              </thead>
              <tbody className="divide-y divide-slate-100">
                {members.map(member => {
                  let totalBags = 0;
                  let totalShipping = 0;
                  DEFAULT_DATES.forEach(date => {
                    if (orders[member]?.[date]) {
                      totalBags += 1;
                      totalShipping += stats.shippingPerPerson[date];
                    }
                  });
                  const totalCost = (totalBags * LETTUCE_PRICE) + totalShipping;

                  return (
                    <tr key={member} className="hover:bg-slate-50 transition-colors">
                      <td className="p-4 font-bold text-slate-700 sticky left-0 bg-white z-10 border-r shadow-[2px_0_5px_rgba(0,0,0,0.05)]">
                        {member}
                      </td>
                      {DEFAULT_DATES.map(date => (
                        <td key={date} className="p-4 text-center">
                          <button
                            onClick={() => toggleOrder(member, date)}
                            className={`w-10 h-10 rounded-xl flex items-center justify-center transition-all ${
                              orders[member]?.[date] 
                                ? 'bg-green-500 text-white shadow-md scale-110' 
                                : 'bg-slate-100 text-slate-300 hover:bg-slate-200'
                            }`}
                          >
                            {orders[member]?.[date] ? <CheckCircle2 size={24} /> : <Plus size={18} />}
                          </button>
                        </td>
                      ))}
                      <td className="p-4 text-right bg-green-50/50 font-mono font-bold text-green-700">
                        ${totalCost.toLocaleString()}
                      </td>
                    </tr>
                  );
                })}
              </tbody>
              <tfoot className="bg-slate-800 text-white">
                <tr>
                  <td className="p-4 font-bold sticky left-0 bg-slate-800">當週小計 (包)</td>
                  {DEFAULT_DATES.map(date => (
                    <td key={date} className="p-4 text-center font-bold">
                      <div className={`inline-block px-3 py-1 rounded-full ${
                        stats.dailyCount[date] >= MAX_BAGS_PER_WEEK ? 'bg-red-500' : 'bg-green-500'
                      }`}>
                        {stats.dailyCount[date]} / {MAX_BAGS_PER_WEEK}
                      </div>
                    </td>
                  ))}
                  <td className="p-4 text-right">
                    <span className="text-xs text-slate-400 block">全體合計</span>
                    ${Object.values(orders).reduce((acc, userOrders) => {
                      let subtotal = 0;
                      DEFAULT_DATES.forEach(date => {
                        if (userOrders?.[date]) {
                          subtotal += LETTUCE_PRICE + stats.shippingPerPerson[date];
                        }
                      });
                      return acc + subtotal;
                    }, 0).toLocaleString()}
                  </td>
                </tr>
                <tr>
                  <td className="p-4 text-sm font-medium sticky left-0 bg-slate-800">每人分擔運費</td>
                  {DEFAULT_DATES.map(date => (
                    <td key={date} className="p-4 text-center text-sm font-mono text-blue-300">
                      <div className="flex flex-center justify-center items-center gap-1">
                        <Truck size={14} />
                        ${stats.shippingPerPerson[date]}
                      </div>
                    </td>
                  ))}
                  <td className="p-4"></td>
                </tr>
              </tfoot>
            </table>
          </div>
        </div>

        {/* Info Cards */}
        <div className="grid grid-cols-1 md:grid-cols-3 gap-6">
          <div className="bg-white p-6 rounded-2xl shadow-md border-l-4 border-green-500">
            <h3 className="text-slate-500 text-sm font-bold flex items-center gap-2 mb-2">
              <ShoppingCart size={18} className="text-green-500" /> 生菜單價
            </h3>
            <p className="text-3xl font-bold text-slate-800">${LETTUCE_PRICE} <span className="text-sm font-normal text-slate-400">/ 包</span></p>
          </div>
          <div className="bg-white p-6 rounded-2xl shadow-md border-l-4 border-blue-500">
            <h3 className="text-slate-500 text-sm font-bold flex items-center gap-2 mb-2">
              <Truck size={18} className="text-blue-500" /> 週運費
            </h3>
            <p className="text-3xl font-bold text-slate-800">${WEEKLY_SHIPPING} <span className="text-sm font-normal text-slate-400">/ 每週</span></p>
          </div>
          <div className="bg-white p-6 rounded-2xl shadow-md border-l-4 border-orange-500">
            <h3 className="text-slate-500 text-sm font-bold flex items-center gap-2 mb-2">
              <AlertCircle size={18} className="text-orange-500" /> 運費分擔規則
            </h3>
            <ul className="text-sm text-slate-600 space-y-1">
              <li>• 12人：$25 /人</li>
              <li>• 11人：$27 /人</li>
              <li>• 10人：$30 /人</li>
            </ul>
          </div>
        </div>

        {/* Member Summary Section */}
        <div className="space-y-4">
          <h2 className="text-xl font-bold text-slate-800 flex items-center gap-2">
            <Users size={24} className="text-green-600" /> 成員繳費清單
          </h2>
          <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
            {members.map(member => {
              let bags = 0;
              let shipTotal = 0;
              DEFAULT_DATES.forEach(date => {
                if (orders[member]?.[date]) {
                  bags += 1;
                  shipTotal += stats.shippingPerPerson[date];
                }
              });
              if (bags === 0) return null;

              return (
                <div key={member} className="bg-white p-4 rounded-xl border border-slate-200 shadow-sm hover:shadow-md transition-shadow">
                  <div className="flex justify-between items-start mb-3">
                    <span className="font-bold text-lg text-slate-800">{member}</span>
                    <span className="bg-green-100 text-green-700 text-xs px-2 py-1 rounded-full font-bold">
                      {bags} 包
                    </span>
                  </div>
                  <div className="space-y-1 text-sm text-slate-500">
                    <div className="flex justify-between">
                      <span>生菜費:</span>
                      <span>${bags * LETTUCE_PRICE}</span>
                    </div>
                    <div className="flex justify-between border-b pb-1">
                      <span>運費分擔:</span>
                      <span>${shipTotal}</span>
                    </div>
                    <div className="flex justify-between pt-1 font-bold text-slate-800 text-base">
                      <span>應付總額:</span>
                      <span className="text-green-600">${(bags * LETTUCE_PRICE + shipTotal).toLocaleString()}</span>
                    </div>
                  </div>
                </div>
              );
            })}
          </div>
        </div>
      </main>

      {/* Footer Instructions */}
      <footer className="max-w-6xl mx-auto px-6 py-8 text-center text-slate-400 text-sm">
        <p>© 2025 生菜訂購統計小幫手</p>
        <p>點擊表格中的方塊即可「登記」或「取消」訂單。系統會即時同步至雲端。</p>
      </footer>
    </div>
  );
}
