            function handleNavChange(e) {
                const chapterId = e.target.value;
                if (chapterId) {
                    const element = document.getElementById(chapterId);
                    if (element) {
                        const headerOffset = dom.header.offsetHeight + 20;
                        const elementPosition = element.getBoundingClientRect().top;
                        const offsetPosition = elementPosition + window.pageYOffset - headerOffset;
                        window.scrollTo({ top: offsetPosition, behavior: "smooth" });
                    }
                }
            }

            function handleDeleteNote() {
                if (state.selectionRange) {
                    const node = state.selectionRange.commonAncestorContainer;
                    const highlightSpan = node.nodeType === Node.TEXT_NODE ? node.parentNode.closest('.highlight') : node.closest('.highlight');
                    
                    if (highlightSpan) {
                        const parent = highlightSpan.parentNode;
                        while (highlightSpan.firstChild) {
                            parent.insertBefore(highlightSpan.firstChild, highlightSpan);

                        parent.removeChild(highlightSpan);
                        parent.normalize();

            async function getAIResponse(prompt) {
                dom.aiOptions.classList.add('hidden');
                dom.aiQuestionArea.classList.add('hidden');
                dom.aiResponseArea.classList.remove('hidden');

                try {

            function toChineseNum(num) {
                const chineseNums = ['〇', '一', '二', '三', '四', '五', '六', '七', '八', '九', '十'];
                if (num <= 10) return chineseNums[num];
                if (num < 20) return '十' + chineseNums[num % 10];
                if (num < 100) {
