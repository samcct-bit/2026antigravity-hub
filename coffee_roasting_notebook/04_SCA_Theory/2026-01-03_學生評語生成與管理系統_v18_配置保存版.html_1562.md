  function buildPrompts() {
    const rows = document.querySelectorAll('#tableBody tr');
    let template = document.getElementById('templateInput').value;
    if (!template.trim()) template = defaultTemplate;
    let prompts = template + "\n";
    
    let has = false;
    rows.forEach(r => {
      const n = r.querySelector('.row-name').value.trim();
      const t = r.querySelector('.row-traits').value.trim();
      const l = r.querySelector('.row-len').value;
      const s = r.querySelector('.row-sub').value;

    });
    return has ? prompts : null;
  }

  async function runAIAutoMode() {
    let key = document.getElementById('apiKeyInput').value.trim();
    if (key) localStorage.setItem('gemini_api_key', key);
    else key = "";
    const mod = document.getElementById('modelSelect').value;
    const pt = buildPrompts();
    if (!pt) { showToast('請先輸入資料'); return; }

    document.getElementById('promptOutput').value = pt;
    document.getElementById('loadingOverlay').style.display = 'flex';

    try {
      const url = `https://generativelanguage.googleapis.com/v1beta/models/${mod}:generateContent?key=${apiKeyInput.value}`;
      const res = await fetch(url, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ contents: [{ parts: [{ text: pt }] }] })
      });
      const data = await res.json();
      if (!res.ok) throw new Error(data.error?.message || "呼叫失敗");
      const txt = data.candidates?.[0]?.content?.parts?.[0]?.text;
      if (!txt) throw new Error("回傳內容為空");
      
      document.getElementById('aiInput').value = txt;
      parseAIResponse();
      showToast('🎉 生成成功！');
    } catch (e) { alert(`發生錯誤：${e.message}`); }
    finally { document.getElementById('loadingOverlay').style.display = 'none'; }
  }

  function escapeRegExp(string) { return string.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'); }

  function parseAIResponse() {
    const aiText = document.getElementById('aiInput').value;
    if (!aiText) return;
    const rows = document.querySelectorAll('#tableBody tr');
    const allNames = Array.from(rows).map(r => r.querySelector('.row-name').value.trim()).filter(n => n);

    rows.forEach(row => {
      const name = row.querySelector('.row-name').value.trim();
      if (!name) return;
      const startRegex = new RegExp(`(?:^|\\n)[\\s\\*\\-\\[]*${escapeRegExp(name)}[\\s\\*\\-\\]]*[:：]`, 'i');
      const match = aiText.match(startRegex);
      if (match) {
        const startIndex = match.index + match[0].length;
        let content = aiText.substring(startIndex);
        let minNextIndex = content.length;
        allNames.forEach(otherName => {
          if (otherName !== name) {

            const nextMatch = content.match(nextRegex);
            if (nextMatch && nextMatch.index < minNextIndex) minNextIndex = nextMatch.index;
          }
        });
        let comment = content.substring(0, minNextIndex).trim().replace(/-{3,}$/g, '').trim();
        const resultCell = row.querySelector('.row-result');
        if (resultCell) {
          resultCell.value = comment;
          resultCell.style.backgroundColor = '#e8f5e9';
          setTimeout(() => resultCell.style.backgroundColor = '', 1500);
        }
      }
    });
  }
