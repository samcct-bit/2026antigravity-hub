    const { createApp, ref, computed } = Vue;

    createApp({
        setup() {
            const title = ref("七月份生菜訂購");
            const priceLarge = ref(230);
            const priceSmall = ref(160);
            const shippingCost = ref(300);

            // 預設四週
            const weeks = ref([
                { date: "7/7 (二)" },
                { date: "7/14 (二)" },
                { date: "7/21 (二)" },
                { date: "7/28 (二)" }
            ]);

            // 預設帶入一些名單
            const defaultNames = ["金成", "蕙敏", "珮萱", "淑芬", "政君"];
            const users = ref(defaultNames.map(name => ({
                name: name,
                orders: [
                    { large: 0, small: 0 }, { large: 0, small: 0 }, 
                    { large: 0, small: 0 }, { large: 0, small: 0 }
                ]
            })));

            // 新增使用者
            const addUser = () => {
                users.value.push({
                    name: "",
                    orders: weeks.value.map(() => ({ large: 0, small: 0 }))
                });
            };

            // 移除使用者
            const removeUser = (index) => {
                users.value.splice(index, 1);
            };

            // 計算某週的總包數 (大包+小包)
            const weeklyTotalBags = (wIndex) => {
                return users.value.reduce((sum, user) => {
                    return sum + (user.orders[wIndex].large || 0) + (user.orders[wIndex].small || 0);
                }, 0);
            };

            // 計算某週每包應分攤的運費 (四捨五入到整數)
            const weeklyShippingPerBag = (wIndex) => {
                const bags = weeklyTotalBags(wIndex);
                if (bags === 0) return 0;
                return Math.round(shippingCost.value / bags);
            };

            // 計算個人總費用
            const calculateUserTotal = (uIndex) => {
                let total = 0;
                const user = users.value[uIndex];

                user.orders.forEach((order, wIndex) => {
                    const lQty = order.large || 0;
                    const sQty = order.small || 0;
                    const totalBagsForWeek = lQty + sQty;

                    if (totalBagsForWeek > 0) {
                        // 菜錢
                        total += (lQty * priceLarge.value) + (sQty * priceSmall.value);
                        // 該週運費 (包數 * 當週單包運費)
                        total += (totalBagsForWeek * weeklyShippingPerBag(wIndex));
                    }
                });
                return total;
            };

            return {
                title, priceLarge, priceSmall, shippingCost,
                weeks, users,
                addUser, removeUser,
                weeklyTotalBags, weeklyShippingPerBag, calculateUserTotal
            };
        }
    }).mount('#app');
</script>
