const INITIAL_DATES = ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"];

export default function App() {
  const [user, setUser] = useState(null);
  const [orders, setOrders] = useState({});
  const [dates, setDates] = useState([]);
  const [isReady, setIsReady] = useState(false);
  const [isAdmin, setIsAdmin] = useState(false);
  const [currentMonth, setCurrentMonth] = useState("");
  const [isProcessing, setIsProcessing] = useState(false);
  const [newMemberName, setNewMemberName] = useState("");
  const [newDateInput, setNewDateInput] = useState("");
  const [statusMsg, setStatusMsg] = useState(null);

    const initAuth = async () => {
      try {
        if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {
          await signInWithCustomToken(auth, __initial_auth_token);
        } else {
          await signInAnonymously(auth);
        }
      } catch (e) { setIsReady(true); }
    };
    initAuth();
    onAuthStateChanged(auth, u => { setUser(u); if(u) setIsReady(true); });
  }, []);

  // 2. 同步資料
  useEffect(() => {
    if (!user) return;
    const configRef = doc(db, `artifacts/${appId}/public/data/config`, 'settings');
    const orderCol = collection(db, `artifacts/${appId}/public/data/lettuce_orders`);

    const unsubDates = onSnapshot(configRef, (docSnap) => {
      if (docSnap.exists()) {
        const d = docSnap.data().activeDates || [];
        setDates(d);
        if (d.length > 0 && !currentMonth) {
          setCurrentMonth(d[d.length - 1].split('/')[0] + '月');
        }
      } else {
        setDoc(configRef, { activeDates: INITIAL_DATES });
      }
    });

    const unsubOrders = onSnapshot(orderCol, (snapshot) => {
      const newOrders = {};
      snapshot.docs.forEach(d => { newOrders[d.id] = d.data().orders || {}; });
      setOrders(newOrders);
    });

    return () => { unsubDates(); unsubOrders(); };
  }, [user]);

  // 3. 計算統計
  const monthList = useMemo(() => {
    const months = new Set();
    dates.forEach(d => months.add(d.split('/')[0] + '月'));
    return Array.from(months).sort((a, b) => parseInt(a) - parseInt(b));
  }, [dates]);

  const currentDates = useMemo(() => {
    return dates.filter(d => d.split('/')[0] + '月' === currentMonth);

    const shippingPerPerson = {};

      dailyCount[date] = count;
      if (count >= 12) shippingPerPerson[date] = 25;

      else if (count === 10) shippingPerPerson[date] = 30;
      else if (count > 0) shippingPerPerson[date] = Math.round(WEEKLY_SHIPPING / count);
      else shippingPerPerson[date] = 0;
    });
    return { dailyCount, shippingPerPerson, memberList };

  const handleImport = async () => {

      Object.entries(INITIAL_IMPORT_DATA).forEach(([name, datesArr]) => {
        const docRef = doc(db, `artifacts/${appId}/public/data/lettuce_orders`, name);

        batch.set(docRef, { orders: o });
      });
      await batch.commit();
      showToast("原始資料匯入成功！");
    } catch (e) { showToast("匯入失敗", "error"); }
    finally { setIsProcessing(false); }

    setIsProcessing(true);
    try {
      const docRef = doc(db, `artifacts/${appId}/public/data/lettuce_orders`, name);
      await updateDoc(docRef, { orders: {} });
      showToast(`${name} 結帳成功，費用已歸零。`);
    } catch (e) { showToast("結帳失敗", "error"); }
    finally { setIsProcessing(false); }
  };

  const addDate = async () => {
    if (!newDateInput.includes('/')) { showToast("格式需為 4/1", "error"); return; }
    const updatedDates = [...dates, newDateInput].sort((a,b) => {
      const [am, ad] = a.split('/').map(Number);
      const [bm, bd] = b.split('/').map(Number);
      return am !== bm ? am - bm : ad - bd;
    });
    await updateDoc(doc(db, `artifacts/${appId}/public/data/config`, 'settings'), { activeDates: updatedDates });
    setNewDateInput("");
    showToast(`新增日期：${newDateInput}`);
  };

  const toggleOrder = async (name, date) => {
    if (isAdmin && showAdminPanel) return; // 管理面板開啟時禁用點選
    const isOrdered = orders[name]?.[date] || false;
    if (!isOrdered && stats.dailyCount[date] >= MAX_BAGS_PER_WEEK) {
      showToast(`${date} 已達上限 12 包！`, 'error');
      return;
    }
    const docRef = doc(db, `artifacts/${appId}/public/data/lettuce_orders`, name);
    await setDoc(docRef, { orders: { [date]: !isOrdered } }, { merge: true });
  };

  const showToast = (text, type = 'success') => {
    setStatusMsg({ text, type });
    setTimeout(() => setStatusMsg(null), 3000);
  };

  if (!isReady) return <div className="h-screen flex items-center justify-center bg-green-50"><Loader2 className="animate-spin text-green-600"/></div>;

          <div className="flex items-center gap-2">
            <ShoppingCart className="w-5 h-5 sm:w-6 sm:h-6" />
            <h1 className="text-sm sm:text-lg font-black tracking-tight">生菜訂購系統 v7</h1>
          </div>
          
          <div className="flex items-center gap-2">
            <button

            <div className="grid grid-cols-1 lg:grid-cols-3 gap-8">
              {/* 匯入與初始化 */}

                <button 
                  onClick={handleImport}

                  匯入原始 14 位成員

                      {currentDates.map(date => (

                          <button
                            onClick={() => toggleOrder(name, date)}
                            className={`mx-auto w-10 h-10 sm:w-11 sm:h-11 rounded-2xl flex items-center justify-center transition-all ${
                              orders[name]?.[date] 
                                ? 'bg-green-600 text-white shadow-lg scale-110' 
                                : 'bg-slate-50 text-slate-200 hover:bg-slate-200'
                            }`}
                          >
                            {orders[name]?.[date] ? <CheckCircle2 size={24} strokeWidth={3} /> : <Plus size={18} />}
                          </button>
                        </td>
                      ))}
                      <td className="p-4 text-right bg-slate-50/80 sticky right-0 z-10 font-mono font-black text-green-700">
                        ${(totalBags * LETTUCE_PRICE + totalShip).toLocaleString()}

                  <td className="p-4 sticky left-0 bg-slate-900 border-r border-slate-800">當週數量</td>

                    <td key={date} className="p-4 text-center">
                      <div className={`px-2 py-1 rounded-full inline-block ${stats.dailyCount[date] >= MAX_BAGS_PER_WEEK ? 'bg-red-500' : 'bg-green-600'}`}>
                        {stats.dailyCount[date]} / 12
                      </div>
                    </td>
                  ))}
                  <td className="p-4 text-right uppercase">Summary</td>
                </tr>
                <tr className="bg-slate-800 text-blue-300">
                  <td className="p-2 sticky left-0 bg-slate-800 border-r border-slate-700 text-center">分擔運費</td>
                  {currentDates.map(date => (
                    <td key={date} className="p-2 text-center font-mono">

                          <Truck size={10} /> ${stats.shippingPerPerson[date]}
                      </div>
                    </td>
                  ))}
                  <td className="p-2"></td>
                </tr>
              </tfoot>
            </table>
          </div>
        </div>

        {/* 個人詳細統計區 */}
        <div className="space-y-4">
          <div className="flex items-center justify-between px-1">
            <h2 className="font-black text-lg text-slate-700 flex items-center gap-2">
              <Users size={20} className="text-green-600"/> 成員繳費卡片
            </h2>
            {isAdmin && <span className="text-[10px] bg-red-100 text-red-600 px-3 py-1 rounded-full font-bold">管理模式：可操作「已匯款結帳」</span>}
          </div>
          <div className="grid grid-cols-2 sm:grid-cols-2 lg:grid-cols-4 gap-3">
            {stats.memberList.map(name => {
              let bags = 0; let ship = 0;
              dates.forEach(d => { if (orders[name]?.[d]) { bags++; ship += stats.shippingPerPerson[d]; } });
              if (bags === 0 && !isAdmin) return null;

              return (
                <div key={name} className="bg-white p-4 rounded-3xl border border-slate-200 shadow-sm relative overflow-hidden flex flex-col justify-between">
                  <div className="absolute top-0 left-0 w-1 h-full bg-green-500"></div>
                  <div>
                    <div className="flex justify-between items-start mb-2">
                      <span className="font-black text-slate-800">{name}</span>
                      <span className="text-[10px] bg-slate-100 px-2 py-0.5 rounded text-slate-500">{bags}包</span>
                    </div>
                    <div className="text-[11px] text-slate-500 space-y-1">
                      <div className="flex justify-between"><span>菜費:</span><span>${bags * LETTUCE_PRICE}</span></div>
                      <div className="flex justify-between border-b border-dashed pb-1"><span>運費:</span><span>${ship}</span></div>
                      <div className="flex justify-between pt-1 font-black text-green-600 text-lg"><span>總計:</span><span>${(bags * LETTUCE_PRICE + ship).toLocaleString()}</span></div>
                    </div>
                  </div>
                  
                  {isAdmin && (
                    <button 
                      onClick={() => handleSettle(name)}
                      disabled={bags === 0}
                      className={`mt-4 w-full py-2 rounded-xl text-[10px] font-black flex items-center justify-center gap-1 transition-all ${
                        bags === 0 ? 'bg-slate-50 text-slate-300' : 'bg-green-100 text-green-700 hover:bg-green-200'
                      }`}
                    >
                      <CreditCard size={14}/> 已匯款(結帳歸零)
                    </button>
                  )}
                </div>
              );
            })}
          </div>
        </div>
      </main>

      {/* 提示彈窗 */}
      {statusMsg && (
        <div className={`fixed bottom-6 left-1/2 -translate-x-1/2 z-[100] flex items-center gap-2 px-8 py-3 rounded-full shadow-2xl animate-in slide-in-from-bottom-4 ${
          statusMsg.type === 'error' ? 'bg-red-600 text-white' : 'bg-green-800 text-white'
        }`}>
          <CheckCircle2 size={18} /> <span className="font-bold text-sm">{statusMsg.text}</span>
        </div>
      )}
    </div>
  );
}
