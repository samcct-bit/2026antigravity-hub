import React, { useState, useEffect, useMemo } from 'react';
import { initializeApp } from 'firebase/app';
import { getAuth, signInAnonymously, onAuthStateChanged, signInWithCustomToken } from 'firebase/auth';
import { getFirestore, collection, doc, onSnapshot, setDoc, updateDoc, deleteDoc, writeBatch, getDoc } from 'firebase/firestore';
import { 
  Users, 
  ShoppingCart, 
  Truck, 
  Calendar, 
  CheckCircle2, 
  AlertCircle,
  Plus,
  Trash2,
  Loader2,
  Database,
  Lock,
  Unlock,
  ChevronLeft,
  ChevronRight,
  Settings,
  CalendarPlus
} from 'lucide-react';

// --- Firebase 配置 ---
const firebaseConfig = JSON.parse(__firebase_config);
const app = initializeApp(firebaseConfig);
const auth = getAuth(app);
const db = getFirestore(app);
const appId = typeof __app_id !== 'undefined' ? __app_id : 'lettuce-master-v6';

const LETTUCE_PRICE = 230;
const WEEKLY_SHIPPING = 300;
const MAX_BAGS_PER_WEEK = 12;

// 預設初始日期
const INITIAL_DATES = ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"];

export default function App() {
  const [user, setUser] = useState(null);
  const [orders, setOrders] = useState({});
  const [dates, setDates] = useState([]); // 從資料庫讀取的動態日期
  const [isReady, setIsReady] = useState(false);
  const [isAdmin, setIsAdmin] = useState(false); // 管理員模式開關
  const [currentMonth, setCurrentMonth] = useState(""); // 當前檢視的月份 (如 "3月")
  const [isImporting, setIsImporting] = useState(false);
  const [newMemberName, setNewMemberName] = useState("");
  const [newDateInput, setNewDateInput] = useState("");
  const [statusMsg, setStatusMsg] = useState(null);
  const [deleteConfirm, setDeleteConfirm] = useState(null);
  const [showAdminPanel, setShowAdminPanel] = useState(false);

  // 1. 認證
  useEffect(() => {
    const initAuth = async () => {
      try {
        if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {
          await signInWithCustomToken(auth, __initial_auth_token);
        } else {
          await signInAnonymously(auth);
        }
      } catch (e) { setIsReady(true); }
    };
    initAuth();
    onAuthStateChanged(auth, u => { setUser(u); if(u) setIsReady(true); });
  }, []);

  // 2. 同步資料 (日期與訂單)
  useEffect(() => {
    if (!user) return;
    const configRef = doc(db, `artifacts/${appId}/public/data/config`, 'settings');
    const orderCol = collection(db, `artifacts/${appId}/public/data/lettuce_orders`);

    // 監聽日期設定
    const unsubDates = onSnapshot(configRef, (docSnap) => {
      if (docSnap.exists()) {
        const d = docSnap.data().activeDates || [];
        setDates(d);
        // 初始化月份分頁
        if (d.length > 0) {
          const latest = d[d.length - 1];
          const m = latest.split('/')[0] + '月';
          if (!currentMonth) setCurrentMonth(m);
        }
      } else {
        // 若無設定，初始化預設日期
        setDoc(configRef, { activeDates: INITIAL_DATES });
      }
    });

    // 監聽訂單
    const unsubOrders = onSnapshot(orderCol, (snapshot) => {
      const newOrders = {};
      snapshot.docs.forEach(d => { newOrders[d.id] = d.data().orders || {}; });
      setOrders(newOrders);
    });

    return () => { unsubDates(); unsubOrders(); };
  }, [user]);

  // 3. 計算與過濾
  const monthList = useMemo(() => {
    const months = new Set();
    dates.forEach(d => months.add(d.split('/')[0] + '月'));
    return Array.from(months).sort((a, b) => parseInt(a) - parseInt(b));
  }, [dates]);

  const currentDates = useMemo(() => {
    return dates.filter(d => d.split('/')[0] + '月' === currentMonth);
  }, [dates, currentMonth]);

  const stats = useMemo(() => {
    const dailyCount = {};
    const shippingPerPerson = {};
    const memberList = Object.keys(orders).sort();
    
    dates.forEach(date => {
      let count = 0;
      memberList.forEach(m => { if (orders[m]?.[date]) count++; });
      dailyCount[date] = count;
      if (count >= 12) shippingPerPerson[date] = 25;
      else if (count === 11) shippingPerPerson[date] = 27;
      else if (count === 10) shippingPerPerson[date] = 30;
      else if (count > 0) shippingPerPerson[date] = Math.round(WEEKLY_SHIPPING / count);
      else shippingPerPerson[date] = 0;
    });
    return { dailyCount, shippingPerPerson, memberList };
  }, [orders, dates]);

  // 4. 功能操作
  const addDate = async () => {
    if (!newDateInput.includes('/')) { showToast("格式需為 4/1", "error"); return; }
    if (dates.includes(newDateInput)) { showToast("日期已存在", "error"); return; }
    const updatedDates = [...dates, newDateInput].sort((a,b) => {
        const [am, ad] = a.split('/').map(Number);
        const [bm, bd] = b.split('/').map(Number);
        return am !== bm ? am - bm : ad - bd;
    });
    await updateDoc(doc(db, `artifacts/${appId}/public/data/config`, 'settings'), { activeDates: updatedDates });
    setNewDateInput("");
    showToast(`已新增日期 ${newDateInput}`);
  };

  const removeDate = async (dateToRemove) => {
    if (!window.confirm(`確定要刪除日期 ${dateToRemove} 嗎？此日期的所有訂購紀錄會同步刪除。`)) return;
    const updatedDates = dates.filter(d => d !== dateToRemove);
    await updateDoc(doc(db, `artifacts/${appId}/public/data/config`, 'settings'), { activeDates: updatedDates });
  };

  const toggleOrder = async (name, date) => {
    const isOrdered = orders[name]?.[date] || false;
    if (!isOrdered && stats.dailyCount[date] >= MAX_BAGS_PER_WEEK) {
      showToast(`${date} 已達上限 ${MAX_BAGS_PER_WEEK} 包！`, 'error');
      return;
    }
    const docRef = doc(db, `artifacts/${appId}/public/data/lettuce_orders`, name);
    await setDoc(docRef, { orders: { [date]: !isOrdered } }, { merge: true });
  };

  const showToast = (text, type = 'success') => {
    setStatusMsg({ text, type });
    setTimeout(() => setStatusMsg(null), 3000);
  };

  if (!isReady) return <div className="h-screen flex items-center justify-center bg-green-50 text-green-700 font-bold"><Loader2 className="animate-spin mr-2"/>系統初始化中...</div>;

  return (
    <div className="min-h-screen bg-slate-50 pb-20 font-sans text-slate-900 overflow-x-hidden">
      {/* 頂部導航 */}
      <nav className="bg-green-800 text-white shadow-xl sticky top-0 z-[60] px-4 py-3">
        <div className="max-w-7xl mx-auto flex items-center justify-between">
          <div className="flex items-center gap-2">
            <ShoppingCart className="w-5 h-5 sm:w-6 sm:h-6" />
            <h1 className="text-sm sm:text-lg font-black tracking-tight">生菜訂購分擔系統</h1>
          </div>
          
          <div className="flex items-center gap-2">
            {/* 管理鎖 */}
            <button 
              onClick={() => setIsAdmin(!isAdmin)}
              className={`flex items-center gap-1 px-3 py-1.5 rounded-full text-xs font-bold transition-all ${
                isAdmin ? 'bg-red-500 text-white animate-pulse' : 'bg-white/20 text-green-100'
              }`}
            >
              {isAdmin ? <Unlock size={14} /> : <Lock size={14} />}
              {isAdmin ? '管理模式：開啟中' : '唯讀模式'}
            </button>
            {isAdmin && (
              <button onClick={() => setShowAdminPanel(!showAdminPanel)} className="p-1.5 bg-white/10 rounded-lg">
                <Settings size={18} />
              </button>
            )}
          </div>
        </div>
      </nav>

      <main className="max-w-7xl mx-auto px-2 sm:px-4 py-6 space-y-6">
        
        {/* 管理面板 (新增成員與日期) */}
        {isAdmin && showAdminPanel && (
          <div className="bg-white p-6 rounded-3xl shadow-2xl border-2 border-red-100 animate-in slide-in-from-top-4">
            <h3 className="text-red-600 font-black flex items-center gap-2 mb-4">
              <Settings size={20} /> 管理員設定面板
            </h3>
            <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
              <div className="space-y-3">
                <label className="text-xs font-bold text-slate-400 uppercase tracking-widest">新增成員</label>
                <div className="flex gap-2">
                  <input 
                    className="flex-1 border p-2 rounded-xl outline-none focus:ring-2 focus:ring-green-400"
                    placeholder="輸入姓名..."
                    value={newMemberName}
                    onChange={e => setNewMemberName(e.target.value)}
                  />
                  <button onClick={async () => {
                    if(!newMemberName) return;
                    await setDoc(doc(db, `artifacts/${appId}/public/data/lettuce_orders`, newMemberName), { orders: {} });
                    setNewMemberName("");
                    showToast("已新增成員");
                  }} className="bg-green-600 text-white px-4 py-2 rounded-xl font-bold">確認</button>
                </div>
              </div>
              <div className="space-y-3">
                <label className="text-xs font-bold text-slate-400 uppercase tracking-widest">新增到貨日期 (如 4/1)</label>
                <div className="flex gap-2">
                  <input 
                    className="flex-1 border p-2 rounded-xl outline-none focus:ring-2 focus:ring-blue-400"
                    placeholder="月/日..."
                    value={newDateInput}
                    onChange={e => setNewDateInput(e.target.value)}
                  />
                  <button onClick={addDate} className="bg-blue-600 text-white px-4 py-2 rounded-xl font-bold">新增日期</button>
                </div>
              </div>
            </div>
            <div className="mt-6 pt-6 border-t border-slate-100">
                <p className="text-xs text-slate-400 mb-2 font-bold">當前所有日期：</p>
                <div className="flex flex-wrap gap-2">
                    {dates.map(d => (
                        <div key={d} className="bg-slate-100 px-3 py-1 rounded-full text-xs font-bold flex items-center gap-2">
                            {d} 
                            <button onClick={() => removeDate(d)} className="text-red-400 hover:text-red-600"><X size={14}/></button>
                        </div>
                    ))}
                </div>
            </div>
          </div>
        )}

                {stats.memberList.map(name => {
                  let bags = 0; let ship = 0;

                  // 當月統計

                              orders[name]?.[date]

        }`}>
          <CheckCircle2 size={18} /> <span className="font-bold text-sm tracking-wide">{statusMsg.text}</span>

        <div className="inline-flex items-center gap-4 px-6 py-2 bg-slate-200 rounded-full text-[10px] font-bold text-slate-500 uppercase tracking-widest mb-4">
            <span>Market Price: $230/Bag</span>
            <span className="w-1 h-1 bg-slate-400 rounded-full"></span>
            <span>Shipment: $300/Weekly</span>
        </div>
        <p className="text-slate-400 text-[10px]">生菜團購管理系統 v6.0 · 支援無限月份擴充與動態權限控管</p>
      </footer>
    </div>
  );
}
