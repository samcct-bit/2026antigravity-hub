## 📋 對話理論筆記
```react
import React, { useState, useEffect, useMemo } from 'react';
import { initializeApp } from 'firebase/app';
import { getAuth, signInAnonymously, onAuthStateChanged } from 'firebase/auth';
import { getFirestore, collection, doc, onSnapshot, setDoc, updateDoc, deleteDoc } from 'firebase/firestore';
import { 
  Users, 
  ShoppingCart, 
  Truck, 
  Calendar, 
  CheckCircle2, 
  AlertCircle,
  Plus,
  Trash2,
  Loader2,
  Database,
  Lock,
  Unlock,
  Settings,
  X,
  CreditCard,
  Wifi,
  WifiOff
} from 'lucide-react';

// --- Firebase 配置 ---
const firebaseConfig = JSON.parse(__firebase_config);
const app = initializeApp(firebaseConfig);
const auth = getAuth(app);
const db = getFirestore(app);
const appId = typeof __app_id !== 'undefined' ? __app_id : 'lettuce-final-v9';

const LETTUCE_PRICE = 230;
const WEEKLY_SHIPPING = 300;
const MAX_BAGS_PER_WEEK = 12;

// 預設資料
const INITIAL_IMPORT_DATA = {
  "金成": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "蕙敏": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "珮萱": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "淑芬": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "阿姿": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "嘉鎂": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "斯涵": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "明智": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "翠花": ["2/24", "3/3", "3/10", "3/24"],
  "政君": ["3/3", "3/17", "3/31"],
  "睿婷": ["3/3", "3/17", "3/31"],
  "亭愷": ["3/3", "3/17", "3/31"],
  "俊明": ["2/24", "3/10", "3/24"],
  "惠偵": ["2/24", "3/10", "3/24"]
};
const INITIAL_DATES = ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"];

export default function App() {
  const [user, setUser] = useState(null);
  const [orders, setOrders] = useState({});
  const [dates, setDates] = useState([]);
  const [isReady, setIsReady] = useState(false);
  const [isAdmin, setIsAdmin] = useState(false);
  const [currentMonth, setCurrentMonth] = useState("");
  const [isProcessing, setIsProcessing] = useState(false);
  const [newMemberName, setNewMemberName] = useState("");
  const [newDateInput, setNewDateInput] = useState("");
  const [statusMsg, setStatusMsg] = useState(null);
  const [showAdminPanel, setShowAdminPanel] = useState(false);

  // 1. 初始化連線 & 登入
  useEffect(() => {
    const initApp = async () => {
      try {
        const result = await signInAnonymously(auth);
        setUser(result.user);
        setIsReady(true);
      } catch (e) {
        console.error("Login failed", e);
        setIsReady(true); // 即使失敗也釋放畫面，讓使用者看到連線錯誤狀態
      }
    };
    initApp();
    const unsubscribe = onAuthStateChanged(auth, setUser);
    return () => unsubscribe();
  }, []);

  // 2. 資料同步 (嚴格路徑監聽)
  useEffect(() => {
    if (!user) return;

    // Rule 1: /artifacts/{appId}/public/data/{collection}
    const configDocRef = doc(db, 'artifacts', appId, 'public', 'data', 'config', 'settings');
    const ordersColRef = collection(db, 'artifacts', appId, 'public', 'data', 'lettuce_orders');

    const unsubDates = onSnapshot(configDocRef, (snapshot) => {
      if (snapshot.exists()) {
        const d = snapshot.data().activeDates || [];
        setDates(d);
        if (d.length > 0 && !currentMonth) {
          setCurrentMonth(d[d.length - 1].split('/')[0] + '月');
        }
      } else {
        // 首次初始化
        setDoc(configDocRef, { activeDates: INITIAL_DATES });
      }
    }, (err) => showToast("日期讀取失敗，請確認權限", "error"));

    const unsubOrders = onSnapshot(ordersColRef, (snapshot) => {
      const o = {};
      snapshot.docs.forEach(d => { o[d.id] = d.data().orders || {}; });
      setOrders(o);
    }, (err) => showToast("訂單讀取失敗", "error"));

    return () => { unsubDates(); unsubOrders(); };
  }, [user, currentMonth]);

  // 3. 核心運算
  const monthList = useMemo(() => {
    const months = new Set();
    dates.forEach(d => months.add(d.split('/')[0] + '月'));
    return Array.from(months).sort((a, b) => parseInt(a) - parseInt(b));
  }, [dates]);

  const currentDates = useMemo(() => {
    return dates.filter(d => d.split('/')[0] + '月' === currentMonth);
  }, [dates, currentMonth]);

  const stats = useMemo(() => {
    const dailyCount = {};
    const shippingPerPerson = {};
    const memberList = Object.keys(orders).sort();
    
    dates.forEach(date => {
      let count = 0;
      memberList.forEach(m => { if (orders[m]?.[date]) count++; });
      dailyCount[date] = count;
      if (count >= 12) shippingPerPerson[date] = 25;
      else if (count === 11) shippingPerPerson[date] = 27;
      else if (count === 10) shippingPerPerson[date] = 30;
      else if (count > 0) shippingPerPerson[date] = Math.round(WEEKLY_SHIPPING / count);
      else shippingPerPerson[date] = 0;
    });
    return { dailyCount, shippingPerPerson, memberList };
  }, [orders, dates]);

  // 4. 功能函數
  const handleImport = async () => {
    if (!user) { showToast("尚未登入，請稍候", "error"); return; }
    setIsProcessing(true);
    try {
      // 逐筆寫入取代 Batch，提高成功率並方便除錯
      const configRef = doc(db, 'artifacts', appId, 'public', 'data', 'config', 'settings');
      await setDoc(configRef, { activeDates: INITIAL_DATES });

      for (const [name, dateArr] of Object.entries(INITIAL_IMPORT_DATA)) {
        const orderDocRef = doc(db, 'artifacts', appId, 'public', 'data', 'lettuce_orders', name);
        const orderObj = {};
        dateArr.forEach(d => orderObj[d] = true);
        await setDoc(orderDocRef, { orders: orderObj });
      }
      
      showToast("匯入完成！");
      setCurrentMonth("3月");
    } catch (e) {
      console.error(e);
      showToast("寫入受阻，請聯絡管理員確認權限設定", "error");
    } finally {
      setIsProcessing(false);
    }
  };

  const handleSettle = async (name) => {
    if (!window.confirm(`確認「${name}」已繳費？`)) return;
    setIsProcessing(true);
    try {
      const docRef = doc(db, 'artifacts', appId, 'public', 'data', 'lettuce_orders', name);
      await updateDoc(docRef, { orders: {} });
      showToast(`${name} 結帳歸零`);
    } catch (e) { showToast("結帳失敗", "error"); }
    finally { setIsProcessing(false); }
  };

  const toggleOrder = async (name, date) => {
    if (isAdmin && showAdminPanel) return;
    const isOrdered = orders[name]?.[date] || false;
    if (!isOrdered && stats.dailyCount[date] >= MAX_BAGS_PER_WEEK) {
      showToast("該週已滿 12 包", "error"); return;
    }
    const docRef = doc(db, 'artifacts', appId, 'public', 'data', 'lettuce_orders', name);
    try {
      await setDoc(docRef, { orders: { [date]: !isOrdered } }, { merge: true });
    } catch (e) { showToast("更新失敗", "error"); }
  };

  const showToast = (text, type = 'success') => {
    setStatusMsg({ text, type });
    setTimeout(() => setStatusMsg(null), 3000);
  };

  if (!isReady) return (
    <div className="h-screen flex flex-col items-center justify-center bg-green-50">
      <Loader2 className="animate-spin text-green-600 mb-2" size={32}/>
      <p className="text-green-800 font-bold">正在建立安全連線...</p>
    </div>
  );

  return (
    <div className="min-h-screen bg-gray-50 pb-20 font-sans text-slate-800">
      {/* 導覽列 */}
      <nav className="bg-green-700 text-white shadow-xl sticky top-0 z-[60] px-4 py-3">
        <div className="max-w-7xl mx-auto flex items-center justify-between">
          <div className="flex items-center gap-2">
            <ShoppingCart className="w-5 h-5" />
            <h1 className="text-base font-black">生菜訂購統計 v9</h1>
          </div>
          
          <div className="flex items-center gap-2">
            {user ? <Wifi size={14} className="text-green-300" /> : <WifiOff size={14} className="text-red-300" />}
            <button 
              onClick={() => setIsAdmin(!isAdmin)}
              className={`px-3 py-1.5 rounded-full text-[10px] font-bold transition-all flex items-center gap-1 ${
                isAdmin ? 'bg-red-500 text-white' : 'bg-white/20'
              }`}
            >
              {isAdmin ? <Unlock size={12}/> : <Lock size={12}/>}
              {isAdmin ? '管理模式' : '檢視'}
            </button>
            {isAdmin && (
              <button onClick={() => setShowAdminPanel(!showAdminPanel)} className="p-1.5 bg-white/10 rounded-lg hover:bg-white/20 transition-all">
                <Settings size={18} />
              </button>
            )}
          </div>
        </div>
      </nav>

      <main className="max-w-7xl mx-auto px-2 sm:px-4 py-6 space-y-6">
        
        {/* 管理介面 */}
        {isAdmin && showAdminPanel && (
          <div className="bg-white p-6 rounded-3xl shadow-2xl border-2 border-red-50 animate-in zoom-in-95">
            <div className="flex items-center justify-between mb-6 border-b pb-4">
              <h3 className="text-red-600 font-black flex items-center gap-2">
                <Settings size={20} /> 管理員控制台
              </h3>
              <button onClick={() => setShowAdminPanel(false)}><X size={20} className="text-slate-400"/></button>
            </div>
            
            <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
              <div className="bg-blue-50 p-5 rounded-2xl border border-blue-100">
                <p className="text-[10px] font-black text-blue-500 uppercase mb-3">資料庫重置 / 匯入</p>
                <button 
                  onClick={handleImport}
                  disabled={isProcessing}
                  className="w-full bg-blue-600 hover:bg-blue-700 text-white py-3 rounded-xl font-bold flex items-center justify-center gap-2 shadow-lg disabled:opacity-50 transition-all active:scale-95"
                >
                  {isProcessing ? <Loader2 className="animate-spin" size={18}/> : <Database size={18} />}
                  一鍵匯入 14 位成員資料
                </button>
                <p className="mt-2 text-[9px] text-blue-400 italic">※ 若遇到失敗，請確認下方連線圖示為綠色。</p>
              </div>

              <div className="space-y-4">
                <div className="flex gap-2">
                  <input className="flex-1 border p-2 rounded-xl text-sm outline-none focus:ring-2 focus:ring-green-400" placeholder="新成員姓名" value={newMemberName} onChange={e => setNewMemberName(e.target.value)}/>
                  <button onClick={async () => {
                    if(!newMemberName) return;
                    await setDoc(doc(db, 'artifacts', appId, 'public', 'data', 'lettuce_orders', newMemberName), { orders: {} });
                    setNewMemberName(""); showToast("已新增成員");
                  }} className="bg-green-600 text-white px-4 py-2 rounded-xl font-bold text-sm">新增成員</button>
                </div>
                <div className="flex gap-2">
                  <input className="flex-1 border p-2 rounded-xl text-sm outline-none focus:ring-2 focus:ring-blue-400" placeholder="新到貨日 (4/7)" value={newDateInput} onChange={e => setNewDateInput(e.target.value)}/>
                  <button onClick={async () => {
                    if(!newDateInput.includes('/')) return;
                    const updated = [...dates, newDateInput].sort();
                    await setDoc(doc(db, 'artifacts', appId, 'public', 'data', 'config', 'settings'), { activeDates: updated });
                    setNewDateInput(""); showToast("已新增日期");
                  }} className="bg-blue-600 text-white px-4 py-2 rounded-xl font-bold text-sm">新增日期</button>
                </div>
              </div>
            </div>
          </div>
        )}

        {/* 月份切換 */}
        <div className="flex items-center gap-2 bg-white p-2 rounded-2xl shadow-sm border border-slate-200 overflow-x-auto scrollbar-hide">
            {monthList.map(m => (
                <button 
                    key={m}
                    onClick={() => setCurrentMonth(m)}
                    className={`px-6 py-2 rounded-xl font-black text-xs transition-all whitespace-nowrap ${
                        currentMonth === m ? 'bg-green-600 text-white shadow-md' : 'text-slate-400 hover:bg-slate-100'
                    }`}
                >
                    {m}份
                </button>
            ))}
        </div>

        {/* 訂購矩陣 */}
        <div className="bg-white rounded-3xl shadow-xl overflow-hidden border border-slate-200">
          <div className="overflow-auto max-h-[60vh]">
            <table className="w-full text-left border-separate border-spacing-0">
              <thead className="sticky top-0 z-40">
                <tr className="bg-slate-900 text-white">
                  <th className="p-4 sticky left-0 z-50 bg-slate-900 border-r border-slate-800 text-center text-[10px] font-black w-24 sm:w-32 shadow-md">成員</th>
                  {currentDates.map(date => (
                    <th key={date} className="p-4 text-center min-w-[80px] border-b border-slate-800">
                      <div className="flex flex-col">
                        <span className="text-[8px] text-slate-500 font-normal">TUE</span>
                        <span className="text-sm font-black">{date}</span>
                      </div>
                    </th>
                  ))}
                  <th className="p-4 text-right border-b border-slate-800 bg-slate-950 text-green-400 text-[10px] font-black">應付金額</th>
                </tr>
              </thead>
              <tbody className="divide-y divide-slate-100">
                {stats.memberList.length === 0 ? (
                  <tr>
                    <td colSpan={currentDates.length + 2} className="p-20 text-center text-slate-400 text-sm">
                      <Database className="mx-auto mb-2 opacity-10" size={40} />
                      <p>資料庫目前為空，請從管理介面匯入資料。</p>
                    </td>
                  </tr>
                ) : (
                  stats.memberList.map(name => {
                    let totalBags = 0; let totalShip = 0;
                    dates.forEach(d => { if (orders[name]?.[d]) { totalBags++; totalShip += stats.shippingPerPerson[d]; } });

                    return (
                      <tr key={name} className="hover:bg-slate-50 transition-colors group">
                        <td className="p-3 font-bold text-slate-700 sticky left-0 bg-white z-20 border-r border-slate-100 flex items-center justify-between gap-1 shadow-sm">
                          <span className="truncate">{name}</span>
                          {isAdmin && (
                            <button onClick={async () => { if(window.confirm(`移除 ${name}？`)) await deleteDoc(doc(db, 'artifacts', appId, 'public', 'data', 'lettuce_orders', name)); }} className="text-red-400 p-1 rounded hover:bg-red-50"><Trash2 size={12}/></button>
                          )}
                        </td>
                        {currentDates.map(date => (
                          <td key={date} className="p-2 text-center">
                            <button
                              onClick={() => toggleOrder(name, date)}
                              className={`mx-auto w-10 h-10 rounded-xl flex items-center justify-center transition-all ${
                                orders[name]?.[date] ? 'bg-green-600 text-white shadow-lg scale-110' : 'bg-gray-50 text-gray-200'
                              }`}
                            >
                              {orders[name]?.[date] ? <CheckCircle2 size={24} /> : <Plus size={16} />}
                            </button>
                          </td>
                        ))}
                        <td className="p-4 text-right bg-slate-50/80 sticky right-0 z-10 font-mono font-black text-green-700 text-sm">
                          ${(totalBags * LETTUCE_PRICE + totalShip).toLocaleString()}
                        </td>
                      </tr>
                    );
                  })
                )}
              </tbody>
              <tfoot className="sticky bottom-0 z-40 bg-slate-900 text-white text-[10px]">
                <tr className="font-bold border-t border-slate-700">
                  <td className="p-4 sticky left-0 bg-slate-900 border-r border-slate-800">當週小計</td>
                  {currentDates.map(date => (
                    <td key={date} className="p-4 text-center">
                      <div className={`px-2 py-0.5 rounded-full inline-block ${stats.dailyCount[date] >= MAX_BAGS_PER_WEEK ? 'bg-red-500' : 'bg-green-600'}`}>
                        {stats.dailyCount[date]} / 12
                      </div>
                    </td>
                  ))}
                  <td className="p-4"></td>
                </tr>
                <tr className="bg-slate-800 text-blue-300">
                  <td className="p-2 sticky left-0 bg-slate-800 border-r border-slate-700 text-center">分擔運費</td>
                  {currentDates.map(date => (
                    <td key={date} className="p-2 text-center font-mono">
                      <div className="flex items-center justify-center gap-1 opacity-70">
                          <Truck size={10} /> ${stats.shippingPerPerson[date]}
                      </div>
                    </td>
                  ))}
                  <td className="p-2"></td>
                </tr>
              </tfoot>
            </table>
          </div>
        </div>

        {/* 結帳卡片區 */}
        <div className="space-y-4">
          <div className="flex items-center gap-2">
            <Users size={18} className="text-green-600"/>
            <h2 className="font-black text-lg text-slate-700">成員繳費總結</h2>
          </div>
          <div className="grid grid-cols-2 sm:grid-cols-2 lg:grid-cols-4 gap-3">
            {stats.memberList.map(name => {
              let bags = 0; let ship = 0;
              dates.forEach(d => { if (orders[name]?.[d]) { bags++; ship += stats.shippingPerPerson[d]; } });
              if (bags === 0 && !isAdmin) return null;

              return (
                <div key={name} className="bg-white p-4 rounded-3xl border border-slate-200 shadow-sm flex flex-col justify-between">
                  <div>
                    <div className="flex justify-between items-start mb-2">
                      <span className="font-black text-slate-800">{name}</span>
                      <span className="text-[10px] text-slate-400 font-bold">{bags} 包</span>
                    </div>
                    <div className="text-[11px] text-slate-500 space-y-1 pb-3 border-b border-dashed">
                      <div className="flex justify-between"><span>菜費:</span><span>${bags * LETTUCE_PRICE}</span></div>
                      <div className="flex justify-between"><span>運費:</span><span>${ship}</span></div>
                      <div className="flex justify-between pt-1 font-black text-green-600 text-lg"><span>總計:</span><span>${(bags * LETTUCE_PRICE + ship).toLocaleString()}</span></div>
                    </div>
                  </div>
                  
                  {isAdmin && (
                    <button 
                      onClick={() => handleSettle(name)}
                      disabled={bags === 0}
                      className={`mt-4 w-full py-2 rounded-xl text-[10px] font-black flex items-center justify-center gap-1 transition-all ${
                        bags === 0 ? 'bg-slate-50 text-slate-300' : 'bg-green-600 text-white hover:bg-green-700 shadow-lg active:scale-95'
                      }`}
                    >
                      <CreditCard size={12}/> 確認已匯款
                    </button>
                  )}
                </div>
              );
            })}
          </div>
        </div>
      </main>

      {/* 底部通知 */}
      {statusMsg && (
        <div className={`fixed bottom-8 left-1/2 -translate-x-1/2 z-[100] flex items-center gap-2 px-8 py-3 rounded-full shadow-2xl animate-in slide-in-from-bottom-8 ${
          statusMsg.type === 'error' ? 'bg-red-600 text-white' : 'bg-green-800 text-white'
        }`}>
          {statusMsg.type === 'error' ? <AlertCircle size={18}/> : <CheckCircle2 size={18} />}
          <span className="font-bold text-sm tracking-tight">{statusMsg.text}</span>
        </div>
      )}

      <footer className="text-center py-12 text-slate-300 text-[10px]">
        <p>生菜訂購分擔助手 v9.0 · 穩定連線版</p>
        <p>所有操作均自動同步雲端</p>
      </footer>
    </div>
  );
}
```
