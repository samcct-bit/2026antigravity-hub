const SlideDeck = () => {
  const [currentSlide, setCurrentSlide] = useState(0);
  const [showNotes, setShowNotes] = useState(false);
  const [menuOpen, setMenuOpen] = useState(false);
  const totalSlides = slidesData.length;

  // 鍵盤導覽
  useEffect(() => {
    const handleKeyDown = (e) => {
      if (e.key === 'ArrowRight' || e.key === ' ' || e.key === 'Enter') {
        nextSlide();
      } else if (e.key === 'ArrowLeft') {
        prevSlide();
      } else if (e.key === 'n') {
        setShowNotes(prev => !prev);
      }
    };
    window.addEventListener('keydown', handleKeyDown);
    return () => window.removeEventListener('keydown', handleKeyDown);
  }, [currentSlide]);

  const nextSlide = () => {
    if (currentSlide < totalSlides - 1) setCurrentSlide(curr => curr + 1);
  };

  const prevSlide = () => {
    if (currentSlide > 0) setCurrentSlide(curr => curr - 1);
  };

  const goToSlide = (index) => {
    setCurrentSlide(index);
    setMenuOpen(false);
  };

  const currentData = slidesData[currentSlide];

  // 渲染不同佈局
  const renderLayout = () => {
    switch (currentData.layout) {
      case 'title':
        return (
          <div className="flex flex-col items-center justify-center h-full text-center space-y-8 animate-fadeIn">
            <div className="p-8 rounded-full bg-amber-900/30 border-2 border-amber-500/50 shadow-[0_0_30px_rgba(245,158,11,0.3)]">
              <Coffee size={80} className="text-amber-400" />
            </div>
            <h1 className="text-5xl md:text-7xl font-bold text-amber-50 tracking-wide font-serif">
              {currentData.title}
            </h1>
            <div className="h-1 w-32 bg-gradient-to-r from-transparent via-amber-500 to-transparent"></div>
            <h2 className="text-2xl md:text-3xl text-stone-300 font-light max-w-2xl leading-relaxed">
              {currentData.subtitle}
            </h2>
            <p className="text-emerald-400 font-mono mt-8">{currentData.author}</p>
          </div>
        );

      case 'list':
        return (
          <div className="flex flex-col h-full justify-center px-8 md:px-20 animate-fadeIn">
            <h2 className="text-4xl font-bold text-amber-100 mb-12 border-l-8 border-amber-600 pl-6">
              {currentData.title}
            </h2>
            <div className="space-y-6">
              {currentData.items.map((item, idx) => (
                <div key={idx} className="flex items-center p-6 bg-stone-800/60 rounded-xl border border-stone-700/50 hover:bg-stone-800 hover:border-amber-500/50 transition-all duration-300 group">
                  <div className="mr-6 p-3 bg-stone-900 rounded-lg text-emerald-400 group-hover:scale-110 transition-transform">
                    {item.icon === 'flask' && <FlaskConical />}
                    {item.icon === 'atom' && <Atom />}
                    {item.icon === 'flame' && <Flame />}
                    {item.icon === 'sun' && <Lightbulb />}
                  </div>
                  <span className="text-xl md:text-2xl text-stone-200 font-medium">{item.text}</span>
                </div>
              ))}
            </div>
          </div>
        );

      case 'split':
        return (
          <div className="flex flex-col h-full px-8 md:px-16 pt-10 animate-fadeIn">
            <div className="mb-10">
              <span className="text-emerald-400 font-mono text-sm uppercase tracking-wider border border-emerald-400/30 px-2 py-1 rounded">Module</span>
              <h2 className="text-4xl font-bold text-amber-50 mt-4">{currentData.title}</h2>
              <p className="text-xl text-stone-400 mt-2">{currentData.subtitle}</p>
            </div>
            <div className="grid grid-cols-1 md:grid-cols-2 gap-8 h-full pb-10">
              <div className="bg-gradient-to-br from-stone-800 to-stone-900 p-8 rounded-2xl border border-amber-900/30 shadow-lg">
                <h3 className="text-2xl text-amber-400 mb-6 flex items-center font-serif">
                  <Coffee className="mr-3" /> {currentData.left.title}
                </h3>
                <ul className="space-y-4">
                  {currentData.left.content.map((t, i) => (
                    <li key={i} className="flex items-start text-stone-300 text-lg">
                      <span className="mr-2 text-amber-600">•</span>{t}
                    </li>
                  ))}
                </ul>
              </div>
              <div className="bg-gradient-to-br from-stone-800 to-stone-900 p-8 rounded-2xl border border-emerald-900/30 shadow-lg relative overflow-hidden">
                 <div className="absolute top-0 right-0 p-4 opacity-10">
                   <FlaskConical size={100} />
                 </div>
                <h3 className="text-2xl text-emerald-400 mb-6 flex items-center font-serif">
                  <Atom className="mr-3" /> {currentData.right.title}
                </h3>
                <ul className="space-y-4">
                  {currentData.right.content.map((t, i) => (
                    <li key={i} className="flex items-start text-stone-300 text-lg">
                      <span className="mr-2 text-emerald-600">→</span>{t}
                    </li>
                  ))}
                </ul>
              </div>
            </div>
          </div>
        );

      case 'cards':
        return (
          <div className="flex flex-col h-full px-8 md:px-16 pt-10 animate-fadeIn">
            <h2 className="text-4xl font-bold text-amber-50 mb-12 text-center">{currentData.title}</h2>
            <div className="grid grid-cols-1 md:grid-cols-2 gap-10 flex-grow pb-12">
              {currentData.cards.map((card, idx) => (
                <div key={idx} className="bg-stone-800 p-8 rounded-2xl border border-stone-600 hover:border-amber-500 transition-colors flex flex-col group">
                  <div className="flex items-center mb-6 text-amber-400 group-hover:text-amber-300">
                    {card.icon === 'thermometer' ? <Thermometer size={32} /> : <BrainCircuit size={32} />}
                    <h3 className="text-2xl font-bold ml-4">{card.title}</h3>
                  </div>
                  <p className="text-xl text-stone-200 mb-4">{card.desc}</p>
                  <div className="mt-auto pt-4 border-t border-stone-700">
                    <p className="text-emerald-300">{card.detail}</p>
                  </div>
                </div>
              ))}
            </div>
          </div>
        );

      case 'code':
        return (
          <div className="flex flex-col h-full px-8 md:px-16 pt-10 animate-fadeIn">
            <h2 className="text-3xl font-bold text-amber-50 mb-2">{currentData.title}</h2>
            <p className="text-stone-400 mb-6">{currentData.promptTitle}</p>
            <div className="bg-gray-900 p-6 rounded-lg border border-gray-700 shadow-inner overflow-auto flex-grow mb-10 font-mono text-sm md:text-base leading-relaxed text-green-400 relative">
              <div className="absolute top-2 right-4 text-xs text-gray-500 select-none">ChatGPT / Claude Prompt</div>
              <pre className="whitespace-pre-wrap">{currentData.code}</pre>
            </div>
          </div>
        );

      case 'table':
        return (
          <div className="flex flex-col h-full px-8 md:px-20 justify-center animate-fadeIn">
            <h2 className="text-4xl font-bold text-amber-100 mb-10 text-center">{currentData.title}</h2>
            <div className="overflow-hidden rounded-xl border border-stone-600 shadow-2xl">
              <table className="w-full text-left border-collapse">
                <thead className="bg-stone-800 text-amber-500">
                  <tr>
                    <th className="p-4 font-bold text-lg">豆款名稱</th>
                    <th className="p-4 font-bold text-lg">科學關鍵字</th>
                    <th className="p-4 font-bold text-lg">對應學科</th>
                  </tr>
                </thead>
                <tbody className="divide-y divide-stone-700 bg-stone-900/90">
                  {currentData.rows.map((row, idx) => (
                    <tr key={idx} className="hover:bg-stone-800 text-stone-300">
                      <td className="p-4 font-medium text-amber-100">{row.item}</td>
                      <td className="p-4">{row.key}</td>
                      <td className="p-4 text-emerald-400">{row.subject}</td>
                    </tr>
                  ))}
                </tbody>
              </table>
            </div>
          </div>
        );

      default:
        return null;
    }
  };

  return (
    <div className="bg-stone-950 min-h-screen w-full text-stone-200 font-sans overflow-hidden flex flex-col relative selection:bg-amber-500/30">
      
      {/* 導覽列 / 進度條 */}
      <div className="h-1.5 w-full bg-stone-900 z-50">
        <div 
          className="h-full bg-gradient-to-r from-amber-600 to-emerald-500 transition-all duration-500 ease-out"
          style={{ width: `${((currentSlide + 1) / totalSlides) * 100}%` }}
        />
      </div>

      {/* 主畫面 */}
      <div className="flex-grow relative overflow-y-auto overflow-x-hidden">
        {/* 背景裝飾 */}
        <div className="absolute top-[-10%] left-[-10%] w-[40%] h-[40%] bg-amber-900/10 rounded-full blur-[100px] pointer-events-none"></div>
        <div className="absolute bottom-[-10%] right-[-10%] w-[40%] h-[40%] bg-emerald-900/10 rounded-full blur-[100px] pointer-events-none"></div>

        <div className="max-w-6xl mx-auto h-full w-full">
           {renderLayout()}
        </div>
      </div>

      {/* 底部控制列 */}
      <div className="h-16 bg-stone-900/80 backdrop-blur border-t border-stone-800 flex items-center justify-between px-6 z-40">
        
        {/* 左側：目錄按鈕 */}
        <div className="flex items-center">
          <button 
            onClick={() => setMenuOpen(!menuOpen)}
            className="p-2 hover:bg-stone-800 rounded-lg text-stone-400 hover:text-amber-500 transition-colors flex items-center gap-2"
          >
            <Menu size={20} />
            <span className="text-sm font-mono hidden md:inline">Slide {currentSlide + 1} / {totalSlides}</span>
          </button>
        </div>

        {/* 中間：翻頁按鈕 */}
        <div className="flex items-center gap-4">
          <button 
            onClick={prevSlide} 
            disabled={currentSlide === 0}
            className={`p-2 rounded-full transition-all ${currentSlide === 0 ? 'text-stone-700 cursor-not-allowed' : 'text-amber-500 hover:bg-stone-800 hover:scale-110 active:scale-95'}`}
          >
            <ChevronLeft size={28} />
          </button>
          <button 
            onClick={nextSlide} 
            disabled={currentSlide === totalSlides - 1}
            className={`p-2 rounded-full transition-all ${currentSlide === totalSlides - 1 ? 'text-stone-700 cursor-not-allowed' : 'text-amber-500 hover:bg-stone-800 hover:scale-110 active:scale-95'}`}
          >
            <ChevronRight size={28} />
          </button>
        </div>

        {/* 右側：備忘稿開關 */}
        <div className="flex items-center">
          <button
             onClick={() => setShowNotes(!showNotes)}
             className={`flex items-center gap-2 px-3 py-1.5 rounded-lg text-sm font-medium transition-colors ${showNotes ? 'bg-emerald-900/50 text-emerald-400' : 'text-stone-500 hover:text-stone-300'}`}
          >
            <FileText size={18} />
            <span className="hidden md:inline">Speaker Notes</span>
          </button>
        </div>
      </div>

      {/* 彈出式選單 (目錄) */}
      {menuOpen && (
        <div className="absolute bottom-16 left-4 w-64 bg-stone-900 border border-stone-700 rounded-lg shadow-2xl overflow-hidden z-50 animate-slideUp max-h-[60vh] overflow-y-auto">
          <div className="p-3 bg-stone-800 border-b border-stone-700 flex justify-between items-center sticky top-0">
            <span className="font-bold text-stone-300">投影片目錄</span>
            <button onClick={() => setMenuOpen(false)}><X size={16} /></button>
          </div>
          <div className="p-2">
            {slidesData.map((slide, idx) => (
              <button
                key={slide.id}
                onClick={() => goToSlide(idx)}
                className={`w-full text-left px-3 py-2 rounded text-sm mb-1 truncate ${currentSlide === idx ? 'bg-amber-900/40 text-amber-400 border border-amber-500/30' : 'text-stone-400 hover:bg-stone-800'}`}
              >
                {idx + 1}. {slide.title}
              </button>
            ))}
          </div>
        </div>
      )}

      {/* 備忘稿 (Overlay) */}
      {showNotes && (
        <div className="absolute bottom-20 right-6 w-80 bg-stone-900/95 border border-emerald-500/30 rounded-lg shadow-2xl p-6 z-50 backdrop-blur animate-slideUp">
           <div className="flex items-center gap-2 mb-3 text-emerald-400 font-bold uppercase text-xs tracking-wider">
             <Lightbulb size={14} />
             <span>講師備忘 (Speaker Notes)</span>
           </div>
           <p className="text-stone-300 leading-relaxed text-sm">
             {currentData.note || '本頁無備忘稿。'}
           </p>
        </div>
      )}

      <style>{`
        @keyframes fadeIn {
          from { opacity: 0; transform: translateY(10px); }
          to { opacity: 1; transform: translateY(0); }
        }
        .animate-fadeIn {
          animation: fadeIn 0.6s cubic-bezier(0.16, 1, 0.3, 1) forwards;
        }
        @keyframes slideUp {
          from { opacity: 0; transform: translateY(10px); }
          to { opacity: 1; transform: translateY(0); }
        }
        .animate-slideUp {
          animation: slideUp 0.3s ease-out forwards;
        }
      `}</style>
    </div>
  );
};

export default SlideDeck;
