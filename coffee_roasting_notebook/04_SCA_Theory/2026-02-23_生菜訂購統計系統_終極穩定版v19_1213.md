## 📋 對話理論筆記
```react
import React, { useState, useEffect, useMemo } from 'react';
import { initializeApp } from 'firebase/app';
import { getAuth, signInAnonymously, onAuthStateChanged, signInWithCustomToken } from 'firebase/auth';
import { getFirestore, collection, doc, onSnapshot, setDoc, deleteDoc } from 'firebase/firestore';
import { 
  Users, 
  ShoppingCart, 
  Truck, 
  Calendar, 
  CheckCircle2, 
  AlertCircle,
  Plus,
  Trash2,
  Loader2,
  Database,
  Lock,
  Unlock,
  Settings,
  X,
  CreditCard,
  Wifi,
  WifiOff,
  Check
} from 'lucide-react';

// --- Firebase 初始化 ---
const firebaseConfig = JSON.parse(__firebase_config);
const app = initializeApp(firebaseConfig);
const auth = getAuth(app);
const db = getFirestore(app);
const appId = typeof __app_id !== 'undefined' ? __app_id : 'lettuce-master-v19';

// 費用常數
const LETTUCE_PRICE = 230;
const WEEKLY_SHIPPING = 300;
const MAX_BAGS_PER_WEEK = 12;

// 預設匯入資料
const INITIAL_IMPORT_DATA = {
  "金成": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "蕙敏": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "珮萱": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "淑芬": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "阿姿": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "嘉鎂": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "斯涵": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "明智": ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"],
  "翠花": ["2/24", "3/3", "3/10", "3/24"],
  "政君": ["3/3", "3/17", "3/31"],
  "睿婷": ["3/3", "3/17", "3/31"],
  "亭愷": ["3/3", "3/17", "3/31"],
  "俊明": ["2/24", "3/10", "3/24"],
  "惠偵": ["2/24", "3/10", "3/24"]
};
const INITIAL_DATES = ["2/24", "3/3", "3/10", "3/17", "3/24", "3/31"];

export default function App() {
  const [user, setUser] = useState(null);
  const [orders, setOrders] = useState({});
  const [dates, setDates] = useState([]);
  const [isReady, setIsReady] = useState(false);
  const [isAdmin, setIsAdmin] = useState(false);
  const [currentMonth, setCurrentMonth] = useState("");
  const [isProcessing, setIsProcessing] = useState(false);
  const [newMemberName, setNewMemberName] = useState("");
  const [newDateInput, setNewDateInput] = useState("");
  const [statusMsg, setStatusMsg] = useState(null);
  const [showAdminPanel, setShowAdminPanel] = useState(false);

  // 1. 登入身分驗證 (Rule 3)
  useEffect(() => {
    const initAuth = async () => {
      try {
        if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {
          await signInWithCustomToken(auth, __initial_auth_token);
        } else {
          await signInAnonymously(auth);
        }
      } catch (e) {
        console.error("Auth Fail", e);
      }
    };
    initAuth();
    onAuthStateChanged(auth, (u) => {
      if (u) {
        setUser(u);
        setIsReady(true);
      }
    });
  }, []);

  // 2. 資料即時同步 (Rule 1 扁平化路徑)
  useEffect(() => {
    if (!user) return;

    // 嚴格遵循扁平化集合路徑
    const settingsDoc = doc(db, 'artifacts', appId, 'public', 'data', 'settings', 'dates');
    const membersCol = collection(db, 'artifacts', appId, 'public', 'data', 'members');

    const unsubDates = onSnapshot(settingsDoc, (snap) => {
      if (snap.exists()) {
        const d = snap.data().list || [];
        setDates(d);
        if (d.length > 0 && !currentMonth) {
          setCurrentMonth(d[0].split('/')[0] + '月');
        }
      } else {
        // 靜默初始化
        setDoc(settingsDoc, { list: INITIAL_DATES });
      }
    });

    const unsubOrders = onSnapshot(membersCol, (snap) => {
      const o = {};
      snap.docs.forEach(d => { o[d.id] = d.data().orders || {}; });
      setOrders(o);
    });

    return () => { unsubDates(); unsubOrders(); };
  }, [user, currentMonth]);

  // 3. 核心統計與計算
  const stats = useMemo(() => {
    const dailyCount = {};
    const shippingPerPerson = {};
    const memberList = Object.keys(orders).sort();
    const months = new Set();
    
    dates.forEach(d => {
      months.add(d.split('/')[0] + '月');
      let count = 0;
      memberList.forEach(m => { if (orders[m]?.[d]) count++; });
      dailyCount[d] = count;
      if (count >= 12) shippingPerPerson[d] = 25;
      else if (count === 11) shippingPerPerson[d] = 27;
      else if (count === 10) shippingPerPerson[d] = 30;
      else if (count > 0) shippingPerPerson[d] = Math.round(WEEKLY_SHIPPING / count);
      else shippingPerPerson[d] = 0;
    });

    return { 
      dailyCount, 
      shippingPerPerson, 
      memberList, 
      monthList: Array.from(months).sort((a,b) => parseInt(a) - parseInt(b)) 
    };
  }, [orders, dates]);

  const currentDates = useMemo(() => {
    return dates.filter(d => d.split('/')[0] + '月' === currentMonth);
  }, [dates, currentMonth]);

  // 4. 管理與資料操作函數
  const showToast = (text, type = 'success') => {
    setStatusMsg({ text, type });
    setTimeout(() => setStatusMsg(null), 3500);
  };

  const handleImport = async () => {
    if (!user || isProcessing) return;
    setIsProcessing(true);
    try {
      // 匯入日期
      const settingsDoc = doc(db, 'artifacts', appId, 'public', 'data', 'settings', 'dates');
      await setDoc(settingsDoc, { list: INITIAL_DATES });

      // 匯入成員
      for (const [name, dateArr] of Object.entries(INITIAL_IMPORT_DATA)) {
        const mRef = doc(db, 'artifacts', appId, 'public', 'data', 'members', name);
        const o = {}; dateArr.forEach(d => o[d] = true);
        await setDoc(mRef, { orders: o });
      }
      showToast("原始資料已匯入！");
      setCurrentMonth("3月");
    } catch (e) {
      showToast(`寫入失敗: ${e.message}`, "error");
    } finally { setIsProcessing(false); }
  };

  const handleResetSettle = async (name) => {
    if (!user || isProcessing) return;
    if (!window.confirm(`確認「${name}」已匯款？費用將歸零。`)) return;
    setIsProcessing(true);
    try {
      const mRef = doc(db, 'artifacts', appId, 'public', 'data', 'members', name);
      // 強制覆寫為空物件，確保金額一定歸零
      await setDoc(mRef, { orders: {} });
      showToast(`${name} 結帳成功，金額已歸零`);
    } catch (e) {
      showToast(`歸零失敗: ${e.message}`, "error");
    } finally { setIsProcessing(false); }
  };

  const handleDeleteMember = async (name) => {
    if (!user || isProcessing) return;
    if (!window.confirm(`確定要完全移除「${name}」嗎？`)) return;
    setIsProcessing(true);
    try {
      const mRef = doc(db, 'artifacts', appId, 'public', 'data', 'members', name);
      await deleteDoc(mRef);
      showToast(`已成功移除成員：${name}`);
    } catch (e) {
      showToast(`移除失敗: ${e.message}`, "error");
    } finally { setIsProcessing(false); }
  };

  const toggleOrder = async (name, date) => {
    if (isAdmin && showAdminPanel) return;
    const currentOrders = orders[name] || {};
    const isOrdered = !!currentOrders[date];

    if (!isOrdered && stats.dailyCount[date] >= MAX_BAGS_PER_WEEK) {
      showToast("該週上限 12 包已滿", "error"); return;
    }

    try {
      const mRef = doc(db, 'artifacts', appId, 'public', 'data', 'members', name);
      const updated = { ...currentOrders, [date]: !isOrdered };
      await setDoc(mRef, { orders: updated });
    } catch (e) {
      showToast("點選失敗，請稍候再試", "error");
    }
  };

  if (!isReady) return (
    <div className="h-screen flex flex-col items-center justify-center bg-green-50">
      <Loader2 className="animate-spin text-green-600 mb-2" size={32}/>
      <p className="text-green-800 font-bold tracking-widest">系統安全連線中...</p>
    </div>
  );

  return (
    <div className="min-h-screen bg-slate-50 pb-20 font-sans text-slate-800 antialiased overflow-x-hidden">
      {/* 頂部導航 */}
      <nav className="bg-green-700 text-white shadow-xl sticky top-0 z-[100] px-4 py-3">
        <div className="max-w-7xl mx-auto flex items-center justify-between gap-3">
          <div className="flex items-center gap-2 overflow-hidden">
            <ShoppingCart className="w-5 h-5 flex-shrink-0" />
            <h1 className="text-sm sm:text-lg font-black tracking-tight truncate">生菜訂購系統 v19</h1>
          </div>
          
          <div className="flex items-center gap-2 flex-shrink-0">
            {user ? <Wifi size={14} className="text-green-300" /> : <WifiOff size={14} className="text-red-400" />}
            <button 
              onClick={() => setIsAdmin(!isAdmin)}
              className={`px-3 py-1.5 rounded-full text-[10px] font-bold flex items-center gap-1 transition-all ${
                isAdmin ? 'bg-red-500 text-white shadow-lg' : 'bg-white/10'
              }`}
            >
              {isAdmin ? <Unlock size={12}/> : <Lock size={12}/>}
              {isAdmin ? '管理中' : '檢視'}
            </button>
            {isAdmin && (
              <button onClick={() => setShowAdminPanel(!showAdminPanel)} className="p-1.5 bg-white/10 rounded-lg active:scale-95">
                <Settings size={18} />
              </button>
            )}
          </div>
        </div>
      </nav>

      <main className="max-w-7xl mx-auto px-2 sm:px-4 py-6 space-y-6">
        
        {/* 管理控制台 */}
        {isAdmin && showAdminPanel && (
          <div className="bg-white p-6 rounded-3xl shadow-2xl border-2 border-red-50 animate-in slide-in-from-top-4">
            <div className="flex items-center justify-between mb-6 border-b pb-4">
              <h3 className="text-red-600 font-black flex items-center gap-2">
                <Settings size={20} /> 管理控制台
              </h3>
              <button onClick={() => setShowAdminPanel(false)}><X size={20} className="text-slate-400"/></button>
            </div>
            
            <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
              <div className="bg-blue-50 p-6 rounded-2xl border border-blue-100 flex flex-col justify-center text-center">
                <button 
                  onClick={handleImport}
                  disabled={isProcessing}
                  className="w-full bg-blue-600 hover:bg-blue-700 text-white py-4 rounded-xl font-bold flex items-center justify-center gap-2 shadow-lg disabled:opacity-50 active:scale-95 transition-all"
                >
                  {isProcessing ? <Loader2 className="animate-spin" size={20}/> : <Database size={20} />}
                  匯入原始 14 位成員訂單
                </button>
                <p className="mt-3 text-[10px] text-blue-500 font-bold uppercase tracking-tight">※ 僅供歷史資料初始化</p>
              </div>

              <div className="space-y-4">
                <div className="flex gap-2">
                  <input className="flex-1 border-2 border-slate-100 p-3 rounded-xl text-sm" placeholder="成員姓名" value={newMemberName} onChange={e => setNewMemberName(e.target.value)}/>
                  <button onClick={async () => {
                    if(!newMemberName) return;
                    await setDoc(doc(db, 'artifacts', appId, 'public', 'data', 'members', newMemberName), { orders: {} });
                    setNewMemberName(""); showToast("已新增成員");
                  }} className="bg-green-600 text-white px-5 rounded-xl font-bold text-xs whitespace-nowrap active:scale-95">新增成員</button>
                </div>
                <div className="flex gap-2">
                  <input className="flex-1 border-2 border-slate-100 p-3 rounded-xl text-sm" placeholder="日期 (如 4/7)" value={newDateInput} onChange={e => setNewDateInput(e.target.value)}/>
                  <button onClick={async () => {
                    if(!newDateInput.includes('/')) return;
                    const updated = [...dates, newDateInput].sort();
                    await setDoc(doc(db, 'artifacts', appId, 'public', 'data', 'settings', 'dates'), { list: updated });
                    setNewDateInput(""); showToast("日期已新增");
                  }} className="bg-blue-600 text-white px-5 rounded-xl font-bold text-xs whitespace-nowrap active:scale-95">新增日期</button>
                </div>
              </div>
            </div>
          </div>
        )}

        {/* 月份選單 */}
        <div className="flex items-center gap-2 bg-white p-2 rounded-2xl shadow-sm border border-slate-200 overflow-x-auto no-scrollbar">
            {stats.monthList.map(m => (
                <button 
                    key={m}
                    onClick={() => setCurrentMonth(m)}
                    className={`px-6 py-2 rounded-xl font-black text-xs transition-all whitespace-nowrap ${
                        currentMonth === m ? 'bg-green-600 text-white shadow-md' : 'text-slate-400 hover:bg-slate-100'
                    }`}
                >
                    {m}份
                </button>
            ))}
        </div>

        {/* 訂購表格 (三重固定：日期頭、左姓名、右總額) */}
        <div className="bg-white rounded-3xl shadow-xl overflow-hidden border border-slate-200 relative">
          <div className="overflow-auto max-h-[60vh] scroll-smooth relative">
            <table className="w-full text-left border-separate border-spacing-0">
              <thead className="sticky top-0 z-[60] bg-slate-900 text-white shadow-md">
                <tr>
                  <th className="p-4 sticky left-0 z-[70] bg-slate-900 border-r border-slate-800 text-center text-[10px] font-black w-24 sm:w-32">姓名</th>
                  {currentDates.map(date => (
                    <th key={date} className="p-4 text-center min-w-[85px] border-b border-slate-800">
                      <div className="flex flex-col">
                        <span className="text-[8px] text-slate-500 font-normal uppercase">Tue</span>
                        <span className="text-sm font-black tracking-tighter">{date}</span>
                      </div>
                    </th>
                  ))}
                  <th className="p-4 text-right border-b border-slate-800 bg-slate-950 text-green-400 text-[10px] font-black min-w-[100px] sticky right-0 z-[70] shadow-[-2px_0_5px_rgba(0,0,0,0.3)]">應付總額</th>
                </tr>
              </thead>
              <tbody className="divide-y divide-slate-100">
                {stats.memberList.length === 0 ? (
                  <tr><td colSpan={currentDates.length + 2} className="p-20 text-center text-slate-400">目前無資料，請點擊一鍵匯入。</td></tr>
                ) : (
                  stats.memberList.map(name => {
                    let totalBags = 0; let totalShip = 0;
                    dates.forEach(d => { if (orders[name]?.[d]) { totalBags++; totalShip += stats.shippingPerPerson[d]; } });

                    return (
                      <tr key={name} className="hover:bg-green-50 transition-colors group">
                        <td className="p-3 font-bold text-slate-700 sticky left-0 z-[40] bg-white border-r border-slate-100 flex items-center justify-between gap-1 shadow-sm">
                          <span className="truncate">{name}</span>
                          {isAdmin && (
                            <button onClick={() => handleDeleteMember(name)} className="text-red-300 hover:text-red-500 p-1 hover:bg-red-50 rounded-lg transition-all"><Trash2 size={12}/></button>
                          )}
                        </td>
                        {currentDates.map(date => (
                          <td key={date} className="p-2 text-center">
                            <button
                              onClick={() => toggleOrder(name, date)}
                              className={`mx-auto w-10 h-10 rounded-xl flex items-center justify-center transition-all ${
                                orders[name]?.[date] ? 'bg-green-600 text-white shadow-lg scale-110' : 'bg-slate-50 text-slate-200 active:scale-95'
                              }`}
                            >
                              {orders[name]?.[date] ? <CheckCircle2 size={24} strokeWidth={3} /> : <Plus size={18} />}
                            </button>
                          </td>
                        ))}
                        <td className="p-4 text-right bg-slate-50 sticky right-0 z-[30] font-mono font-black text-green-700 text-sm border-l border-slate-100 shadow-[-2px_0_5px_rgba(0,0,0,0.02)]">
                          ${(totalBags * LETTUCE_PRICE + totalShip).toLocaleString()}
                        </td>
                      </tr>
                    );
                  })
                )}
              </tbody>
              <tfoot className="sticky bottom-0 z-[60] bg-slate-900 text-white text-[10px] border-t border-slate-700 shadow-[0_-2px_10px_rgba(0,0,0,0.1)]">
                <tr className="font-bold">
                  <td className="p-4 sticky left-0 z-[70] bg-slate-900 border-r border-slate-800 uppercase tracking-widest text-slate-500">Total</td>
                  {currentDates.map(date => (
                    <td key={date} className="p-4 text-center">
                      <div className={`px-2 py-0.5 rounded-full inline-block ${stats.dailyCount[date] >= MAX_BAGS_PER_WEEK ? 'bg-red-500' : 'bg-green-600'}`}>
                        {stats.dailyCount[date]} / 12
                      </div>
                    </td>
                  ))}
                  <td className="p-4 text-right sticky right-0 z-[70] bg-slate-950 text-slate-400">BAGS</td>
                </tr>
                <tr className="bg-slate-800 text-blue-300">
                  <td className="p-2 sticky left-0 z-[70] bg-slate-800 border-r border-slate-700 text-center uppercase text-[7px] tracking-tight">Ship Fee</td>
                  {currentDates.map(date => (
                    <td key={date} className="p-2 text-center font-mono opacity-80">
                      ${stats.shippingPerPerson[date]}
                    </td>
                  ))}
                  <td className="p-2 sticky right-0 z-[70] bg-slate-800"></td>
                </tr>
              </tfoot>
            </table>
          </div>
        </div>

        {/* 結帳繳費統計區 */}
        <div className="space-y-4">
          <div className="flex items-center gap-2 px-1">
            <Users size={18} className="text-green-600"/>
            <h2 className="font-black text-lg text-slate-700">成員繳費明細總覽</h2>
          </div>
          <div className="grid grid-cols-2 lg:grid-cols-4 gap-3">
            {stats.memberList.map(name => {
              let bags = 0; let ship = 0;
              dates.forEach(d => { if (orders[name]?.[d]) { bags++; ship += stats.shippingPerPerson[d]; } });
              if (bags === 0 && !isAdmin) return null;

              return (
                <div key={name} className="bg-white p-5 rounded-3xl border border-slate-200 shadow-sm flex flex-col justify-between hover:shadow-xl transition-all relative overflow-hidden group">
                  <div className="absolute top-0 left-0 w-1.5 h-full bg-green-500 transition-all group-hover:w-2"></div>
                  <div>
                    <div className="flex justify-between items-start mb-3">
                      <span className="font-black text-slate-800 text-base">{name}</span>
                      <span className="text-[10px] bg-slate-50 px-2 py-0.5 rounded text-slate-500 font-bold tracking-tight">{bags} 包</span>
                    </div>
                    <div className="text-[11px] text-slate-500 space-y-1 pb-4 border-b border-dashed border-slate-100">
                      <div className="flex justify-between"><span>生菜費用:</span><span>${bags * LETTUCE_PRICE}</span></div>
                      <div className="flex justify-between"><span>累計運費:</span><span>${ship}</span></div>
                      <div className="flex justify-between pt-2 font-black text-green-600 text-xl tracking-tighter"><span>結算總額:</span><span>${(bags * LETTUCE_PRICE + ship).toLocaleString()}</span></div>
                    </div>
                  </div>
                  
                  {isAdmin && (
                    <button 
                      onClick={() => handleResetSettle(name)}
                      disabled={bags === 0 || isProcessing}
                      className={`mt-4 w-full py-2.5 rounded-xl text-[10px] font-black flex items-center justify-center gap-2 transition-all shadow-md ${
                        bags === 0 ? 'bg-slate-50 text-slate-300' : 'bg-green-600 text-white hover:bg-green-700 hover:scale-[1.02] active:scale-95'
                      }`}
                    >
                      <Check size={14}/> 確認已匯款並歸零
                    </button>
                  )}
                </div>
              );
            })}
          </div>
        </div>
      </main>

      {/* 底部提示彈窗 */}
      {statusMsg && (
        <div className={`fixed bottom-8 left-1/2 -translate-x-1/2 z-[200] flex items-center gap-2 px-10 py-4 rounded-full shadow-2xl animate-in slide-in-from-bottom-10 transition-all ${
          statusMsg.type === 'error' ? 'bg-red-600 text-white' : 'bg-green-800 text-white'
        }`}>
          {statusMsg.type === 'error' ? <AlertCircle size={20}/> : <CheckCircle2 size={20} />}
          <span className="font-black text-sm tracking-tight">{statusMsg.text}</span>
        </div>
      )}
    </div>
  );
}
```
