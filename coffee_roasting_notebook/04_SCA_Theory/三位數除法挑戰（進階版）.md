<script>
    let score = 0;
    let currentStep = 1;
    const totalSteps = 5;
    let timeLeft = 30;
    let timerInterval;
    let problem = { dividend: 0, divisor: 0, q: 0, r: 0 };

    function startGame() {
        document.getElementById('start-screen').classList.add('hidden');
        document.getElementById('game-board').classList.remove('hidden');
        generateQuestion();
    }

    function generateQuestion() {
        // 產生 100-999 的被除數
        const dividend = Math.floor(Math.random() * 900) + 100;
        // 產生 2-9 的除數
        const divisor = Math.floor(Math.random() * 8) + 2;
        
        const q = Math.floor(dividend / divisor);
        const r = dividend % divisor;

        problem = { dividend, divisor, q, r };
        
        document.getElementById('q-num').innerText = currentStep;
        document.getElementById('question-text').innerText = `${dividend} ÷ ${divisor}`;
        document.getElementById('input-q').value = '';
        document.getElementById('input-r').value = '';
        document.getElementById('input-q').focus();
        
        updateProgress();
        startTimer();
    }

    function startTimer() {
        clearInterval(timerInterval);
        timeLeft = 30;
        updateTimerUI();
        
        timerInterval = setInterval(() => {
            timeLeft--;
            updateTimerUI();
            if (timeLeft <= 0) {
                clearInterval(timerInterval);
                checkAnswer(true); // 時間到
            }
        }, 1000);
    }

    function updateTimerUI() {
        const timerEl = document.getElementById('timer-text');
        timerEl.innerText = `剩餘時間：${timeLeft}s`;
        if (timeLeft <= 10) timerEl.style.color = "#ef4444";
        else timerEl.style.color = "#1e293b";
    }

    function checkAnswer(isTimeout = false) {
        clearInterval(timerInterval);
        
        const userQ = parseInt(document.getElementById('input-q').value);
        const userR = parseInt(document.getElementById('input-r').value);
        
        // 如果沒填寫且不是時間到，不允許送出
        if (!isTimeout && isNaN(userQ)) {
            document.getElementById('input-q').classList.add('animate-shake');
            setTimeout(() => document.getElementById('input-q').classList.remove('animate-shake'), 500);
            startTimer(); // 恢復計時
            return;
        }

        const isCorrect = (userQ === problem.q && (isNaN(userR) ? 0 : userR) === problem.r);
        showFeedback(isCorrect, isTimeout);
    }

    function showFeedback(isCorrect, isTimeout) {
        const fb = document.getElementById('feedback');
        const icon = document.getElementById('fb-icon');
        const title = document.getElementById('fb-title');
        const desc = document.getElementById('fb-desc');
        
        fb.style.display = 'flex';
        
        if (isCorrect) {
            score += 20;
            document.getElementById('current-score').innerText = score;
            icon.innerText = "⭐";
            icon.className = "fb-icon animate-bounce";
            title.innerText = "答對了！太強了！";
            title.style.color = "var(--success)";
            desc.innerHTML = `計算正確：${problem.dividend} ÷ ${problem.divisor} = <b>${problem.q}</b> ⋯ <b>${problem.r}</b>`;
        } else {
            icon.innerText = isTimeout ? "⏰" : "❌";
            icon.className = "fb-icon animate-shake";
            title.innerText = isTimeout ? "時間到了！" : "再接再厲！";
            title.style.color = "var(--danger)";
            
            desc.innerHTML = `
                正確答案是：<span class="math-highlight">${problem.q} ⋯ ${problem.r}</span><br><br>
                <small>💡 小撇步驗算：<br>
                ${problem.divisor} (除數) × ${problem.q} (商) + ${problem.r} (餘數)<br>
                = ${problem.divisor * problem.q} + ${problem.r} = <b>${problem.dividend}</b></small>
            `;
        }
    }

    function nextQuestion() {
        document.getElementById('feedback').style.display = 'none';
        if (currentStep < totalSteps) {
            currentStep++;
            generateQuestion();
        } else {
            showFinalResult();
        }
    }

    function updateProgress() {
        const p = ((currentStep - 1) / totalSteps) * 100;
        document.getElementById('progress-fill').style.width = `${p}%`;
    }

    function showFinalResult() {
        document.getElementById('game-board').classList.add('hidden');
        document.getElementById('result-screen').classList.remove('hidden');
        document.getElementById('final-score-val').innerText = score;
        
        let comment = "";
        let icon = "";
        if (score >= 100) {
            icon = "👑";
            comment = "傳說級除法大師！所有題目都難不倒你！";
        } else if (score >= 60) {
            icon = "🥈";
            comment = "表現優異！離滿分只差一點點了！";
        } else {
            icon = "🧱";
            comment = "練習是通往成功的路，再挑戰一次吧！";
        }
        document.getElementById('final-rank-icon').innerText = icon;
        document.getElementById('final-comment').innerText = comment;
    }
</script>
