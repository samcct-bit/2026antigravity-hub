    <script>
        const allQuestions = [
            { q: "王獻之寫____字，跟人一樣高。", a: "的", e: "「字」是名詞，前面用「的」來修飾。" },
            { q: "毛筆被獻之牢牢____抓在手裡。", a: "地", e: "「抓」是動詞，前面用「地」描述動作的狀態。" },
            { q: "你寫字時，筆握____牢。", a: "得", e: "「握」是動詞，「牢」是補充說明的結果，中間用「得」。" },
            { q: "大家讚美他寫的字和父親一樣____好。", a: "得", e: "「寫」的好，表示動作結果的程度，故用「得」。" },
            { q: "美麗____風景就在鹿谷。", a: "的", e: "「風景」是名詞，前面用「的」連接形容詞。" },
            { q: "他興奮____跳了起來。", a: "地", e: "「跳」是動作，前面的「興奮」是形容動作如何進行，故用「地」。" },
            { q: "天黑____看不見路。", a: "得", e: "形容「黑」到了一種程度，故用「得」。" },
            { q: "這是一本很有趣____書。", a: "的", e: "「書」是名詞，前面用「的」。" },
            { q: "老師大聲____責備頑皮的學生。", a: "地", e: "「責備」是動詞，修飾動詞的副詞後面用「地」。" },
            { q: "這件事辦____非常出色。", a: "得", e: "「辦」是動詞，「出色」是補充結果，中間用「得」。" }
        ];

        let currentQuestions = [];
        let currentIndex = 0;
        let score = 0;
        let answered = false;

        function startGame() {
            // Shuffle and pick 10 questions (or all if less than 10)
            currentQuestions = [...allQuestions].sort(() => Math.random() - 0.5);
            currentIndex = 0;
            score = 0;
            
            document.getElementById('start-screen').classList.add('hidden');
            document.getElementById('result-screen').classList.add('hidden');
            document.getElementById('quiz-screen').classList.remove('hidden');
            
            updateUI();
        }

        function updateUI() {
            answered = false;
            const q = currentQuestions[currentIndex];
            document.getElementById('q-text').innerText = (currentIndex + 1) + ". " + q.q;
            document.getElementById('q-count').innerText = `題目 ${currentIndex + 1} / ${currentQuestions.length}`;
            document.getElementById('score-count').innerText = `得分: ${score}`;
            document.getElementById('progress').style.width = `${(currentIndex / currentQuestions.length) * 100}%`;
            
            document.getElementById('feedback').classList.add('hidden');
            
            // Re-enable option buttons
            const btns = document.querySelectorAll('.option-btn');
            btns.forEach(b => {
                b.disabled = false;
                b.classList.remove('border-emerald-500', 'bg-emerald-50', 'border-red-500', 'bg-red-50');
            });
        }

        function handleAnswer(choice) {
            if (answered) return;
            answered = true;
            
            const q = currentQuestions[currentIndex];
            const feedback = document.getElementById('feedback');
            const feedbackBox = document.getElementById('feedback-box');
            const feedbackMsg = document.getElementById('feedback-msg');
            const explanation = document.getElementById('explanation');
            
            // Disable all buttons
            const btns = document.querySelectorAll('.option-btn');
            btns.forEach(b => {
                b.disabled = true;
                if (b.innerText.includes(q.a)) {
                    b.classList.add('border-emerald-500', 'bg-emerald-50');
                }
            });

            if (choice === q.a) {
                score += 10;
                feedbackMsg.innerText = "✅ 答對了！";
                feedbackBox.className = "p-4 rounded-xl mb-4 bg-emerald-100 text-emerald-800";
            } else {
                feedbackMsg.innerText = "❌ 答錯囉！";
                feedbackBox.className = "p-4 rounded-xl mb-4 bg-red-100 text-red-800";
                // Mark the selected wrong button
                btns.forEach(b => {
                    if (b.innerText.includes(choice)) {
                        b.classList.add('border-red-500', 'bg-red-50');
                    }
                });
            }

            explanation.innerText = q.e;
            feedback.classList.remove('hidden');
            document.getElementById('score-count').innerText = `得分: ${score}`;
        }

        function nextQuestion() {
            currentIndex++;
            if (currentIndex < currentQuestions.length) {
                updateUI();
            } else {
                showResult();
            }
        }

        function showResult() {
            document.getElementById('quiz-screen').classList.add('hidden');
            document.getElementById('result-screen').classList.remove('hidden');
            document.getElementById('progress').style.width = "100%";
            
            document.getElementById('final-score').innerText = score;
            
            const msg = document.getElementById('final-msg');
            if (score === 100) {
                msg.innerText = "超完美！你已經是國文大師了！";
            } else if (score >= 70) {
                msg.innerText = "很棒！用法掌握得相當不錯。";
            } else {
                msg.innerText = "再接再厲，多練習幾次就會進步喔！";
            }
        }
    </script>
